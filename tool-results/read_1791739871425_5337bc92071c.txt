     3→/**
     4→ * useOeChat — lógica del chat premium con OE AI LLM.
     5→ *
     6→ * - Comunica con POST /api/chat (SSE: meta → stage/search → delta → done | error).
     7→ * - El backend entrega la respuesta completa (streaming de tokens no verificado);
     8→ *   la revelación progresiva es un efecto de presentación local (máquina de
     9→ *   escribir adaptativa) para una experiencia fluida, sin fingir datos.
    10→ * - Gestiona historial multi-conversación vía /api/chat/sessions y
    11→ *   /api/chat/history; herramientas: imágenes (/api/tools/image), búsqueda web
    12→ *   real con fuentes, edición de mensajes (recorte), renombrar y exportar.
    13→ */

    27→export interface ChatMsgMeta {
    28→  coreLabel?: string | null
    29→  routeNote?: string | null
    30→  providerReportedModel?: string | null
    31→  latencyMs?: number | null
    32→  mode?: 'instant' | 'reflective'
    33→  webSearchUsed?: boolean
    34→  /** true: el turno analizó una imagen adjuntada (visión). */
    35→  visionUsed?: boolean
    36→  /** true: los deltas llegan incrementales en tiempo real (streaming verificado). */
    37→  streaming?: boolean
    38→}

    40→export interface ChatMsg {
    41→  id: string
    42→  role: 'user' | 'assistant'
    43→  /** Texto visible (durante la revelación es un prefijo de fullContent). */
    44→  content: string
    45→  /** Texto completo recibido del proveedor. */
    46→  fullContent: string
    47→  /** true cuando la revelación terminó (render markdown). */
    48→  complete: boolean
    49→  meta?: ChatMsgMeta
    50→  /** Tipo: texto normal o imagen generada por la herramienta del núcleo. */
    51→  kind?: 'text' | 'image'
    52→  /** Para kind="image": URL del activo servido por /api/assets/[id]. */
    53→  assetUrl?: string | null
    54→  /** Placeholder mientras se genera la imagen. */
    55→  pendingImage?: boolean
    56→  /** Fuentes web (búsqueda real) adjuntas a la respuesta. */
    57→  sources?: WebSource[]
    58→  /** Aviso honesto si la búsqueda web no estuvo disponible. */
    59→  searchNote?: string | null
    60→  /** Imagen adjuntada por el usuario (visión): /api/assets/[id]. */
    61→  attachmentUrl?: string | null
    62→}

    82→const ACTIVE_KEY = 'oe-ai-active-session'

    84→function newId(): string {
    85→  if (typeof crypto !== 'undefined' && 'randomUUID' in crypto) return crypto.randomUUID()
    86→  return `${Date.now()}-${Math.random().toString(36).slice(2)}-${Math.random().toString(36).slice(2)}`
    87→}

    89→export function useOeChat() {
    90→  const [conversations, setConversations] = useState<ConversationSummary[]>([])
    91→  const [activeId, setActiveId] = useState<string | null>(null)
    92→  const [messages, setMessages] = useState<ChatMsg[]>([])
    93→  const [phase, setPhase] = useState<ChatPhase>('idle')
    94→  const [error, setError] = useState<ChatError | null>(null)
    95→  const [routeNote, setRouteNote] = useState<string | null>(null)
    96→  const [focusModel, setFocusModel] = useState<string | null>(null)
    97→  const [loadingHistory, setLoadingHistory] = useState(false)
    98→  const [lastUserMsg, setLastUserMsg] = useState<string | null>(null)
    99→  const [webSearching, setWebSearching] = useState(false)
   100→  const [imagePending, setImagePending] = useState(false)
   101→  const [guestQuota, setGuestQuota] = useState<GuestQuotaState | null>(null)
   102→  const [guestBlocked, setGuestBlocked] = useState(false)
   103→
   104→  const abortRef = useRef<AbortController | null>(null)
   105→  const rafRef = useRef<number | null>(null)
   106→  const activeIdRef = useRef<string | null>(null)
   107→  const pendingRef = useRef(false)
   108→  const lastOptsRef = useRef<{ webSearch?: boolean; attachment?: { assetId: string; previewUrl: string } }>({})
   109→  const auth = useAuth()
   110→
   111→  useEffect(() => {
   112→    activeIdRef.current = activeId
   113→  }, [activeId])
   114→
   115→  /** Estado de la cuota (GET /api/chat/quota): nula si hay sesión iniciada.
   116→   *  IMPORTANTE: envía el token de Supabase cuando existe — sin él el endpoint
   117→   *  sigue viendo la cookie de invitado agotada y volvía a bloquear a usuarios
   118→   *  que YA habían iniciado sesión (la respuesta asíncrona reactivaba la
   119→   *  compuerta después de que el efecto la levantaba). */
   120→  const refreshQuota = useCallback(async () => {
   121→    try {
   122→      const token = await auth.getAccessToken().catch(() => null)
   123→      const res = await fetch('/api/chat/quota', {
   124→        cache: 'no-store',
   125→        ...(token ? { headers: { Authorization: `Bearer ${token}` } } : {}),
   126→      })
   127→      if (!res.ok) return
   128→      const data = (await res.json()) as {
   129→        authenticated: boolean
   130→        used: number | null
   131→        limit: number | null
   132→        remaining: number | null
   133→        exhausted?: boolean
   134→      }
   135→      if (data.authenticated) {
   136→        setGuestQuota(null)
   137→        setGuestBlocked(false)
   138→        return
   139→      }
   140→      const state: GuestQuotaState = {
   141→        used: data.used ?? 0,
   142→        limit: data.limit ?? 5,
   143→        remaining: data.remaining ?? 0,
   144→        exhausted: Boolean(data.exhausted),
   145→      }
   146→      setGuestQuota(state)
   147→      // Con sesión iniciada el servidor ignora la cuota de invitado: nunca bloquear.
   148→      if (state.exhausted && !auth.user) setGuestBlocked(true)
   149→    } catch {
   150→      // sin estado de cuota visible; el servidor seguirá aplicándola
   151→    }
   152→  }, [auth])
   153→
   154→  // Cuota al arrancar y al cambiar la sesión (login/cierre): tras iniciar sesión
   155→  // la compuerta se levanta (usuarios verificados: chats ilimitados).
   156→  useEffect(() => {
   157→    void refreshQuota()
   158→    if (auth.user) setGuestBlocked(false)
   159→  }, [auth.user, refreshQuota])
   160→
   161→  // Sincroniza el enfoque de red (chatModel del store) con el estado local,
   162→  // también cuando cambia estando montado (p. ej. "Usar en el chat" desde Conexiones).
   163→  useEffect(() => {
   164→    const syncFocus = (model: string) => {
   165→      setFocusModel(model && model !== 'auto' ? model : null)
   166→    }
   167→    syncFocus(useUiStore.getState().chatModel)
   168→    return useUiStore.subscribe((state) => syncFocus(state.chatModel))
   169→  }, [])
   170→
   171→  /** Lista de conversaciones para sidebar/buscador. */
   172→  const refreshConversations = useCallback(async () => {
   173→    try {
   174→      const res = await fetch('/api/chat/sessions')
   175→      if (!res.ok) return
   176→      const data = (await res.json()) as { sessions?: ConversationSummary[] }
   177→      if (data.sessions) {
   178→        setConversations(data.sessions.filter((s) => s.messageCount > 0))
   179→        useUiStore.getState().setConversations(data.sessions.filter((s) => s.messageCount > 0))
   180→      }
   181→    } catch {
   182→      // sin historial visible; la app sigue funcionando
   183→    }
   184→  }, [])
   185→
   186→  /** Carga los mensajes de una conversación existente (texto e imágenes). */
   187→  const loadConversation = useCallback(async (id: string) => {
   188→    if (pendingRef.current) return
   189→    setError(null)
   190→    setRouteNote(null)
   191→    setLoadingHistory(true)
   192→    try {
   193→      const res = await fetch(`/api/chat/history?sessionId=${encodeURIComponent(id)}`)
   194→      if (!res.ok) return
   195→      const data = (await res.json()) as { messages?: Array<Record<string, unknown>> }
   196→      const restored: ChatMsg[] = (data.messages ?? []).map((m) => {
   197→        const kind = m.kind === 'image' ? 'image' : 'text'
   198→        const role = m.role === 'user' ? ('user' as const) : ('assistant' as const)
   199→        let sources: WebSource[] | undefined
   200→        if (typeof m.sourcesJson === 'string' && m.sourcesJson) {
   201→          try {
   202→            const parsed = JSON.parse(m.sourcesJson) as WebSource[]
   203→            if (Array.isArray(parsed) && parsed.length) sources = parsed
   204→          } catch {
   205→            // JSON inválido: se ignora
   206→          }
   207→        }
   208→        return {
   209→          id: String(m.id),
   210→          role,
   211→          content: String(m.content ?? ''),
   212→          fullContent: String(m.content ?? ''),
   213→          complete: true,
   214→          kind,
   215→          assetUrl: kind === 'image' && typeof m.assetId === 'string' ? `/api/assets/${m.assetId}` : null,
   216→          attachmentUrl: role === 'user' && typeof m.attachmentAssetId === 'string' ? `/api/assets/${m.attachmentAssetId}` : null,
   217→          sources,
   218→          meta: {
   219→            providerReportedModel: (m.providerReportedModel as string) ?? null,
   220→            routeNote: (m.resolvedModelId as string) ?? null,
   221→          },
   222→        }
   223→      })
   224→      setMessages(restored)
   225→      setActiveId(id)
   226→      activeIdRef.current = id
   227→      try {
   228→        window.localStorage.setItem(ACTIVE_KEY, id)
   229→      } catch {
   230→        // almacenamiento no disponible
   231→      }
   232→    } finally {
   233→      setLoadingHistory(false)
   234→    }
   235→  }, [])
   236→
   237→  // Arranque: historial + restauración de la conversación activa + enfoque desde /apis
   238→  useEffect(() => {
   239→    void refreshConversations()
   240→    let initialId: string | null = null
   241→    try {
   242→      initialId = window.localStorage.getItem(ACTIVE_KEY)
   243→    } catch {
   244→      // sin almacenamiento
   245→    }
   246→    if (initialId) void loadConversation(initialId)
   247→
   248→    // Enfoque de red llegado desde el Centro de APIs (/apis)
   249→    try {
   250→      const pending = window.sessionStorage.getItem('oe-ai-pending-model')
   251→      if (pending) {
   252→        window.sessionStorage.removeItem('oe-ai-pending-model')
   253→        useUiStore.getState().openChatWithModel(pending)
   254→      }
   255→    } catch {
   256→      // ignorar
   257→    }
   258→    const focus = useUiStore.getState().chatModel
   259→    if (focus && focus !== 'auto') setFocusModel(focus)
   260→  }, [loadConversation, refreshConversations])
   261→
   262→  /** Revelación progresiva (máquina de escribir adaptativa, duración acotada).
   263→   *  Contenido estructurado (tablas/código): sin typewriter → render Markdown
   264→   *  inmediato y bien formateado (nunca se ve la sintaxis cruda). */
   265→  const startTypewriter = useCallback((assistantId: string, full: string) => {
   266→    if (/```|(^|\n)\s*\|/.test(full)) {
   267→      setMessages((prev) =>
   268→        prev.map((m) => (m.id === assistantId ? { ...m, content: full, complete: true } : m)),
   269→      )
   270→      setPhase('idle')
   271→      return
   272→    }
   273→    setPhase('typing')
   274→    const duration = Math.min(2400, Math.max(700, full.length * 4))
   275→    const start = performance.now()
   276→    const step = (now: number) => {
   277→      const p = Math.min(1, (now - start) / duration)
   278→      const eased = 1 - Math.pow(1 - p, 2.1)
   279→      const chars = Math.max(1, Math.floor(eased * full.length))
   280→      setMessages((prev) =>
   281→        prev.map((m) => (m.id === assistantId ? { ...m, content: full.slice(0, chars) } : m)),
   282→      )
   283→      if (p < 1) {
   284→        rafRef.current = requestAnimationFrame(step)
   285→      } else {
   286→        setMessages((prev) =>
   287→          prev.map((m) => (m.id === assistantId ? { ...m, content: full, complete: true } : m)),
   288→        )
   289→        setPhase('idle')
   290→      }
   291→    }
   292→    rafRef.current = requestAnimationFrame(step)
   293→  }, [])
   294→
   295→  /** Envía un mensaje al modelo unificado OE AI LLM (con búsqueda web y/o imagen adjunta opcional). */
   296→  const send = useCallback(
   297→    async (
   298→      rawText: string,
   299→      opts?: {
   300→        isRetry?: boolean
   301→        webSearch?: boolean
   302→        /** Imagen adjuntada (visión): el núcleo la analiza junto al texto. */
   303→        attachment?: { assetId: string; previewUrl: string }
   304→        /** Texto de documento leído (enlace/PDF): contexto efímero del turno. */
   305→        docContext?: string
   306→        /** Título del documento leído (se muestra en el prompt de sistema). */
   307→        docTitle?: string
   308→      },
   309→    ) => {
   310→      const content = rawText.trim()
   311→      if (!content || pendingRef.current) return
   312→      const mode = useUiStore.getState().mode
   313→      const focus = useUiStore.getState().chatModel
   314→      const webSearch = Boolean(opts?.webSearch)
   315→      const attachment = opts?.attachment
   316→      const docContext = typeof opts?.docContext === 'string' ? opts.docContext : undefined
   317→      const docTitle = typeof opts?.docTitle === 'string' ? opts.docTitle : undefined
   318→      lastOptsRef.current = { webSearch, attachment }
   319→
   320→      // Conversación nueva si no hay una activa
   321→      let convId = activeIdRef.current
   322→      if (!convId) {
   323→        convId = newId()
   324→        activeIdRef.current = convId
   325→        setActiveId(convId)
   326→        try {
   327→          window.localStorage.setItem(ACTIVE_KEY, convId)
   328→        } catch {
   329→          // sin almacenamiento
   330→        }
   331→      }
   332→
   333→      setError(null)
   334→      setRouteNote(null)
   335→      setPhase('thinking')
   336→      setWebSearching(false)
   337→      pendingRef.current = true
   338→      if (!opts?.isRetry) {
   339→        setLastUserMsg(content)
   340→        setMessages((prev) => [
   341→          ...prev,
   342→          {
   343→            id: `u-${newId()}`,
   344→            role: 'user',
   345→            content,
   346→            fullContent: content,
   347→            complete: true,
   348→            attachmentUrl: attachment?.previewUrl ?? null,
   349→          },
   350→        ])
   351→      }
   352→
   353→      const assistantId = `a-${newId()}`
   354→      setMessages((prev) => [
   355→        ...prev,
   356→        { id: assistantId, role: 'assistant', content: '', fullContent: '', complete: false },
   357→      ])
   358→
   359→      const controller = new AbortController()
   360→      abortRef.current = controller
   361→      let doneMeta: ChatMsgMeta | null = null
   362→      let serverStreaming = false
   363→
   364→      const handleEvent = (event: string, data: unknown) => {
   365→        if (event === 'meta') {
   366→          const meta = data as { routeNote?: string; coreLabel?: string; mode?: 'instant' | 'reflective'; streaming?: boolean }
   367→          serverStreaming = Boolean(meta.streaming)
   368→          if (meta.routeNote) setRouteNote(meta.routeNote)
   369→          setMessages((prev) =>
   370→            prev.map((m) =>
   371→              m.id === assistantId
   372→                ? { ...m, meta: { ...m.meta, coreLabel: meta.coreLabel ?? null, routeNote: meta.routeNote ?? null, mode: meta.mode, streaming: serverStreaming } }
   373→                : m,
   374→            ),
   375→          )
   376→        } else if (event === 'stage') {
   377→          const { stage } = data as { stage?: string }
   378→          if (stage === 'web_search') setWebSearching(true)
   379→        } else if (event === 'search') {
   380→          const { sources } = data as { sources?: WebSource[] }
   381→          setWebSearching(false)
   382→          setMessages((prev) =>
   383→            prev.map((m) => (m.id === assistantId ? { ...m, sources: Array.isArray(sources) ? sources : [] } : m)),
   384→          )
   385→        } else if (event === 'search_error') {
   386→          const { message } = data as { message?: string }
   387→          setWebSearching(false)
   388→          setMessages((prev) =>
   389→            prev.map((m) => (m.id === assistantId ? { ...m, searchNote: message ?? 'La búsqueda web no estuvo disponible.' } : m)),
   390→          )
   391→        } else if (event === 'delta') {
   392→          const { text } = data as { text: string }
   393→          if (serverStreaming) {
   394→            // Streaming real del proveedor: cada delta es un incremento.
   395→            setPhase('typing')
   396→            setMessages((prev) =>
   397→              prev.map((m) => {
   398→                if (m.id !== assistantId) return m
   399→                const full = m.fullContent + text
   400→                return { ...m, fullContent: full, content: full }
   401→              }),
   402→            )
   403→          } else {
   404→            // Respuesta única: revelación progresiva local (efecto de presentación).
   405→            setMessages((prev) =>
   406→              prev.map((m) => (m.id === assistantId ? { ...m, fullContent: text } : m)),
   407→            )
   408→            startTypewriter(assistantId, text)
   409→          }
   410→        } else if (event === 'done') {
   411→          const d = data as { latencyMs?: number; providerReportedModel?: string | null; webSearchUsed?: boolean; visionUsed?: boolean }
   412→          doneMeta = {
   413→            latencyMs: d.latencyMs ?? null,
   414→            providerReportedModel: d.providerReportedModel ?? null,
   415→            webSearchUsed: Boolean(d.webSearchUsed),
   416→            visionUsed: Boolean(d.visionUsed),
   417→          }
   418→          if (serverStreaming) {
   419→            setMessages((prev) =>
   420→              prev.map((m) => (m.id === assistantId ? { ...m, content: m.fullContent, complete: true } : m)),
   421→            )
   422→            setPhase('idle')
   423→          }
   424→        } else if (event === 'error') {
   425→          const e = data as { message: string; errorType?: string }
   426→          setError({ message: e.message, errorType: e.errorType })
   427→          // Si ya había contenido parcial, se conserva completo; si no, se retira.
   428→          setMessages((prev) =>
   429→            prev
   430→              .map((m) => (m.id === assistantId && m.fullContent ? { ...m, content: m.fullContent, complete: true } : m))
   431→              .filter((m) => !(m.id === assistantId && !m.fullContent)),
   432→          )
   433→          setPhase('idle')
   434→        }
   435→      }
   436→
   437→      try {
   438→        const token = await auth.getAccessToken()
   439→        const res = await fetch('/api/chat', {
   440→          method: 'POST',
   441→          headers: {
   442→            'Content-Type': 'application/json',
   443→            ...(token ? { Authorization: `Bearer ${token}` } : {}),
   444→          },
   445→          body: JSON.stringify({
   446→            sessionId: convId,
   447→            message: content,
   448→            model: focus && focus !== 'auto' ? focus : undefined,
   449→            allowFallback: true,
   450→            mode,
   451→            webSearch,
   452→            attachmentId: attachment?.assetId,
   453→            docContext,
   454→            docTitle,
   455→          }),
   456→          signal: controller.signal,
   457→        })
   458→
   459→        if (!res.ok || !res.body) {
   460→          const data = (await res.json().catch(() => null)) as { error?: string; errorType?: string } | null
   461→          throw Object.assign(new Error(data?.error ?? `Error ${res.status} al iniciar la conversación.`), {
   462→            errorType: data?.errorType,
   463→          })
   464→        }
   465→
   466→        const reader = res.body.getReader()
   467→        const decoder = new TextDecoder()
   468→        let buffer = ''
   469→
   470→        while (true) {
   471→          const { done, value } = await reader.read()
   472→          if (done) break
   473→          buffer += decoder.decode(value, { stream: true })
   474→          let sep: number
   475→          while ((sep = buffer.indexOf('\n\n')) !== -1) {
   476→            const rawEvent = buffer.slice(0, sep)
   477→            buffer = buffer.slice(sep + 2)
   478→            let eventName = 'message'
   479→            let dataRaw = ''
   480→            for (const line of rawEvent.split('\n')) {
   481→              if (line.startsWith('event:')) eventName = line.slice(6).trim()
   482→              else if (line.startsWith('data:')) dataRaw += line.slice(5).trim()
   483→            }
   484→            if (dataRaw) {
   485→              try {
   486→                handleEvent(eventName, JSON.parse(dataRaw))
   487→              } catch {
   488→                // evento malformado: ignorar de forma segura
   489→              }
   490→            }
   491→          }
   492→        }
   493→
   494→        // Adjuntar metadatos finales (latencia, modelo reportado)
   495→        if (doneMeta) {
   496→          const meta = doneMeta as ChatMsgMeta
   497→          setMessages((prev) => prev.map((m) => (m.id === assistantId ? { ...m, meta: { ...m.meta, ...meta } } : m)))
   498→        }
   499→        void refreshConversations()
   500→        void refreshQuota()
   501→      } catch (err) {
   502→        if ((err as Error).name === 'AbortError') {
   503→          // Cancelación: si ya hay texto, se revela completo; si no, se retira el marcador.
   504→          setMessages((prev) =>
   505→            prev
   506→              .map((m) => (m.id === assistantId && m.fullContent ? { ...m, content: m.fullContent, complete: true } : m))
   507→              .filter((m) => !(m.id === assistantId && !m.fullContent)),
   508→          )
   509→        } else {
   510→          const e = err as Error & { errorType?: string }
   511→          if (e.errorType === 'guest_limit_reached') {
   512→            // Cuota agotada: se activa la compuerta de inicio de sesión (sin error genérico).
   513→            setGuestBlocked(true)
   514→            void refreshQuota()
   515→          } else {
   516→            setError({ message: e.message, errorType: e.errorType })
   517→          }
   518→          setMessages((prev) => prev.filter((m) => m.id !== assistantId))
   519→        }
   520→      } finally {
   521→        abortRef.current = null
   522→        pendingRef.current = false
   523→        setWebSearching(false)
   524→        if (!rafRef.current) setPhase('idle')
   525→      }
   526→    },

   530→  /** Genera una imagen con la herramienta del núcleo y la inserta en la conversación. */
   531→  const generateImage = useCallback(
   532→    async (rawPrompt: string) => {
   533→      const prompt = rawPrompt.trim()

   560→      setMessages((prev) => [

   566→      abortRef.current = controller

   576→          body: JSON.stringify({ sessionId: convId, prompt }),
   577→          signal: controller.signal,
   578→        })
   579→        const data = (await res.json().catch(() => null)) as {
   580→          url?: string
   581→          latencyMs?: number
   582→          error?: string
   583→          errorType?: string
   584→        } | null

   606→        void refreshQuota()
   607→      } catch (err) {
   608→        setMessages((prev) => prev.filter((m) => m.id !== placeholderId))
   609→        if ((err as Error).name !== 'AbortError') {
   610→          const e = err as Error & { errorType?: string }
   611→          if (e.errorType === 'guest_limit_reached') {
   612→            setGuestBlocked(true)
   613→            void refreshQuota()
   614→          } else {
   615→            setError({ message: e.message, errorType: e.errorType })
   616→          }
   617→        }
   618→      } finally {
   619→        abortRef.current = null
   620→        pendingRef.current = false
   621→        setImagePending(false)
   622→        setPhase('idle')
   623→      }
   624→    },
   625→    [auth, refreshConversations, refreshQuota],

   666→      ])
   667→
   668→      const placeholderId = `vid-${newId()}`
   669→      setMessages((prev) => [
   670→        ...prev,
   671→        { id: placeholderId, role: 'assistant', content: '', fullContent: '', complete: false },
   672→      ])
   673→

   700→        } | null

   703→            errorType: data?.errorType,

   718→          ),

   721→        void refreshQuota()

   723→      } catch (err) {

   842→  /** Detiene la generación, la revelación o la herramienta en curso. */
   843→  const stop = useCallback(() => {
   844→    abortRef.current?.abort()
   845→    abortRef.current = null
   846→    pendingRef.current = false
   847→    if (rafRef.current !== null) {
   848→      cancelAnimationFrame(rafRef.current)
   849→      rafRef.current = null
   850→    }
   851→    setPhase('idle')
   852→    setWebSearching(false)
   853→    setImagePending(false)
   854→    setMessages((prev) => prev.map((m) => (m.complete || m.pendingImage ? m : { ...m, content: m.fullContent || m.content, complete: true })))
   855→  }, [])
   856→
   857→  /** Regenera la última respuesta (conserva la búsqueda web y la imagen adjunta del turno). */
   858→  const regenerate = useCallback(() => {
   859→    if (!lastUserMsg || pendingRef.current) return
   860→    // Se elimina la última respuesta asistente incompleta o final
   861→    setMessages((prev) => {
   862→      const idx = prev.map((m) => m.role).lastIndexOf('assistant')
   863→      return idx === -1 ? prev : prev.filter((_, i) => i !== idx)
   864→    })
   865→    void send(lastUserMsg, {
   866→      isRetry: true,
   867→      webSearch: lastOptsRef.current.webSearch,
   868→      attachment: lastOptsRef.current.attachment,
   869→    })
   870→  }, [lastUserMsg, send])
   871→
   872→  /**
   873→   * Edita un mensaje del usuario: recorta la rama posterior (servidor y cliente)
   874→   * y reenvía el texto corregido como mensaje nuevo. Si el mensaje original
   875→   * tenía una imagen adjunta (visión), se reutiliza su attachmentAssetId —
   876→   * mismo contrato que regenerate, que conserva el adjunto del turno.
   877→   */
   878→  const editMessage = useCallback(
   879→    async (messageId: string, newContent: string) => {
   880→      const text = newContent.trim()
   881→      if (!text || pendingRef.current) return
   882→      const idx = messages.findIndex((m) => m.id === messageId)
   883→      if (idx === -1 || messages[idx]?.role !== 'user') return
   884→      const convId = activeIdRef.current
   885→      if (!convId) return
   886→
   887→      // Adjunto de visión conservado: el assetId se deriva de la URL del preview.
   888→      const original = messages[idx]
   889→      let attachment: { assetId: string; previewUrl: string } | undefined
   890→      if (original.attachmentUrl?.startsWith('/api/assets/')) {
   891→        const assetId = original.attachmentUrl.slice('/api/assets/'.length)
   892→        if (assetId) attachment = { assetId, previewUrl: original.attachmentUrl }
   893→      }
   894→
   895→      // Recorte en servidor: se conservan solo los mensajes anteriores a la edición.
   896→      try {
   897→        await fetch('/api/chat/trim', {
   898→          method: 'POST',
   899→          headers: { 'Content-Type': 'application/json' },
   900→          body: JSON.stringify({ sessionId: convId, keepCount: idx }),
   901→        })
   902→      } catch {
   903→        // si el recorte falla se continúa: la conversación local manda en la vista
   904→      }
   905→      setMessages((prev) => prev.slice(0, idx))
   906→      void send(text, { attachment })
   907→    },
   908→    [messages, send],
   909→  )
   910→
   911→  /** Renombra una conversación (PATCH) y actualiza sidebar/buscador. */
   912→  const renameConversation = useCallback(
   913→    async (id: string, title: string) => {
   914→      const clean = title.trim().slice(0, 80)
   915→      if (!clean) return
   916→      setConversations((prev) => prev.map((c) => (c.id === id ? { ...c, title: clean } : c)))
   917→      useUiStore.setState((s) => ({ conversations: s.conversations.map((c) => (c.id === id ? { ...c, title: clean } : c)) }))
   918→      try {
   919→        await fetch(`/api/chat/sessions/${encodeURIComponent(id)}`, {
   920→          method: 'PATCH',
   921→          headers: { 'Content-Type': 'application/json' },
   922→          body: JSON.stringify({ title: clean }),
   923→        })
   924→      } catch {
   925→        // el título local se mantiene; el servidor quedará coherente al refrescar
   926→      }
   927→      void refreshConversations()
   928→    },
   929→    [refreshConversations],
   930→  )
   931→
   932→  /** Exporta la conversación activa como Markdown descargable. */
   933→  const exportActiveConversation = useCallback(() => {
   934→    if (messages.length === 0) return
   935→    const title = conversations.find((c) => c.id === activeId)?.title ?? 'Conversación OE AI'
   936→    exportMessagesAsMarkdown(title, messages)
   937→  }, [activeId, conversations, messages])
   938→
   939→  /** Nueva conversación (Ctrl+Shift+O). */
   940→  const newChat = useCallback(() => {
   941→    if (pendingRef.current) abortRef.current?.abort()
   942→    activeIdRef.current = null
   943→    setActiveId(null)
   944→    setMessages([])
   945→    setError(null)
   946→    setRouteNote(null)
   947→    setLastUserMsg(null)
   948→    setFocusModel(null)
   949→    setWebSearching(false)
   950→    setImagePending(false)
   951→    try {
   952→      window.localStorage.removeItem(ACTIVE_KEY)
   953→      window.sessionStorage.removeItem('oe-ai-pending-model')
   954→      useUiStore.setState({ chatModel: 'auto' })
   955→    } catch {
   956→      // sin almacenamiento
   957→    }
   958→  }, [])
   959→
   960→  /** Elimina una conversación del historial. */
   961→  const deleteConversation = useCallback(
   962→    async (id: string) => {
   963→      try {
   964→        await fetch(`/api/chat/sessions/${encodeURIComponent(id)}`, { method: 'DELETE' })
   965→      } catch {
   966→        // si falla, la lista se refresca igualmente
   967→      }
   968→      if (activeIdRef.current === id) newChat()
   969→      void refreshConversations()
   970→    },
   971→    [newChat, refreshConversations],
   972→  )
   973→
   974→  // Limpieza al desmontar
   975→  useEffect(() => {
   976→    return () => {
   977→      if (rafRef.current !== null) cancelAnimationFrame(rafRef.current)
   978→      abortRef.current?.abort()
   979→    }
   980→  }, [])
   981→
   982→  const pending = phase !== 'idle'
   983→
   984→  return {
   985→    conversations,
   986→    activeId,
   987→    messages,
   988→    phase,
   989→    pending,
   990→    error,
   991→    routeNote,
   992→    focusModel,
   993→    loadingHistory,
   994→    lastUserMsg,
   995→    webSearching,
   996→    imagePending,
   997→    guestQuota,
   998→    guestBlocked,
   999→    refreshQuota,
  1000→    send,
  1001→    stop,
  1002→    regenerate,
  1003→    newChat,
  1004→    loadConversation,
  1005→    deleteConversation,
  1006→    refreshConversations,
  1007→    setFocusModel,
  1008→    generateImage,
  1009→    generateVideo,
  1010→    generateImageEdit,
  1011→    editMessage,
  1012→    renameConversation,
  1013→    exportActiveConversation,
  1014→  }
  1015→}
