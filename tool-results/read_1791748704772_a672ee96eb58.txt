    99→interface BuildDirectivePayload {
   100→  name?: string

   102→  mode?: string
   103→  projectId?: string

   108→function extractBuildDirective(text: string): BuildDirectivePayload | null {
   109→  const webIdx = text.indexOf('```oe-web')
   110→  const appIdx = text.indexOf('```oe-build')
   111→  const kind: 'app' | 'web' = webIdx !== -1 && (appIdx === -1 || webIdx < appIdx) ? 'web' : 'app'
   112→  const fenceStart = kind === 'web' ? webIdx : appIdx
   113→  if (fenceStart === -1) return null
   114→  const bodyStart = text.indexOf('\n', fenceStart)
   115→  if (bodyStart === -1) return null
   116→  const fenceEnd = text.indexOf('```', bodyStart)
   117→  if (fenceEnd === -1) return null
   118→  const raw = text.slice(bodyStart, fenceEnd).trim()
   119→  const start = raw.indexOf('{')
   120→  if (start === -1) return null
   121→  let depth = 0
   122→  let inString = false
   123→  let escaped = false
   124→  let end = -1
   125→  for (let i = start; i < raw.length; i++) {
   126→    const ch = raw[i]
   127→    if (inString) {
   128→      if (escaped) escaped = false
   129→      else if (ch === '\\') escaped = true
   130→      else if (ch === '"') inString = false
   131→      continue
   132→    }
   133→    if (ch === '"') inString = true
   134→    else if (ch === '{') depth++
   135→    else if (ch === '}') {
   136→      depth--
   137→      if (depth === 0) {
   138→        end = i
   139→        break
   140→      }
   141→    }
   142→  }
   143→  if (end === -1) return null
   144→  try {
   145→    const parsed = JSON.parse(raw.slice(start, end + 1)) as {
   146→      name?: unknown
   147→      brief?: unknown
   148→      mode?: unknown
   149→      projectId?: unknown
   150→    }
   151→    const brief = typeof parsed.brief === 'string' ? parsed.brief.trim() : ''
   152→    if (!brief) return null
   153→    const prefix = text
   154→      .slice(0, fenceStart)
   155→      .replace(/```[a-z]*\s*$/, '')
   156→      .trim()
   157→    return {
   158→      brief: brief.slice(0, 6000),
   159→      name: typeof parsed.name === 'string' ? parsed.name : undefined,
   160→      mode: typeof parsed.mode === 'string' ? parsed.mode : undefined,
   161→      projectId: typeof parsed.projectId === 'string' ? parsed.projectId : undefined,
   162→      kind,
   163→      prefix,
   164→    }
   165→  } catch {
   166→    return null
   167→  }
   168→}

   170→/** Restaura el último proyecto construido desde el historial (bloques oeproject). */
   171→function extractLastBuildFromHistory(
   172→  messages: Array<{ content: string }>,
   173→): { projectId: string; title: string; files: string[]; kind: 'app' | 'web' } | null {
   174→  const re = /```oeproject\s*\n([\s\S]*?)```/g
   175→  let last: { projectId: string; title: string; files: string[]; kind: 'app' | 'web' } | null = null
   176→  for (const m of messages) {
   177→    let match: RegExpExecArray | null
   178→    re.lastIndex = 0
   179→    while ((match = re.exec(m.content)) !== null) {
   180→      try {
   181→        const data = JSON.parse(match[1]) as { id?: unknown; title?: unknown; kind?: unknown; files?: Array<{ path?: unknown }> }
   182→        if (typeof data.id === 'string' && Array.isArray(data.files)) {
   183→          last = {
   184→            projectId: data.id,
   185→            title: typeof data.title === 'string' ? data.title : 'App construida',
   186→            files: data.files
   187→              .map((f) => (typeof f?.path === 'string' ? f.path : ''))
   188→              .filter(Boolean)
   189→              .slice(0, 12),
   190→            kind: data.kind === 'web' ? 'web' : 'app',
   191→          }
   192→        }
   193→      } catch {
   194→        // bloque inválido: ignorar
   195→      }
   196→    }
   197→  }
   198→  return last
   199→}

   378→  /** Revelación progresiva (máquina de escribir adaptativa, duración acotada).
   379→   *  Contenido estructurado (tablas/código): sin typewriter → render Markdown
   380→   *  inmediato y bien formateado (nunca se ve la sintaxis cruda). */
   381→  const startTypewriter = useCallback((assistantId: string, full: string) => {
   382→    if (/```|(^|\n)\s*\|/.test(full)) {
   383→      setMessages((prev) =>
   384→        prev.map((m) => (m.id === assistantId ? { ...m, content: full, complete: true } : m)),
   385→      )
   386→      setPhase('idle')
   387→      return
   388→    }
   389→    setPhase('typing')
   390→    const duration = Math.min(2400, Math.max(700, full.length * 4))
   391→    const start = performance.now()
   392→    const step = (now: number) => {
   393→      const p = Math.min(1, (now - start) / duration)
   394→      const eased = 1 - Math.pow(1 - p, 2.1)
   395→      const chars = Math.max(1, Math.floor(eased * full.length))
   396→      setMessages((prev) =>
   397→        prev.map((m) => (m.id === assistantId ? { ...m, content: full.slice(0, chars) } : m)),
   398→      )
   399→      if (p < 1) {
   400→        rafRef.current = requestAnimationFrame(step)
   401→      } else {
   402→        setMessages((prev) =>
   403→          prev.map((m) => (m.id === assistantId ? { ...m, content: full, complete: true } : m)),
   404→        )
   405→        setPhase('idle')
   406→      }
   407→    }
   408→    rafRef.current = requestAnimationFrame(step)
   409→  }, [])

   411→  /**
   412→   * Flujo del Constructor de Apps: el asistente emitió una directiva ```oe-build.
   413→   * Reemplaza la directiva por el placeholder aurora, llama a
   414→   * /api/tools/build-app y actualiza el mensaje con el bloque ```oeproject
   415→   * (tarjetas de archivos + descarga .zip), persistiendo el cambio.
   416→   */
   417→  const runBuildFlow = useCallback(
   418→    async (
   419→      directive: { brief: string; mode?: string; projectId?: string; kind: 'app' | 'web'; prefix: string },
   420→      assistantId: string,
   421→      serverMessageId: string,
   422→      convId: string,
   423→    ) => {
   424→      const prefix = directive.prefix
   425→      const isWeb = directive.kind === 'web'
   426→      const waiting = prefix || (isWeb ? 'Diseñando tu sitio web…' : 'Construyendo tu app…')
   427→      setMessages((prev) =>
   428→        prev.map((m) =>
   429→          m.id === assistantId
   430→            ? { ...m, content: waiting, fullContent: waiting, pendingProject: true, complete: true }
   431→            : m,
   432→        ),
   433→      )
   434→      try {
   435→        const token = await auth.getAccessToken()
   436→        const res = await fetch(isWeb ? '/api/web-factory/generate' : '/api/tools/build-app', {
   437→          method: 'POST',
   438→          headers: {
   439→            'Content-Type': 'application/json',
   440→            ...(token ? { Authorization: `Bearer ${token}` } : {}),
   441→          },
   442→          body: JSON.stringify({
   443→            sessionId: convId,
   444→            brief: directive.brief,
   445→            mode: directive.mode,
   446→            projectId: directive.projectId,
   447→          }),
   448→        })
   449→        const data = (await res.json().catch(() => null)) as {
   450→          id?: string
   451→          title?: string
   452→          summary?: string
   453→          mode?: string
   454→          isUpdate?: boolean
   455→          changedPaths?: string[]
   456→          files?: Array<{ path: string; language: string; bytes: number }>
   457→          latencyMs?: number
   458→          error?: string
   459→          errorType?: string
   460→          // Campos extra de Web Factory (vista previa, diseño, procedencia, validación)
   461→          previewUrl?: string
   462→          designSystem?: Record<string, string>
   463→          references?: Array<{ name: string; repoUrl: string; license: string; usage: string }>
   464→          checks?: Array<{ check: string; status: string; detail: string }>
   465→          warnings?: string[]
   466→        } | null
   467→        if (!res.ok || !data?.id) {
   468→          throw Object.assign(new Error(data?.error ?? `No se pudo construir la app (error ${res.status}).`), {
   469→            errorType: data?.errorType,
   470→          })
   471→        }
   472→        const projectJson = JSON.stringify({
   473→          id: data.id,
   474→          title: data.title ?? 'App construida',
   475→          summary: data.summary ?? '',
   476→          mode: data.mode ?? 'fullstack',
   477→          kind: isWeb ? 'web' : 'app',
   478→          isUpdate: Boolean(data.isUpdate),
   479→          changedPaths: data.changedPaths ?? [],
   480→          files: data.files ?? [],
   481→          ...(isWeb
   482→            ? {
   483→                previewUrl: data.previewUrl,
   484→                designSystem: data.designSystem,
   485→                references: data.references,
   486→                checks: data.checks,
   487→                warnings: data.warnings,
   488→              }
   489→            : {}),
   490→        })
   491→        const newContent = `${prefix ? `${prefix}\n\n` : ''}\`\`\`oeproject\n${projectJson}\n\`\`\``
   492→        setMessages((prev) =>
   493→          prev.map((m) =>
   494→            m.id === assistantId
   495→              ? {
   496→                  ...m,
   497→                  content: newContent,
   498→                  fullContent: newContent,
   499→                  pendingProject: false,
   500→                  complete: true,
   501→                  meta: { ...m.meta, latencyMs: data.latencyMs ?? m.meta?.latencyMs ?? null },
   502→                }
   503→              : m,
   504→          ),
   505→        )
   506→        // Persistir el mensaje actualizado (directiva → tarjetas) para el historial.
   507→        void fetch('/api/chat/messages', {
   508→          method: 'PATCH',
   509→          headers: { 'Content-Type': 'application/json' },
   510→          body: JSON.stringify({ id: serverMessageId, content: newContent }),
   511→        }).catch(() => undefined)
   512→        lastBuildRef.current = {
   513→          projectId: data.id,
   514→          title: data.title ?? 'App construida',
   515→          files: (data.files ?? []).map((f) => f.path).slice(0, 12),
   516→          kind: isWeb ? 'web' : 'app',
   517→        }
   518→        void refreshQuota()
   519→        void refreshConversations()
   520→      } catch (err) {
   521→        const e = err as Error & { errorType?: string }
   522→        if (e.errorType === 'guest_limit_reached') {
   523→          setGuestBlocked(true)
   524→          void refreshQuota()

   794→      } finally {

   796→        pendingRef.current = false

  1087→                  fullContent: content,
  1088→                  complete: true,
  1089→                  meta: { latencyMs: data.latencyMs ?? null },
  1090→                }
  1091→              : m,
  1092→          ),
  1093→        )
  1094→        void refreshConversations()
  1095→        void refreshQuota()
  1096→        return { ok: true }
  1097→      } catch (err) {
  1098→        setMessages((prev) => prev.filter((m) => m.id !== placeholderId))
  1099→        if ((err as Error).name === 'AbortError') return { ok: false, error: 'Edición cancelada.' }
  1100→        const e = err as Error & { errorType?: string }
  1101→        if (e.errorType === 'guest_limit_reached') {
  1102→          setGuestBlocked(true)
  1103→          void refreshQuota()
  1104→        } else {
  1105→          setError({ message: e.message, errorType: e.errorType })
  1106→        }
  1107→        return { ok: false, error: e.message, errorType: e.errorType }
  1108→      } finally {
  1109→        abortRef.current = null
  1110→        pendingRef.current = false
  1111→        setPhase('idle')
  1112→      }
  1113→    },
  1114→    [auth, refreshConversations, refreshQuota],
  1115→  )
  1116→
  1117→  /** Detiene la generación, la revelación o la herramienta en curso. */
  1118→  const stop = useCallback(() => {
  1119→    abortRef.current?.abort()
  1120→    abortRef.current = null
  1121→    pendingRef.current = false
  1122→    if (rafRef.current !== null) {
  1123→      cancelAnimationFrame(rafRef.current)
  1124→      rafRef.current = null
  1125→    }
  1126→    setPhase('idle')
  1127→    setWebSearching(false)
  1128→    setImagePending(false)
  1129→    setMessages((prev) => prev.map((m) => (m.complete || m.pendingImage ? m : { ...m, content: m.fullContent || m.content, complete: true })))
  1130→  }, [])
  1131→
  1132→  /** Regenera la última respuesta (conserva la búsqueda web y la imagen adjunta del turno). */
  1133→  const regenerate = useCallback(() => {
  1134→    if (!lastUserMsg || pendingRef.current) return
  1135→    // Se elimina la última respuesta asistente incompleta o final
  1136→    setMessages((prev) => {
  1137→      const idx = prev.map((m) => m.role).lastIndexOf('assistant')
  1138→      return idx === -1 ? prev : prev.filter((_, i) => i !== idx)
  1139→    })
  1140→    void send(lastUserMsg, {
  1141→      isRetry: true,
  1142→      webSearch: lastOptsRef.current.webSearch,
  1143→      attachment: lastOptsRef.current.attachment,
  1144→    })
  1145→  }, [lastUserMsg, send])
  1146→
  1147→  /**
  1148→   * Edita un mensaje del usuario: recorta la rama posterior (servidor y cliente)
  1149→   * y reenvía el texto corregido como mensaje nuevo. Si el mensaje original
  1150→   * tenía una imagen adjunta (visión), se reutiliza su attachmentAssetId —
  1151→   * mismo contrato que regenerate, que conserva el adjunto del turno.
  1152→   */
  1153→  const editMessage = useCallback(
  1154→    async (messageId: string, newContent: string) => {
  1155→      const text = newContent.trim()
  1156→      if (!text || pendingRef.current) return
  1157→      const idx = messages.findIndex((m) => m.id === messageId)
  1158→      if (idx === -1 || messages[idx]?.role !== 'user') return
  1159→      const convId = activeIdRef.current
  1160→      if (!convId) return
  1161→
  1162→      // Adjunto de visión conservado: el assetId se deriva de la URL del preview.
  1163→      const original = messages[idx]
  1164→      let attachment: { assetId: string; previewUrl: string } | undefined
  1165→      if (original.attachmentUrl?.startsWith('/api/assets/')) {
  1166→        const assetId = original.attachmentUrl.slice('/api/assets/'.length)
  1167→        if (assetId) attachment = { assetId, previewUrl: original.attachmentUrl }
  1168→      }
  1169→
  1170→      // Recorte en servidor: se conservan solo los mensajes anteriores a la edición.
  1171→      try {
  1172→        await fetch('/api/chat/trim', {
  1173→          method: 'POST',
  1174→          headers: { 'Content-Type': 'application/json' },
  1175→          body: JSON.stringify({ sessionId: convId, keepCount: idx }),
  1176→        })
  1177→      } catch {
  1178→        // si el recorte falla se continúa: la conversación local manda en la vista
  1179→      }
  1180→      setMessages((prev) => prev.slice(0, idx))
  1181→      void send(text, { attachment })
  1182→    },
  1183→    [messages, send],
  1184→  )
  1185→
  1186→  /** Renombra una conversación (PATCH) y actualiza sidebar/buscador. */
  1187→  const renameConversation = useCallback(
  1188→    async (id: string, title: string) => {
  1189→      const clean = title.trim().slice(0, 80)
  1190→      if (!clean) return
  1191→      setConversations((prev) => prev.map((c) => (c.id === id ? { ...c, title: clean } : c)))
  1192→      useUiStore.setState((s) => ({ conversations: s.conversations.map((c) => (c.id === id ? { ...c, title: clean } : c)) }))
  1193→      try {
  1194→        await fetch(`/api/chat/sessions/${encodeURIComponent(id)}`, {
  1195→          method: 'PATCH',
  1196→          headers: { 'Content-Type': 'application/json' },
  1197→          body: JSON.stringify({ title: clean }),
  1198→        })
  1199→      } catch {
  1200→        // el título local se mantiene; el servidor quedará coherente al refrescar
  1201→      }
  1202→      void refreshConversations()
  1203→    },
  1204→    [refreshConversations],
  1205→  )
  1206→
  1207→  /** Exporta la conversación activa como Markdown descargable. */
  1208→  const exportActiveConversation = useCallback(() => {
  1209→    if (messages.length === 0) return
  1210→    const title = conversations.find((c) => c.id === activeId)?.title ?? 'Conversación OE AI'
  1211→    exportMessagesAsMarkdown(title, messages)
  1212→  }, [activeId, conversations, messages])
  1213→
  1214→  /** Nueva conversación (Ctrl+Shift+O). */
  1215→  const newChat = useCallback(() => {
  1216→    if (pendingRef.current) abortRef.current?.abort()
  1217→    activeIdRef.current = null
  1218→    setActiveId(null)
  1219→    setMessages([])
  1220→    setError(null)
  1221→    setRouteNote(null)
  1222→    setLastUserMsg(null)
  1223→    setFocusModel(null)
  1224→    setWebSearching(false)
  1225→    setImagePending(false)
  1226→    lastBuildRef.current = null
  1227→    try {
  1228→      window.localStorage.removeItem(ACTIVE_KEY)
  1229→      window.sessionStorage.removeItem('oe-ai-pending-model')
  1230→      useUiStore.setState({ chatModel: 'auto' })
  1231→    } catch {
  1232→      // sin almacenamiento
  1233→    }
  1234→  }, [])
  1235→
  1236→  /** Elimina una conversación del historial. */
  1237→  const deleteConversation = useCallback(
  1238→    async (id: string) => {
  1239→      try {
  1240→        await fetch(`/api/chat/sessions/${encodeURIComponent(id)}`, { method: 'DELETE' })
  1241→      } catch {
  1242→        // si falla, la lista se refresca igualmente
  1243→      }
  1244→      if (activeIdRef.current === id) newChat()
  1245→      void refreshConversations()
  1246→    },
  1247→    [newChat, refreshConversations],
  1248→  )
  1249→
  1250→  // Limpieza al desmontar
  1251→  useEffect(() => {
  1252→    return () => {
  1253→      if (rafRef.current !== null) cancelAnimationFrame(rafRef.current)
  1254→      abortRef.current?.abort()
  1255→    }
  1256→  }, [])
  1257→
  1258→  const pending = phase !== 'idle'
  1259→
  1260→  return {
  1261→    conversations,
  1262→    activeId,
  1263→    messages,
  1264→    phase,
  1265→    pending,
  1266→    error,
  1267→    routeNote,
  1268→    focusModel,
  1269→    loadingHistory,
  1270→    lastUserMsg,
  1271→    webSearching,
  1272→    imagePending,
  1273→    guestQuota,
  1274→    guestBlocked,
  1275→    refreshQuota,
  1276→    send,
  1277→    stop,
  1278→    regenerate,
  1279→    newChat,
  1280→    loadConversation,
  1281→    deleteConversation,
  1282→    refreshConversations,
  1283→    setFocusModel,
  1284→    generateImage,
  1285→    generateVideo,
  1286→    generateImageEdit,
  1287→    editMessage,
  1288→    renameConversation,
  1289→    exportActiveConversation,
  1290→  }
  1291→}
  1292→
  1293→export type OeChatApi = ReturnType<typeof useOeChat>
