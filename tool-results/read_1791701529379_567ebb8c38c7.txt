     1→/**
     2→ * Pipeline del Modo Agente de OE AI — TODO REAL, nada simulado.
     3→ *
     4→ * Orquesta el turno completo contra núcleos LLM reales y contra la máquina:
     5→ *   1. Plan (agente planificador → JSON) → se convierte en la lista de Todos
     6→ *   2. Código (agente de código → archivos completos) → escritura REAL en disco
     7→ *   3. `bun install` REAL (salida en streaming al terminal)
     8→ *   4. Servidor `next dev` REAL (puerto propio, salida en streaming)
     9→ *   5. Verificación REAL de la página (HTTP) + bucle de reparación con el
    10→ *      Agente Reparador si el servidor devuelve un error real
    11→ *   6. Resumen final compuesto localmente (sin llamada extra al LLM)
    12→ *
    13→ * Cada paso emite eventos SSE (thought / todo / cmd_* / file / preview_status)
    14→ * y persiste el terminal real en AgentLog. Nada se simula: si un paso falla,
    15→ * se informa con honestidad y el proyecto queda en 'error' con su progreso.
    16→ */

    54→function normalizePath(raw: string): string | null {
    55→  if (typeof raw !== 'string') return null
    56→  const clean = raw.trim().replace(/^\.\//, '').replace(/^\/+/, '')
    57→  if (!clean || clean.includes('..') || clean.length > 200) return null
    58→  return clean
    59→}
    60→
    61→/** Normaliza y valida la lista de archivos del agente de código. */
    62→function sanitizeFiles(raw: unknown, maxFiles: number): AgentGeneratedFile[] {
    63→  if (!Array.isArray(raw)) return []
    64→  const seen = new Set<string>()
    65→  const out: AgentGeneratedFile[] = []
    66→  for (const item of raw.slice(0, maxFiles + 4)) {
    67→    if (!item || typeof item !== 'object') continue
    68→    const path = normalizePath(String((item as { path?: unknown }).path ?? ''))
    69→    const content = String((item as { content?: unknown }).content ?? '')
    70→    if (!path || !content.trim() || seen.has(path)) continue
    71→    if (path === 'preview.html' || path.startsWith('node_modules')) continue
    72→    seen.add(path)
    73→    out.push({ path, content: content.slice(0, 40000) })
    74→    if (out.length >= maxFiles) break
    75→  }
    76→  return out
    77→}
    78→
    79→function asPlan(raw: Record<string, unknown>): AgentPlan {
    80→  const strArr = (v: unknown, max: number): string[] =>
    81→    Array.isArray(v) ? v.filter((x) => typeof x === 'string' && x.trim()).slice(0, max).map((s) => String(s).trim()) : []
    82→
    83→  const filesRaw = Array.isArray(raw.files) ? raw.files : []
    84→  const files = filesRaw
    85→    .slice(0, 14)
    86→    .map((f) => {
    87→      const obj = (f ?? {}) as { path?: unknown; purpose?: unknown; language?: unknown }
    88→      return {
    89→        path: String(obj.path ?? ''),
    90→        purpose: String(obj.purpose ?? '').slice(0, 200),
    91→        language: String(obj.language ?? 'text').slice(0, 20),
    92→      }
    93→    })
    94→    .filter((f) => f.path)
    95→
    96→  return {
    97→    title: String(raw.title ?? 'Proyecto OE AI').slice(0, 80),
    98→    summary: String(raw.summary ?? '').slice(0, 500),
    99→    appSlug: String(raw.appSlug ?? 'oe-app')
   100→      .toLowerCase()
   101→      .replace(/[^a-z0-9-]+/g, '-')
   102→      .replace(/^-+|-+$/g, '')
   103→      .slice(0, 48) || 'oe-app',
   104→    features: strArr(raw.features, 8),
   105→    pages: Array.isArray(raw.pages)
   106→      ? raw.pages.slice(0, 8).map((p) => {
   107→          const obj = (p ?? {}) as { route?: unknown; purpose?: unknown }
   108→          return { route: String(obj.route ?? '/').slice(0, 100), purpose: String(obj.purpose ?? '').slice(0, 200) }
   109→        })
   110→      : [],
   111→    apiEndpoints: Array.isArray(raw.apiEndpoints)
   112→      ? raw.apiEndpoints.slice(0, 8).map((e) => {
   113→          const obj = (e ?? {}) as { method?: unknown; path?: unknown; purpose?: unknown }
   114→          return {
   115→            method: String(obj.method ?? 'GET').slice(0, 8).toUpperCase(),
   116→            path: String(obj.path ?? '/api').slice(0, 120),
   117→            purpose: String(obj.purpose ?? '').slice(0, 200),
   118→          }
   119→        })
   120→      : [],
   121→    files,
   122→  }
   123→}

   131→export class AgentPipeline {
   132→  private todos: AgentTodo[] = []
   133→  private thoughts: string[] = []
   134→  private turnFiles: string[] = []
   135→
   136→  constructor(
   137→    private readonly projectId: string,
   138→    private readonly send: PipelineSend,
   139→    private readonly signal: AbortSignal,
   140→  ) {}
   141→
   142→  private async log(level: 'info' | 'command' | 'success' | 'warn' | 'error', text: string) {
   143→    const max = await db.agentLog.aggregate({ _max: { seq: true }, where: { projectId: this.projectId } })
   144→    const seq = (max._max.seq ?? 0) + 1
   145→    await db.agentLog.create({ data: { projectId: this.projectId, seq, level, text: text.slice(0, 2000) } })
   146→    this.send('log', { seq, level, text })
   147→  }
   148→
   149→  private thought(text: string) {
   150→    this.thoughts.push(text)
   151→    this.send('thought', { text })
   152→  }
   153→
   154→  private setTodos(todos: AgentTodo[]) {
   155→    this.todos = todos
   156→    this.send('todos_reset', { todos })
   157→  }
   158→
   159→  private setTodo(id: string, status: AgentTodo['status']) {
   160→    this.todos = this.todos.map((t) => (t.id === id ? { ...t, status } : t))
   161→    this.send('todo', { id, status })
   162→  }
   163→
   164→  private async setPhase(phase: AgentPhase, note?: string) {
   165→    const status = phase === 'idle' ? 'draft' : phase
   166→    await db.agentProject.update({ where: { id: this.projectId }, data: { status, errorType: null } })
   167→    this.send('phase', { phase, note })
   168→  }
   169→
   170→  private snapshot(): TurnSnapshot {
   171→    return {
   172→      todos: this.todos,
   173→      thoughts: [...this.thoughts],
   174→      files: [...this.turnFiles],
   175→    }
   176→  }
   177→
   178→  /** Ejecuta un comando REAL registrándolo en el terminal en vivo y persistido. */
   179→  private async runRealCommand(command: string, opts: { timeoutMs?: number } = {}) {
   180→    const cmdId = crypto.randomUUID()
   181→    await this.log('command', command)
   182→    this.send('cmd_start', { id: cmdId, command, cwd: 'workspace del proyecto' })
   183→    const result = await runCommand(this.projectId, command, {
   184→      signal: this.signal,
   185→      timeoutMs: opts.timeoutMs,
   186→      onLine: (line: RunLine) => {
   187→        this.send('cmd_output', { id: cmdId, stream: line.stream, text: line.text })
   188→        // Persistir salida real (recortada) como log del terminal
   189→        void this.log(line.stream === 'err' ? 'warn' : 'info', line.text)
   190→      },
   191→    })
   192→    this.send('cmd_end', {
   193→      id: cmdId,
   194→      code: result.code,
   195→      durationMs: result.durationMs,
   196→      timedOut: result.timedOut,
   197→    })
   198→    return result
   199→  }
   200→
   201→  /** Escribe un archivo REAL en disco + base de datos y emite el evento. */
   202→  private async saveFile(path: string, content: string, action: 'create' | 'update' | 'base') {
   203→    const bytes = Buffer.byteLength(content, 'utf8')
   204→    const language = languageFromPath(path)
   205→    const existing = await db.agentFile.findUnique({
   206→      where: { projectId_path: { projectId: this.projectId, path } },
   207→      select: { id: true },
   208→    })
   209→    const finalAction = existing ? (action === 'base' ? 'base' : 'update') : 'create'
   210→    await db.agentFile.upsert({
   211→      where: { projectId_path: { projectId: this.projectId, path } },
   212→      create: { projectId: this.projectId, path, content, language, bytes, action: finalAction },
   213→      update: { content, language, bytes, action: finalAction },
   214→    })
   215→    // Escritura REAL en el workspace (servidor next dev la toma en caliente)
   216→    await writeProjectFile(this.projectId, path, content)
   217→    this.send('file', { path, language, bytes, action: finalAction, content })
   218→    if (!this.turnFiles.includes(path)) this.turnFiles.push(path)
   219→    await this.log('success', `escribir ${path} (${bytes >= 1024 ? `${(bytes / 1024).toFixed(1)} kB` : `${bytes} B`}) · disco`)
   220→  }
   221→
   222→  private aborted(): boolean {
   223→    return this.signal.aborted
   224→  }
   225→
   226→  /** Ejecuta un turno completo del agente (inicial o de continuación). */
   227→  async runTurn(opts: { isInitial: boolean; mode: AgentMode; userMessage: string }) {
   228→    const started = Date.now()
   229→    const mode = opts.mode
   230→
   231→    try {
   232→      await this.log('command', opts.isInitial
   233→        ? `iniciar agente (${mode}) · pipeline real: plan → código → bun install → next dev → verificar`
   234→        : 'iniciar agente (iteración real) sobre el proyecto en disco')
   235→
   236→      // ── 1) Plan (solo turno inicial o proyecto sin plan) ────────────────────
   237→      let plan = await this.loadPlan()
   238→      if (!plan) {
   239→        this.setTodos([
   240→          { id: 'analyze', label: 'Analizar la petición', status: 'in_progress' },
   241→          { id: 'plan', label: 'Planificar la arquitectura', status: 'pending' },
   242→          { id: 'code', label: 'Escribir los archivos del proyecto', status: 'pending' },
   243→          { id: 'install', label: 'Instalar dependencias (bun install real)', status: 'pending' },
   244→          { id: 'devserver', label: 'Iniciar servidor Next.js real', status: 'pending' },
   245→          { id: 'verify', label: 'Verificar la página y reparar errores reales', status: 'pending' },
   246→          { id: 'summary', label: 'Resumen del turno', status: 'pending' },
   247→        ])
   248→        this.thought('Voy a analizar tu petición y definir la arquitectura del proyecto Next.js.')
   249→        this.setTodo('analyze', 'done')
   250→        this.setTodo('plan', 'in_progress')
   251→        await this.setPhase('planning')
   252→        await this.log('info', 'agente planificador analizando la petición')
   253→        const planner = await agentChatJson<Record<string, unknown>>(
   254→          plannerSystemPrompt(),
   255→          plannerUserPrompt(opts.userMessage, mode),
   256→          this.signal,
   257→        )
   258→        this.reportCore(planner.core)
   259→        if (this.aborted()) return this.finishAborted()
   260→        plan = asPlan(planner.data)
   261→        await db.agentProject.update({
   262→          where: { id: this.projectId },
   263→          data: { planJson: JSON.stringify(plan), title: plan.title },
   264→        })
   265→        this.send('plan', { plan })
   266→        this.thought(`Plan listo: ${plan.files.length} archivos, ${plan.pages.length} página(s) y ${plan.apiEndpoints.length} endpoint(s). Ahora escribo el código real en el workspace.`)
   267→        await this.log('success', `plan listo: ${plan.files.length} archivos · ${plan.pages.length} páginas · ${plan.apiEndpoints.length} endpoints`)
   268→        this.setTodo('plan', 'done')
   269→      } else {
   270→        this.setTodos([
   271→          { id: 'analyze', label: 'Analizar la petición', status: 'in_progress' },
   272→          { id: 'code', label: 'Aplicar cambios en el código', status: 'pending' },
   273→          { id: 'devserver', label: 'Compilar y servir (next dev real)', status: 'pending' },
   274→          { id: 'verify', label: 'Verificar la página y reparar errores reales', status: 'pending' },
   275→          { id: 'summary', label: 'Resumen del turno', status: 'pending' },
   276→        ])
   277→        this.thought('Continúo el proyecto existente: voy a aplicar tu petición sobre los archivos reales en disco.')
   278→      }
   279→
   280→      // ── 2) Código ───────────────────────────────────────────────────────────
   281→      await this.setPhase('coding')
   282→      let filesChanged: AgentGeneratedFile[] = []
   283→      let followUpSummary: string | null = null
   284→
   285→      if (!plan) throw new AgentLlmError('El agente no disponía de plan válido.', 'json_parse')
   286→
   287→      if (opts.isInitial) {
   288→        this.setTodo('code', 'in_progress')
   289→        await this.log('info', 'agente de código implementando los archivos del plan')
   290→        const coder = await agentChatJson<{ files?: unknown }>(
   291→          coderSystemPrompt(),
   292→          coderUserPrompt(JSON.stringify(plan), mode),
   293→          this.signal,
   294→        )
   295→        this.reportCore(coder.core)
   296→        if (this.aborted()) return this.finishAborted()
   297→        const files = sanitizeFiles(coder.data.files, mode === 'long' ? 10 : 8)
   298→        if (files.length === 0) {
   299→          throw new AgentLlmError('El agente de código no devolvió archivos utilizables.', 'json_parse')
   300→        }
   301→        const plannedPaths = new Set(plan.files.map((f) => f.path))
   302→        const missing = [...plannedPaths].filter((p) => !files.some((f) => f.path === p))
   303→        if (missing.length) await this.log('warn', `archivos planificados no entregados: ${missing.slice(0, 4).join(', ')}`)
   304→        filesChanged = files
   305→      } else {
   306→        this.setTodo('code', 'in_progress')
   307→        const project = await db.agentProject.findUnique({ where: { id: this.projectId }, select: { planJson: true } })
   308→        const existingFiles = await db.agentFile.findMany({ where: { projectId: this.projectId }, select: { path: true } })
   309→        const recent = await db.agentMessage.findMany({
   310→          where: { projectId: this.projectId },
   311→          orderBy: { createdAt: 'desc' },
   312→          take: 6,
   313→          select: { role: true, content: true },
   314→        })
   315→        const recentTurns = recent
   316→          .reverse()
   317→          .map((m) => `${m.role === 'user' ? 'Usuario' : 'Agente'}: ${m.content.slice(0, 220)}`)
   318→          .join('\n')
   319→        await this.log('info', 'agente de código aplicando la petición sobre el proyecto existente')
   320→        const fu = await agentChatJson<{ summary?: unknown; files?: unknown }>(
   321→          followupSystemPrompt(),
   322→          followupUserPrompt({
   323→            planJson: project?.planJson ?? JSON.stringify(plan),
   324→            filePaths: existingFiles.map((f) => f.path),
   325→            recentTurns,
   326→            message: opts.userMessage,
   327→            mode,
   328→          }),
   329→          this.signal,
   330→        )
   331→        this.reportCore(fu.core)
   332→        if (this.aborted()) return this.finishAborted()
   333→        followUpSummary = String(fu.data.summary ?? 'Cambios aplicados.').slice(0, 400)
   334→        filesChanged = sanitizeFiles(fu.data.files, 4)
   335→        if (filesChanged.length === 0) {
   336→          await this.log('warn', 'la iteración no produjo cambios de archivo')
   337→        }
   338→      }
   339→
   340→      for (const f of filesChanged) {
   341→        if (this.aborted()) return this.finishAborted()
   342→        await this.saveFile(f.path, f.content, 'update')
   343→      }
   344→      this.setTodo('code', 'done')
   345→
   346→      // Plantilla base real (config Next.js ejecutable) + respaldos mínimos
   347→      if (opts.isInitial) {
   348→        this.thought('Añado la plantilla base real: package.json, tsconfig, next.config y configuración de Tailwind 4.')
   349→        await this.log('info', 'añadiendo plantilla base (package.json, tsconfig, next.config, postcss)')
   350→        for (const bf of BASE_FILES) await this.saveFile(bf.path, bf.content, 'base')
   351→        const savedPaths = new Set(filesChanged.map((f) => f.path))
   352→        for (const fb of FALLBACK_FILES) {
   353→          if (!savedPaths.has(fb.path)) await this.saveFile(fb.path, fb.content, 'base')
   354→        }
   355→      }
   356→
   357→      // ── 3) Instalación REAL (solo turno inicial) ────────────────────────────
   358→      let readyMs: number | undefined
   359→      if (opts.isInitial) {
   360→        this.setTodo('install', 'in_progress')
   361→        await this.setPhase('installing')
   362→        this.send('preview_status', { status: 'installing', detail: 'bun install en curso' })
   363→        this.thought('Dependencias listas para instalar: ejecuto `bun install` de verdad y transmito su salida al terminal.')
   364→        const install = await this.runRealCommand('bun install --prefer-offline --no-progress', { timeoutMs: 240_000 })
   365→        if (install.code !== 0) {
   366→          const tail = install.lines.filter((l) => l.stream === 'err').slice(-6).map((l) => l.text).join(' | ')
   367→          throw new AgentLlmError(
   368→            `bun install falló (código ${install.code ?? 'nulo'}${install.timedOut ? ', timeout' : ''}). ${tail.slice(0, 300)}`,
   369→            'provider_error',
   370→          )
   371→        }
   372→        await this.log('success', `bun install completado en ${(install.durationMs / 1000).toFixed(1)} s`)
   373→        this.setTodo('install', 'done')
   374→      }
   375→
   376→      // ── 4) Servidor Next.js REAL ────────────────────────────────────────────
   377→      this.setTodo('devserver', 'in_progress')
   378→      await this.setPhase('starting')
   379→      this.send('preview_status', { status: 'starting', detail: 'next dev arrancando' })
   380→      this.thought(opts.isInitial
   381→        ? 'Ahora arranco el servidor de desarrollo REAL (next dev) del proyecto. Su salida en vivo llega al terminal.'
   382→        : 'El servidor recogerá los cambios en caliente; verifico que compile y sirva la página actualizada.')
   383→      const devId = `dev-${this.projectId.slice(0, 8)}`
   384→      this.send('cmd_start', { id: devId, command: 'next dev --hostname 127.0.0.1 (servidor del proyecto)', cwd: 'workspace del proyecto' })
   385→      const dev = await ensureDevServer(this.projectId, {
   386→        signal: this.signal,
   387→        onLine: (line) => {
   388→          this.send('cmd_output', { id: devId, stream: line.stream, text: line.text })
   389→          void this.log(line.stream === 'err' ? 'warn' : 'info', `[next dev] ${line.text}`)
   390→        },
   391→      })
   392→      this.send('cmd_end', { id: devId, code: 0, durationMs: dev.readyMs, timedOut: false })
   393→      await this.log('success', `next dev activo en el puerto ${dev.port}${dev.alreadyRunning ? ' (reutilizado)' : `, listo en ${(dev.readyMs / 1000).toFixed(1)} s`}`)
   394→      readyMs = dev.readyMs || undefined
   395→      this.setTodo('devserver', 'done')
   396→
   397→      // ── 5) Verificación REAL + bucle de reparación ──────────────────────────
   398→      this.setTodo('verify', 'in_progress')
   399→      await this.setPhase('verifying')
   400→      let fixAttempts = 0
   401→      let verification = await verifyProjectPage(this.projectId, dev.port)
   402→      if (!verification.ok) {
   403→        this.thought('La verificación real falló. Leo el error real del servidor y activo al Agente Reparador.')
   404→      }
   405→      while (!verification.ok && fixAttempts < MAX_FIX_ATTEMPTS && !this.aborted()) {
   406→        fixAttempts++
   407→        const errText = verification.errorText ?? `HTTP ${verification.status}`
   408→        this.send('fix_attempt', { attempt: fixAttempts, error: errText })
   409→        await this.log('error', `verificación real fallida (intento ${fixAttempts}): ${errText.slice(0, 300)}`)
   410→        const fixed = await this.repair(errText)
   411→        if (!fixed.length) break
   412→        for (const f of fixed) {
   413→          if (this.aborted()) return this.finishAborted()
   414→          await this.saveFile(f.path, f.content, 'update')
   415→        }
   416→        // El servidor recompila en caliente; esperar antes de reverificar
   417→        await new Promise((r) => setTimeout(r, 2500))
   418→        verification = await verifyProjectPage(this.projectId, dev.port)
   419→      }
   420→      if (!verification.ok) {
   421→        throw new AgentLlmError(
   422→          `La página del proyecto sigue fallando tras ${MAX_FIX_ATTEMPTS} intentos de reparación. Último error real: ${(verification.errorText ?? `HTTP ${verification.status}`).slice(0, 300)}`,
   423→          'provider_error',
   424→        )
   425→      }
   426→      this.thought('Verificación superada: el servidor respondió correctamente. La vista previa en vivo ya es la app real.')
   427→      await this.log('success', `página verificada (HTTP ${verification.status}) · vista previa real activa`)
   428→      this.setTodo('verify', 'done')
   429→      this.send('preview_status', { status: 'ready', port: dev.port, readyMs })
   430→
   431→      // ── 6) Resumen final (compuesto localmente, sin LLM extra) ─────────────
   432→      this.setTodo('summary', 'in_progress')
   433→      const savedFiles = await db.agentFile.findMany({
   434→        where: { projectId: this.projectId, path: { notIn: ['preview.html'] } },
   435→        select: { path: true, bytes: true },
   436→        orderBy: { path: 'asc' },
   437→      })
   438→      const changedForSummary = filesChanged.length
   439→        ? filesChanged.map((f) => ({ path: f.path, bytes: Buffer.byteLength(f.content, 'utf8') }))
   440→        : savedFiles.slice(0, 8)
   441→      const summary = finalSummaryText({
   442→        title: plan.title,
   443→        planSummary: plan.summary,
   444→        files: changedForSummary,
   445→        isFollowUp: !opts.isInitial,
   446→        followUpSummary: followUpSummary ?? undefined,
   447→        readyMs,
   448→        fixAttempts,
   449→        devPort: dev.port,
   450→      })
   451→

   488→  /** Reparación real: pide al LLM corregir con el error real del servidor. */
   489→  private async repair(errorText: string): Promise<AgentGeneratedFile[]> {
   490→    const files = await db.agentFile.findMany({
   491→      where: { projectId: this.projectId, path: { notIn: ['preview.html'] } },
   492→      select: { path: true, content: true },
   493→      orderBy: { path: 'asc' },
   494→    })
   495→    // Heurística: incluir archivos que el error menciona + página/layout
   496→    const lower = errorText.toLowerCase()
   497→    const mentioned = files.filter((f) => errorText.includes(f.path) || lower.includes(f.path.split('/').pop()?.toLowerCase() ?? '###'))
   498→    const priority = ['app/page.tsx', 'app/layout.tsx', 'app/globals.css', 'lib/data.ts']
   499→    const relevant: Array<{ path: string; content: string }> = []
   500→    for (const p of priority) {
   501→      const f = files.find((x) => x.path === p)
   502→      if (f) relevant.push(f)
   503→    }
   504→    for (const f of mentioned.slice(0, 3)) {
   505→      if (!relevant.some((r) => r.path === f.path)) relevant.push(f)
   506→    }
   507→    if (relevant.length === 0 && files[0]) relevant.push(files[0])
   508→
   509→    const fixer = await agentChatJson<{ files?: unknown; summary?: unknown }>(
   510→      fixerSystemPrompt(),
   511→      fixerUserPrompt({
   512→        error: errorText,
   513→        filePaths: files.map((f) => f.path),
   514→        relevantFiles: relevant.slice(0, 3),
   515→      }),
   516→      this.signal,
   517→    )
   518→    this.reportCore(fixer.core)
   519→    const fixed = sanitizeFiles(fixer.data.files, 3)
   520→    if (fixed.length === 0) await this.log('warn', 'el reparador no devolvió archivos utilizables')
   521→    else this.thought(`Reparación aplicada: ${fixed.map((f) => f.path).join(', ')}. Reverifico contra el servidor real.`)
   522→    return fixed
   523→  }
   524→
   525→  private reportCore(core: AgentCore) {
   526→    void this.log('info', `núcleo LLM que respondió: ${core}`)
   527→  }

   530→    await this.log('warn', 'ejecución interrumpida por el usuario')
   531→    const filesCount = await db.agentFile.count({ where: { projectId: this.projectId, path: { not: 'preview.html' } } })
   532→    await db.agentProject.update({

   535→    })
   536→    this.send('done', { projectId: this.projectId, status: 'ready', filesCount, latencyMs: 0 })
   537→  }

   539→  private async finishError(err: unknown) {
   540→    const isLlm = err instanceof AgentLlmError
   541→    const errorType = isLlm ? err.errorType : 'provider_error'
   542→    const message = isLlm
   543→      ? err.message
   544→      : 'El agente no pudo completar el turno. Se conserva el progreso real del proyecto en disco.'
   545→    await this.log('error', `error del agente (${errorType}): ${message}`)
   546→    await db.agentProject.update({
   547→      where: { id: this.projectId },
   548→      data: { status: 'error', errorType, updatedAt: new Date() },
   549→    })
   550→    this.send('preview_status', { status: 'error', detail: message.slice(0, 200) })
   551→    this.send('error', { message, errorType, requestId: this.projectId })
   552→  }

   565→/** Terminal REAL del usuario (usado por la ruta /terminal): ejecuta, persiste y emite. */
   566→export async function runUserTerminalCommand(
   567→  projectId: string,
   568→  command: string,
   569→  emit: (event: string, data: unknown) => void,
   570→): Promise<void> {
   571→  const cmdId = crypto.randomUUID()
   572→  const max = await db.agentLog.aggregate({ _max: { seq: true }, where: { projectId } })
   573→  let seq = (max._max.seq ?? 0) + 1
   574→  const nextSeq = () => seq++
   575→  const persist = (level: 'info' | 'command' | 'warn' | 'error' | 'success', text: string) => {
   576→    const s = nextSeq()
   577→    return db.agentLog.create({ data: { projectId, seq: s, level, text: text.slice(0, 2000) } }).catch(() => undefined)
   578→  }
   579→
   580→  await persist('command', command)
   581→  emit('cmd_start', { id: cmdId, command, cwd: 'workspace del proyecto' })
   582→  const result = await execUserCommand(projectId, command, {
   583→    onLine: (line) => {
   584→      emit('cmd_output', { id: cmdId, stream: line.stream, text: line.text })
   585→      void persist(line.stream === 'err' ? 'warn' : 'info', line.text)
   586→    },
   587→  })
   588→  emit('cmd_end', { id: cmdId, code: result.code, durationMs: result.durationMs, timedOut: result.timedOut })
   589→  await persist(result.code === 0 ? 'success' : 'error', `proceso terminado (código ${result.code ?? 'nulo'}${result.timedOut ? ', timeout' : ''}) en ${(result.durationMs / 1000).toFixed(1)} s`)
   590→}
