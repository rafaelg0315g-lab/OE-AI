     4→     4→ * useOeChat — lógica del chat premium con OE AI LLM.
     5→     5→ *
     6→     6→ * - Comunica con POST /api/chat (SSE: meta → stage/search → delta → done | error).
     7→     7→ * - El backend entrega la respuesta completa (streaming de tokens no verificado);
     8→     8→ *   la revelación progresiva es un efecto de presentación local (máquina de
     9→     9→ *   escribir adaptativa) para una experiencia fluida, sin fingir datos.
    10→    10→ * - Gestiona historial multi-conversación vía /api/chat/sessions y
    11→    11→ *   /api/chat/history; herramientas: imágenes (/api/tools/image), búsqueda web
    12→    12→ *   real con fuentes, edición de mensajes (recorte), renombrar y exportar.

   180→   180→      const restored: ChatMsg[] = (data.messages ?? []).map((m) => {
   181→   181→        const kind = m.kind === 'image' ? 'image' : 'text'
   182→   182→        let sources: WebSource[] | undefined

   244→   244→  /** Revelación progresiva (máquina de escribir adaptativa, duración acotada).
   245→   245→   *  Contenido estructurado (tablas/código): sin typewriter → render Markdown
   246→   246→   *  inmediato y bien formateado (nunca se ve la sintaxis cruda). */
   247→   247→  const startTypewriter = useCallback((assistantId: string, full: string) => {
   248→   248→    if (/```|(^|\n)\s*\|/.test(full)) {
   249→   249→      setMessages((prev) =>
   250→   250→        prev.map((m) => (m.id === assistantId ? { ...m, content: full, complete: true } : m)),
   251→   251→      )
   252→   252→      setPhase('idle')
   253→   253→      return
   254→   254→    }
   255→   255→    setPhase('typing')
   256→   256→    const duration = Math.min(2400, Math.max(700, full.length * 4))
   257→   257→    const start = performance.now()
   258→   258→    const step = (now: number) => {
   259→   259→      const p = Math.min(1, (now - start) / duration)
   260→   260→      const eased = 1 - Math.pow(1 - p, 2.1)
   261→   261→      const chars = Math.max(1, Math.floor(eased * full.length))
   262→   262→      setMessages((prev) =>
   263→   263→        prev.map((m) => (m.id === assistantId ? { ...m, content: full.slice(0, chars) } : m)),
   264→   264→      )
   265→   265→      if (p < 1) {
   266→   266→        rafRef.current = requestAnimationFrame(step)
   267→   267→      } else {
   268→   268→        setMessages((prev) =>
   269→   269→          prev.map((m) => (m.id === assistantId ? { ...m, content: full, complete: true } : m)),
   270→   270→        )
   271→   271→        setPhase('idle')
   272→   272→      }
   273→   273→    }
   274→   274→    rafRef.current = requestAnimationFrame(step)
   275→   275→  }, [])
   276→   276→
   277→   277→  /** Envía un mensaje al modelo unificado OE AI LLM (con búsqueda web opcional). */
   278→   278→  const send = useCallback(
   279→   279→    async (rawText: string, opts?: { isRetry?: boolean; webSearch?: boolean }) => {
   280→   280→      const content = rawText.trim()
   281→   281→      if (!content || pendingRef.current) return
   282→   282→      const mode = useUiStore.getState().mode
   283→   283→      const focus = useUiStore.getState().chatModel
   284→   284→      const webSearch = Boolean(opts?.webSearch)
   285→   285→      lastOptsRef.current = { webSearch }
   286→   286→
   287→   287→      // Conversación nueva si no hay una activa
   288→   288→      let convId = activeIdRef.current
   289→   289→      if (!convId) {
   290→   290→        convId = newId()
   291→   291→        activeIdRef.current = convId
   292→   292→        setActiveId(convId)
   293→   293→        try {
   294→   294→          window.localStorage.setItem(ACTIVE_KEY, convId)
   295→   295→        } catch {
   296→   296→          // sin almacenamiento
   297→   297→        }
   298→   298→      }
   299→   299→
   300→   300→      setError(null)
   301→   301→      setRouteNote(null)
   302→   302→      setPhase('thinking')
   303→   303→      setWebSearching(false)
   304→   304→      pendingRef.current = true
   305→   305→      if (!opts?.isRetry) {
   306→   306→        setLastUserMsg(content)
   307→   307→        setMessages((prev) => [
   308→   308→          ...prev,
   309→   309→          { id: `u-${newId()}`, role: 'user', content, fullContent: content, complete: true },
   310→   310→        ])
   311→   311→      }
   312→   312→
   313→   313→      const assistantId = `a-${newId()}`
   314→   314→      setMessages((prev) => [
   315→   315→        ...prev,
   316→   316→        { id: assistantId, role: 'assistant', content: '', fullContent: '', complete: false },
   317→   317→      ])
   318→   318→
   319→   319→      const controller = new AbortController()
   320→   320→      abortRef.current = controller
   321→   321→      let doneMeta: ChatMsgMeta | null = null
   322→   322→      let serverStreaming = false
   323→   323→
   324→   324→      const handleEvent = (event: string, data: unknown) => {
   325→   325→        if (event === 'meta') {
   326→   326→          const meta = data as { routeNote?: string; coreLabel?: string; mode?: 'instant' | 'reflective'; streaming?: boolean }
   327→   327→          serverStreaming = Boolean(meta.streaming)
   328→   328→          if (meta.routeNote) setRouteNote(meta.routeNote)
   329→   329→          setMessages((prev) =>
   330→   330→            prev.map((m) =>
   331→   331→              m.id === assistantId
   332→   332→                ? { ...m, meta: { ...m.meta, coreLabel: meta.coreLabel ?? null, routeNote: meta.routeNote ?? null, mode: meta.mode, streaming: serverStreaming } }
   333→   333→                : m,
   334→   334→            ),
   335→   335→          )
   336→   336→        } else if (event === 'stage') {
   337→   337→          const { stage } = data as { stage?: string }
   338→   338→          if (stage === 'web_search') setWebSearching(true)
   339→   339→        } else if (event === 'search') {
   340→   340→          const { sources } = data as { sources?: WebSource[] }
   341→   341→          setWebSearching(false)
   342→   342→          setMessages((prev) =>
   343→   343→            prev.map((m) => (m.id === assistantId ? { ...m, sources: Array.isArray(sources) ? sources : [] } : m)),
   344→   344→          )
   345→   345→        } else if (event === 'search_error') {
   346→   346→          const { message } = data as { message?: string }
   347→   347→          setWebSearching(false)
   348→   348→          setMessages((prev) =>
   349→   349→            prev.map((m) => (m.id === assistantId ? { ...m, searchNote: message ?? 'La búsqueda web no estuvo disponible.' } : m)),
   350→   350→          )
   351→   351→        } else if (event === 'delta') {
   352→   352→          const { text } = data as { text: string }
   353→   353→          if (serverStreaming) {
   354→   354→            // Streaming real del proveedor: cada delta es un incremento.
   355→   355→            setPhase('typing')
   356→   356→            setMessages((prev) =>
   357→   357→              prev.map((m) => {
   358→   358→                if (m.id !== assistantId) return m
   359→   359→                const full = m.fullContent + text
   360→   360→                return { ...m, fullContent: full, content: full }
   361→   361→              }),
   362→   362→            )
   363→   363→          } else {
   364→   364→            // Respuesta única: revelación progresiva local (efecto de presentación).
   365→   365→            setMessages((prev) =>
   366→   366→              prev.map((m) => (m.id === assistantId ? { ...m, fullContent: text } : m)),
   367→   367→            )
   368→   368→            startTypewriter(assistantId, text)
   369→   369→          }
   370→   370→        } else if (event === 'done') {
   371→   371→          const d = data as { latencyMs?: number; providerReportedModel?: string | null; webSearchUsed?: boolean }
   372→   372→          doneMeta = {
   373→   373→            latencyMs: d.latencyMs ?? null,
   374→   374→            providerReportedModel: d.providerReportedModel ?? null,
   375→   375→            webSearchUsed: Boolean(d.webSearchUsed),
   376→   376→          }
   377→   377→          if (serverStreaming) {
   378→   378→            setMessages((prev) =>
   379→   379→              prev.map((m) => (m.id === assistantId ? { ...m, content: m.fullContent, complete: true } : m)),
   380→   380→            )
   381→   381→            setPhase('idle')
   382→   382→          }
   383→   383→        } else if (event === 'error') {
   384→   384→          const e = data as { message: string; errorType?: string }
   385→   385→          setError({ message: e.message, errorType: e.errorType })
   386→   386→          // Si ya había contenido parcial, se conserva completo; si no, se retira.
   387→   387→          setMessages((prev) =>
   388→   388→            prev
   389→   389→              .map((m) => (m.id === assistantId && m.fullContent ? { ...m, content: m.fullContent, complete: true } : m))
   390→   390→              .filter((m) => !(m.id === assistantId && !m.fullContent)),
   391→   391→          )
   392→   392→          setPhase('idle')
   393→   393→        }
   394→   394→      }
   395→   395→
   396→   396→      try {
   397→   397→        const token = await auth.getAccessToken()
   398→   398→        const res = await fetch('/api/chat', {
   399→   399→          method: 'POST',
   400→   400→          headers: {
   401→   401→            'Content-Type': 'application/json',
   402→   402→            ...(token ? { Authorization: `Bearer ${token}` } : {}),
   403→   403→          },
   404→   404→          body: JSON.stringify({
   405→   405→            sessionId: convId,
   406→   406→            message: content,
   407→   407→            model: focus && focus !== 'auto' ? focus : undefined,
   408→   408→            allowFallback: true,
   409→   409→            mode,
   410→   410→            webSearch,
   411→   411→          }),
   412→   412→          signal: controller.signal,
   413→   413→        })
   414→   414→
   415→   415→        if (!res.ok || !res.body) {
   416→   416→          const data = (await res.json().catch(() => null)) as { error?: string; errorType?: string } | null
   417→   417→          throw Object.assign(new Error(data?.error ?? `Error ${res.status} al iniciar la conversación.`), {
   418→   418→            errorType: data?.errorType,
   419→   419→          })
   420→   420→        }
   421→   421→
   422→   422→        const reader = res.body.getReader()
   423→   423→        const decoder = new TextDecoder()
   424→   424→        let buffer = ''
   425→   425→
   426→   426→        while (true) {
   427→   427→          const { done, value } = await reader.read()
   428→   428→          if (done) break
   429→   429→          buffer += decoder.decode(value, { stream: true })
   430→   430→          let sep: number
   431→   431→          while ((sep = buffer.indexOf('\n\n')) !== -1) {
   432→   432→            const rawEvent = buffer.slice(0, sep)
   433→   433→            buffer = buffer.slice(sep + 2)
   434→   434→            let eventName = 'message'
   435→   435→            let dataRaw = ''
   436→   436→            for (const line of rawEvent.split('\n')) {
   437→   437→              if (line.startsWith('event:')) eventName = line.slice(6).trim()
   438→   438→              else if (line.startsWith('data:')) dataRaw += line.slice(5).trim()
   439→   439→            }
   440→   440→            if (dataRaw) {
   441→   441→              try {
   442→   442→                handleEvent(eventName, JSON.parse(dataRaw))
   443→   443→              } catch {
   444→   444→                // evento malformado: ignorar de forma segura
   445→   445→              }
   446→   446→            }
   447→   447→          }
   448→   448→        }
   449→   449→
   450→   450→        // Adjuntar metadatos finales (latencia, modelo reportado)
   451→   451→        if (doneMeta) {
   452→   452→          const meta = doneMeta as ChatMsgMeta
   453→   453→          setMessages((prev) => prev.map((m) => (m.id === assistantId ? { ...m, meta: { ...m.meta, ...meta } } : m)))
   454→   454→        }
   455→   455→        void refreshConversations()
   456→   456→        void refreshQuota()
   457→   457→      } catch (err) {
   458→   458→        if ((err as Error).name === 'AbortError') {
   459→   459→          // Cancelación: si ya hay texto, se revela completo; si no, se retira el marcador.
   460→   460→          setMessages((prev) =>
   461→   461→            prev
   462→   462→              .map((m) => (m.id === assistantId && m.fullContent ? { ...m, content: m.fullContent, complete: true } : m))
   463→   463→              .filter((m) => !(m.id === assistantId && !m.fullContent)),
   464→   464→          )
   465→   465→        } else {
   466→   466→          const e = err as Error & { errorType?: string }
   467→   467→          if (e.errorType === 'guest_limit_reached') {
   468→   468→            // Cuota agotada: se activa la compuerta de inicio de sesión (sin error genérico).
   469→   469→            setGuestBlocked(true)
   470→   470→            void refreshQuota()
   471→   471→          } else {
   472→   472→            setError({ message: e.message, errorType: e.errorType })
   473→   473→          }
   474→   474→          setMessages((prev) => prev.filter((m) => m.id !== assistantId))
   475→   475→        }
   476→   476→      } finally {
   477→   477→        abortRef.current = null
   478→   478→        pendingRef.current = false
   479→   479→        setWebSearching(false)
   480→   480→        if (!rafRef.current) setPhase('idle')
   481→   481→      }
   482→   482→    },
   483→   483→    [auth, refreshConversations, refreshQuota, startTypewriter],
   484→   484→  )
   485→   485→
   486→   486→  /** Genera una imagen con la herramienta del núcleo y la inserta en la conversación. */
   487→   487→  const generateImage = useCallback(
   488→   488→    async (rawPrompt: string) => {
   489→   489→      const prompt = rawPrompt.trim()
   490→   490→      if (!prompt || pendingRef.current) return
   491→   491→
   492→   492→      let convId = activeIdRef.current
   493→   493→      if (!convId) {
   494→   494→        convId = newId()
   495→   495→        activeIdRef.current = convId
   496→   496→        setActiveId(convId)
   497→   497→        try {
   498→   498→          window.localStorage.setItem(ACTIVE_KEY, convId)
   499→   499→        } catch {
   500→   500→          // sin almacenamiento
   501→   501→        }
   502→   502→      }
   503→   503→
   504→   504→      setError(null)
   505→   505→      setRouteNote(null)
   506→   506→      setPhase('thinking')
   507→   507→      setImagePending(true)
   508→   508→      pendingRef.current = true
   509→   509→      setLastUserMsg(null) // la regeneración de texto no aplica tras una imagen
   510→   510→      setMessages((prev) => [
   511→   511→        ...prev,
   512→   512→        { id: `u-${newId()}`, role: 'user', content: prompt, fullContent: prompt, complete: true },
   513→   513→      ])
   514→   514→
   515→   515→      const placeholderId = `img-${newId()}`
   516→   516→      setMessages((prev) => [
   517→   517→        ...prev,
   518→   518→        { id: placeholderId, role: 'assistant', content: '', fullContent: '', complete: false, kind: 'image', pendingImage: true },
   519→   519→      ])
   520→   520→
   521→   521→      const controller = new AbortController()
   522→   522→      abortRef.current = controller
   523→   523→
   524→   524→      try {
   525→   525→        const token = await auth.getAccessToken()
   526→   526→        const res = await fetch('/api/tools/image', {
   527→   527→          method: 'POST',
   528→   528→          headers: {
   529→   529→            'Content-Type': 'application/json',
   530→   530→            ...(token ? { Authorization: `Bearer ${token}` } : {}),
   531→   531→          },
   532→   532→          body: JSON.stringify({ sessionId: convId, prompt }),
   533→   533→          signal: controller.signal,
   534→   534→        })
   535→   535→        const data = (await res.json().catch(() => null)) as {
   536→   536→          url?: string
   537→   537→          latencyMs?: number
   538→   538→          error?: string
   539→   539→          errorType?: string
   540→   540→        } | null
   541→   541→        if (!res.ok || !data?.url) {
   542→   542→          throw Object.assign(new Error(data?.error ?? `No se pudo generar la imagen (error ${res.status}).`), {
   543→   543→            errorType: data?.errorType,
   544→   544→          })
   545→   545→        }
   546→   546→        setMessages((prev) =>
   547→   547→          prev.map((m) =>
   548→   548→            m.id === placeholderId
   549→   549→              ? {
   550→   550→                  ...m,
   551→   551→                  content: prompt,
   552→   552→                  fullContent: prompt,
   553→   553→                  assetUrl: data.url ?? null,
   554→   554→                  pendingImage: false,
   555→   555→                  complete: true,
   556→   556→                  meta: { latencyMs: data.latencyMs ?? null },
   557→   557→                }
   558→   558→              : m,
   559→   559→          ),
   560→   560→        )
   561→   561→        void refreshConversations()
   562→   562→        void refreshQuota()
   563→   563→      } catch (err) {
   564→   564→        setMessages((prev) => prev.filter((m) => m.id !== placeholderId))
   565→   565→        if ((err as Error).name !== 'AbortError') {
   566→   566→          const e = err as Error & { errorType?: string }
   567→   567→          if (e.errorType === 'guest_limit_reached') {
   568→   568→            setGuestBlocked(true)
   569→   569→            void refreshQuota()
   570→   570→          } else {
   571→   571→            setError({ message: e.message, errorType: e.errorType })
   572→   572→          }
   573→   573→        }
   574→   574→      } finally {
   575→   575→        abortRef.current = null
   576→   576→        pendingRef.current = false
   577→   577→        setImagePending(false)
   578→   578→        setPhase('idle')
   579→   579→      }
   580→   580→    },
   581→   581→    [auth, refreshConversations, refreshQuota],
   582→   582→  )
   583→   583→
   584→   584→  /** Detiene la generación, la revelación o la herramienta en curso. */
   585→   585→  const stop = useCallback(() => {
   586→   586→    abortRef.current?.abort()
   587→   587→    abortRef.current = null
   588→   588→    pendingRef.current = false
   589→   589→    if (rafRef.current !== null) {
   590→   590→      cancelAnimationFrame(rafRef.current)
   591→   591→      rafRef.current = null
   592→   592→    }
   593→   593→    setPhase('idle')
   594→   594→    setWebSearching(false)
   595→   595→    setImagePending(false)
   596→   596→    setMessages((prev) => prev.map((m) => (m.complete || m.pendingImage ? m : { ...m, content: m.fullContent || m.content, complete: true })))
   597→   597→  }, [])
   598→   598→
   599→   599→  /** Regenera la última respuesta. */
   600→   600→  const regenerate = useCallback(() => {
   601→   601→    if (!lastUserMsg || pendingRef.current) return
   602→   602→    // Se elimina la última respuesta asistente incompleta o final
   603→   603→    setMessages((prev) => {
   604→   604→      const idx = prev.map((m) => m.role).lastIndexOf('assistant')
   605→   605→      return idx === -1 ? prev : prev.filter((_, i) => i !== idx)
   606→   606→    })
   607→   607→    void send(lastUserMsg, { isRetry: true, webSearch: lastOptsRef.current.webSearch })
   608→   608→  }, [lastUserMsg, send])
   609→   609→
   610→   610→  /**
   611→   611→   * Edita un mensaje del usuario: recorta la rama posterior (servidor y cliente)
   612→   612→   * y reenvía el texto corregido como mensaje nuevo.
   613→   613→   */
   614→   614→  const editMessage = useCallback(
   615→   615→    async (messageId: string, newContent: string) => {
   616→   616→      const text = newContent.trim()
   617→   617→      if (!text || pendingRef.current) return
   618→   618→      const idx = messages.findIndex((m) => m.id === messageId)
   619→   619→      if (idx === -1 || messages[idx]?.role !== 'user') return
   620→   620→      const convId = activeIdRef.current
   621→   621→      if (!convId) return
   622→   622→
   623→   623→      // Recorte en servidor: se conservan solo los mensajes anteriores a la edición.
   624→   624→      try {
   625→   625→        await fetch('/api/chat/trim', {
   626→   626→          method: 'POST',
   627→   627→          headers: { 'Content-Type': 'application/json' },
   628→   628→          body: JSON.stringify({ sessionId: convId, keepCount: idx }),
   629→   629→        })
   630→   630→      } catch {
   631→   631→        // si el recorte falla se continúa: la conversación local manda en la vista
   632→   632→      }
   633→   633→      setMessages((prev) => prev.slice(0, idx))
   634→   634→      void send(text)
   635→   635→    },
   636→   636→    [messages, send],
   637→   637→  )
   638→   638→
   639→   639→  /** Renombra una conversación (PATCH) y actualiza sidebar/buscador. */
   640→   640→  const renameConversation = useCallback(
   641→   641→    async (id: string, title: string) => {
   642→   642→      const clean = title.trim().slice(0, 80)
   643→   643→      if (!clean) return
   644→   644→      setConversations((prev) => prev.map((c) => (c.id === id ? { ...c, title: clean } : c)))
   645→   645→      useUiStore.setState((s) => ({ conversations: s.conversations.map((c) => (c.id === id ? { ...c, title: clean } : c)) }))
   646→   646→      try {
   647→   647→        await fetch(`/api/chat/sessions/${encodeURIComponent(id)}`, {
   648→   648→          method: 'PATCH',
   649→   649→          headers: { 'Content-Type': 'application/json' },
   650→   650→          body: JSON.stringify({ title: clean }),
   651→   651→        })
   652→   652→      } catch {
   653→   653→        // el título local se mantiene; el servidor quedará coherente al refrescar
   654→   654→      }
   655→   655→      void refreshConversations()
   656→   656→    },
   657→   657→    [refreshConversations],
   658→   658→  )
   659→   659→
   660→   660→  /** Exporta la conversación activa como Markdown descargable. */
   661→   661→  const exportActiveConversation = useCallback(() => {
   662→   662→    if (messages.length === 0) return
   663→   663→    const title = conversations.find((c) => c.id === activeId)?.title ?? 'Conversación OE AI'
   664→   664→    exportMessagesAsMarkdown(title, messages)
   665→   665→  }, [activeId, conversations, messages])
   666→   666→
   667→   667→  /** Nueva conversación (Ctrl+Shift+O). */
   668→   668→  const newChat = useCallback(() => {
   669→   669→    if (pendingRef.current) abortRef.current?.abort()
   670→   670→    activeIdRef.current = null
   671→   671→    setActiveId(null)
   672→   672→    setMessages([])
   673→   673→    setError(null)
   674→   674→    setRouteNote(null)
   675→   675→    setLastUserMsg(null)
   676→   676→    setFocusModel(null)
   677→   677→    setWebSearching(false)
   678→   678→    setImagePending(false)
   679→   679→    try {
   680→   680→      window.localStorage.removeItem(ACTIVE_KEY)
   681→   681→      window.sessionStorage.removeItem('oe-ai-pending-model')
   682→   682→      useUiStore.setState({ chatModel: 'auto' })
   683→   683→    } catch {
   684→   684→      // sin almacenamiento
   685→   685→    }
   686→   686→  }, [])
   687→   687→
   688→   688→  /** Elimina una conversación del historial. */
   689→   689→  const deleteConversation = useCallback(
   690→   690→    async (id: string) => {
   691→   691→      try {
   692→   692→        await fetch(`/api/chat/sessions/${encodeURIComponent(id)}`, { method: 'DELETE' })
   693→   693→      } catch {
   694→   694→        // si falla, la lista se refresca igualmente
   695→   695→      }
   696→   696→      if (activeIdRef.current === id) newChat()
   697→   697→      void refreshConversations()
   698→   698→    },
   699→   699→    [newChat, refreshConversations],
   700→   700→  )
   701→   701→
   702→   702→  // Limpieza al desmontar
   703→   703→  useEffect(() => {
   704→   704→    return () => {
   705→   705→      if (rafRef.current !== null) cancelAnimationFrame(rafRef.current)
   706→   706→      abortRef.current?.abort()
   707→   707→    }
   708→   708→  }, [])
   709→   709→
   710→   710→  const pending = phase !== 'idle'
   711→   711→
   712→   712→  return {
   713→   713→    conversations,
   714→   714→    activeId,
   715→   715→    messages,
   716→   716→    phase,
   717→   717→    pending,
   718→   718→    error,
   719→   719→    routeNote,
   720→   720→    focusModel,
   721→   721→    loadingHistory,
   722→   722→    lastUserMsg,
   723→   723→    webSearching,
   724→   724→    imagePending,
   725→   725→    guestQuota,
   726→   726→    guestBlocked,
   727→   727→    refreshQuota,
   728→   728→    send,
   729→   729→    stop,
   730→   730→    regenerate,
   731→   731→    newChat,
   732→   732→    loadConversation,
   733→   733→    deleteConversation,
   734→   734→    refreshConversations,
   735→   735→    setFocusModel,
   736→   736→    generateImage,
   737→   737→    editMessage,
   738→   738→    renameConversation,
   739→   739→    exportActiveConversation,
   740→   740→  }
   741→   741→}
