     3→/**
     4→ * useOeChat — lógica del chat premium con OE AI LLM.
     5→ *
     6→ * - Comunica con POST /api/chat (SSE: meta → stage/search → delta → done | error).
     7→ * - El backend entrega la respuesta completa (streaming de tokens no verificado);
     8→ *   la revelación progresiva es un efecto de presentación local (máquina de
     9→ *   escribir adaptativa) para una experiencia fluida, sin fingir datos.
    10→ * - Gestiona historial multi-conversación vía /api/chat/sessions y
    11→ *   /api/chat/history; herramientas: imágenes (/api/tools/image), búsqueda web
    12→ *   real con fuentes, edición de mensajes (recorte), renombrar y exportar.
    13→ */

    81→function newId(): string {
    82→  if (typeof crypto !== 'undefined' && 'randomUUID' in crypto) return crypto.randomUUID()
    83→  return `${Date.now()}-${Math.random().toString(36).slice(2)}-${Math.random().toString(36).slice(2)}`
    84→}

   102→  const rafRef = useRef<number | null>(null)

   104→  const pendingRef = useRef(false)

   112→  /** Estado de la cuota (GET /api/chat/quota): nula si hay sesión iniciada. */
   113→  const refreshQuota = useCallback(async () => {
   114→    try {
   115→      const res = await fetch('/api/chat/quota', { cache: 'no-store' })

   117→      const data = (await res.json()) as {
   118→        authenticated: boolean
   119→        used: number | null
   120→        limit: number | null
   121→        remaining: number | null
   122→        exhausted?: boolean
   123→      }
   124→      if (data.authenticated) {
   125→        setGuestQuota(null)
   126→        setGuestBlocked(false)
   127→        return
   128→      }

   131→        limit: data.limit ?? 5,
   132→        remaining: data.remaining ?? 0,

   141→

   174→  /** Carga los mensajes de una conversación existente (texto e imágenes). */
   175→  const loadConversation = useCallback(async (id: string) => {
   176→    if (pendingRef.current) return
   177→    setError(null)
   178→    setRouteNote(null)
   179→    setLoadingHistory(true)
   180→    try {
   181→      const res = await fetch(`/api/chat/history?sessionId=${encodeURIComponent(id)}`)
   182→      if (!res.ok) return
   183→      const data = (await res.json()) as { messages?: Array<Record<string, unknown>> }
   184→      const restored: ChatMsg[] = (data.messages ?? []).map((m) => {
   185→        const kind = m.kind === 'image' ? 'image' : 'text'
   186→        const role = m.role === 'user' ? ('user' as const) : ('assistant' as const)
   187→        let sources: WebSource[] | undefined
   188→        if (typeof m.sourcesJson === 'string' && m.sourcesJson) {
   189→          try {
   190→            const parsed = JSON.parse(m.sourcesJson) as WebSource[]

   192→          } catch {
   193→            // JSON inválido: se ignora
   194→          }
   195→        }
   196→        return {
   197→          id: String(m.id),
   198→          role,
   199→          content: String(m.content ?? ''),
   200→          fullContent: String(m.content ?? ''),
   201→          complete: true,
   202→          kind,
   203→          assetUrl: kind === 'image' && typeof m.assetId === 'string' ? `/api/assets/${m.assetId}` : null,
   204→          attachmentUrl: role === 'user' && typeof m.attachmentAssetId === 'string' ? `/api/assets/${m.attachmentAssetId}` : null,
   205→          sources,
   206→          meta: {
   207→            providerReportedModel: (m.providerReportedModel as string) ?? null,
   208→            routeNote: (m.resolvedModelId as string) ?? null,

   210→        }
   211→      })

   213→      setActiveId(id)

   217→      } catch {

   223→  }, [])

   229→    try {
   230→      initialId = window.localStorage.getItem(ACTIVE_KEY)

   233→    }

   250→  /** Revelación progresiva (máquina de escribir adaptativa, duración acotada).
   251→   *  Contenido estructurado (tablas/código): sin typewriter → render Markdown
   252→   *  inmediato y bien formateado (nunca se ve la sintaxis cruda). */
   253→  const startTypewriter = useCallback((assistantId: string, full: string) => {
   254→    if (/```|(^|\n)\s*\|/.test(full)) {
   255→      setMessages((prev) =>
   256→        prev.map((m) => (m.id === assistantId ? { ...m, content: full, complete: true } : m)),
   257→      )
   258→      setPhase('idle')
   259→      return
   260→    }
   261→    setPhase('typing')
   262→    const duration = Math.min(2400, Math.max(700, full.length * 4))
   263→    const start = performance.now()
   264→    const step = (now: number) => {
   265→      const p = Math.min(1, (now - start) / duration)
   266→      const eased = 1 - Math.pow(1 - p, 2.1)
   267→      const chars = Math.max(1, Math.floor(eased * full.length))
   268→      setMessages((prev) =>
   269→        prev.map((m) => (m.id === assistantId ? { ...m, content: full.slice(0, chars) } : m)),
   270→      )
   271→      if (p < 1) {
   272→        rafRef.current = requestAnimationFrame(step)
   273→      } else {
   274→        setMessages((prev) =>
   275→          prev.map((m) => (m.id === assistantId ? { ...m, content: full, complete: true } : m)),
   276→        )
   277→        setPhase('idle')
   278→      }
   279→    }
   280→    rafRef.current = requestAnimationFrame(step)
   281→  }, [])
   282→
   283→  /** Envía un mensaje al modelo unificado OE AI LLM (con búsqueda web y/o imagen adjunta opcional). */
   284→  const send = useCallback(
   285→    async (
   286→      rawText: string,
   287→      opts?: {
   288→        isRetry?: boolean
   289→        webSearch?: boolean
   290→        /** Imagen adjuntada (visión): el núcleo la analiza junto al texto. */
   291→        attachment?: { assetId: string; previewUrl: string }
   292→      },
   293→    ) => {
   294→      const content = rawText.trim()
   295→      if (!content || pendingRef.current) return
   296→      const mode = useUiStore.getState().mode
   297→      const focus = useUiStore.getState().chatModel
   298→      const webSearch = Boolean(opts?.webSearch)
   299→      const attachment = opts?.attachment
   300→      lastOptsRef.current = { webSearch, attachment }
   301→
   302→      // Conversación nueva si no hay una activa
   303→      let convId = activeIdRef.current
   304→      if (!convId) {
   305→        convId = newId()
   306→        activeIdRef.current = convId
   307→        setActiveId(convId)
   308→        try {
   309→          window.localStorage.setItem(ACTIVE_KEY, convId)
   310→        } catch {
   311→          // sin almacenamiento
   312→        }
   313→      }
   314→
   315→      setError(null)
   316→      setRouteNote(null)
   317→      setPhase('thinking')
   318→      setWebSearching(false)
   319→      pendingRef.current = true
   320→      if (!opts?.isRetry) {
   321→        setLastUserMsg(content)
   322→        setMessages((prev) => [
   323→          ...prev,
   324→          {
   325→            id: `u-${newId()}`,
   326→            role: 'user',
   327→            content,
   328→            fullContent: content,
   329→            complete: true,
   330→            attachmentUrl: attachment?.previewUrl ?? null,
   331→          },
   332→        ])
   333→      }
   334→
   335→      const assistantId = `a-${newId()}`
   336→      setMessages((prev) => [
   337→        ...prev,
   338→        { id: assistantId, role: 'assistant', content: '', fullContent: '', complete: false },
   339→      ])
   340→
   341→      const controller = new AbortController()
   342→      abortRef.current = controller
   343→      let doneMeta: ChatMsgMeta | null = null
   344→      let serverStreaming = false
   345→
   346→      const handleEvent = (event: string, data: unknown) => {
   347→        if (event === 'meta') {
   348→          const meta = data as { routeNote?: string; coreLabel?: string; mode?: 'instant' | 'reflective'; streaming?: boolean }
   349→          serverStreaming = Boolean(meta.streaming)
   350→          if (meta.routeNote) setRouteNote(meta.routeNote)
   351→          setMessages((prev) =>
   352→            prev.map((m) =>
   353→              m.id === assistantId
   354→                ? { ...m, meta: { ...m.meta, coreLabel: meta.coreLabel ?? null, routeNote: meta.routeNote ?? null, mode: meta.mode, streaming: serverStreaming } }
   355→                : m,
   356→            ),
   357→          )
   358→        } else if (event === 'stage') {
   359→          const { stage } = data as { stage?: string }
   360→          if (stage === 'web_search') setWebSearching(true)
   361→        } else if (event === 'search') {
   362→          const { sources } = data as { sources?: WebSource[] }
   363→          setWebSearching(false)
   364→          setMessages((prev) =>
   365→            prev.map((m) => (m.id === assistantId ? { ...m, sources: Array.isArray(sources) ? sources : [] } : m)),
   366→          )
   367→        } else if (event === 'search_error') {
   368→          const { message } = data as { message?: string }
   369→          setWebSearching(false)
   370→          setMessages((prev) =>
   371→            prev.map((m) => (m.id === assistantId ? { ...m, searchNote: message ?? 'La búsqueda web no estuvo disponible.' } : m)),
   372→          )
   373→        } else if (event === 'delta') {
   374→          const { text } = data as { text: string }
   375→          if (serverStreaming) {
   376→            // Streaming real del proveedor: cada delta es un incremento.
   377→            setPhase('typing')
   378→            setMessages((prev) =>
   379→              prev.map((m) => {
   380→                if (m.id !== assistantId) return m
   381→                const full = m.fullContent + text
   382→                return { ...m, fullContent: full, content: full }
   383→              }),
   384→            )
   385→          } else {
   386→            // Respuesta única: revelación progresiva local (efecto de presentación).
   387→            setMessages((prev) =>
   388→              prev.map((m) => (m.id === assistantId ? { ...m, fullContent: text } : m)),
   389→            )
   390→            startTypewriter(assistantId, text)
   391→          }
   392→        } else if (event === 'done') {
   393→          const d = data as { latencyMs?: number; providerReportedModel?: string | null; webSearchUsed?: boolean; visionUsed?: boolean }
   394→          doneMeta = {
   395→            latencyMs: d.latencyMs ?? null,
   396→            providerReportedModel: d.providerReportedModel ?? null,
   397→            webSearchUsed: Boolean(d.webSearchUsed),
   398→            visionUsed: Boolean(d.visionUsed),
   399→          }
   400→          if (serverStreaming) {
   401→            setMessages((prev) =>
   402→              prev.map((m) => (m.id === assistantId ? { ...m, content: m.fullContent, complete: true } : m)),
   403→            )
   404→            setPhase('idle')
   405→          }
   406→        } else if (event === 'error') {
   407→          const e = data as { message: string; errorType?: string }
   408→          setError({ message: e.message, errorType: e.errorType })
   409→          // Si ya había contenido parcial, se conserva completo; si no, se retira.
   410→          setMessages((prev) =>
   411→            prev
   412→              .map((m) => (m.id === assistantId && m.fullContent ? { ...m, content: m.fullContent, complete: true } : m))
   413→              .filter((m) => !(m.id === assistantId && !m.fullContent)),
   414→          )
   415→          setPhase('idle')
   416→        }
   417→      }
   418→
   419→      try {
   420→        const token = await auth.getAccessToken()
   421→        const res = await fetch('/api/chat', {
   422→          method: 'POST',
   423→          headers: {
   424→            'Content-Type': 'application/json',
   425→            ...(token ? { Authorization: `Bearer ${token}` } : {}),
   426→          },
   427→          body: JSON.stringify({
   428→            sessionId: convId,
   429→            message: content,
   430→            model: focus && focus !== 'auto' ? focus : undefined,
   431→            allowFallback: true,
   432→            mode,
   433→            webSearch,
   434→            attachmentId: attachment?.assetId,
   435→          }),
   436→          signal: controller.signal,
   437→        })
   438→
   439→        if (!res.ok || !res.body) {
   440→          const data = (await res.json().catch(() => null)) as { error?: string; errorType?: string } | null
   441→          throw Object.assign(new Error(data?.error ?? `Error ${res.status} al iniciar la conversación.`), {
   442→            errorType: data?.errorType,
   443→          })
   444→        }
   445→
   446→        const reader = res.body.getReader()
   447→        const decoder = new TextDecoder()
   448→        let buffer = ''
   449→
   450→        while (true) {
   451→          const { done, value } = await reader.read()
   452→          if (done) break
   453→          buffer += decoder.decode(value, { stream: true })
   454→          let sep: number
   455→          while ((sep = buffer.indexOf('\n\n')) !== -1) {
   456→            const rawEvent = buffer.slice(0, sep)
   457→            buffer = buffer.slice(sep + 2)
   458→            let eventName = 'message'
   459→            let dataRaw = ''
   460→            for (const line of rawEvent.split('\n')) {
   461→              if (line.startsWith('event:')) eventName = line.slice(6).trim()
   462→              else if (line.startsWith('data:')) dataRaw += line.slice(5).trim()
   463→            }
   464→            if (dataRaw) {
   465→              try {
   466→                handleEvent(eventName, JSON.parse(dataRaw))
   467→              } catch {
   468→                // evento malformado: ignorar de forma segura
   469→              }
   470→            }
   471→          }
   472→        }
   473→
   474→        // Adjuntar metadatos finales (latencia, modelo reportado)
   475→        if (doneMeta) {
   476→          const meta = doneMeta as ChatMsgMeta
   477→          setMessages((prev) => prev.map((m) => (m.id === assistantId ? { ...m, meta: { ...m.meta, ...meta } } : m)))
   478→        }
   479→        void refreshConversations()
   480→        void refreshQuota()
   481→      } catch (err) {
   482→        if ((err as Error).name === 'AbortError') {
   483→          // Cancelación: si ya hay texto, se revela completo; si no, se retira el marcador.
   484→          setMessages((prev) =>
   485→            prev
   486→              .map((m) => (m.id === assistantId && m.fullContent ? { ...m, content: m.fullContent, complete: true } : m))
   487→              .filter((m) => !(m.id === assistantId && !m.fullContent)),
   488→          )
   489→        } else {
   490→          const e = err as Error & { errorType?: string }
   491→          if (e.errorType === 'guest_limit_reached') {
   492→            // Cuota agotada: se activa la compuerta de inicio de sesión (sin error genérico).
   493→            setGuestBlocked(true)
   494→            void refreshQuota()
   495→          } else {
   496→            setError({ message: e.message, errorType: e.errorType })
   497→          }
   498→          setMessages((prev) => prev.filter((m) => m.id !== assistantId))
   499→        }
   500→      } finally {
   501→        abortRef.current = null
   502→        pendingRef.current = false
   503→        setWebSearching(false)
   504→        if (!rafRef.current) setPhase('idle')
   505→      }
   506→    },
   507→    [auth, refreshConversations, refreshQuota, startTypewriter],
   508→  )

   510→  /** Genera una imagen con la herramienta del núcleo y la inserta en la conversación. */
   511→  const generateImage = useCallback(
   512→    async (rawPrompt: string) => {
   513→      const prompt = rawPrompt.trim()
   514→      if (!prompt || pendingRef.current) return
   515→
   516→      let convId = activeIdRef.current
   517→      if (!convId) {
   518→        convId = newId()
   519→        activeIdRef.current = convId
   520→        setActiveId(convId)
   521→        try {
   522→          window.localStorage.setItem(ACTIVE_KEY, convId)
   523→        } catch {
   524→          // sin almacenamiento
   525→        }
   526→      }
   527→
   528→      setError(null)
   529→      setRouteNote(null)
   530→      setPhase('thinking')
   531→      setImagePending(true)
   532→      pendingRef.current = true
   533→      setLastUserMsg(null) // la regeneración de texto no aplica tras una imagen
   534→      setMessages((prev) => [
   535→        ...prev,
   536→        { id: `u-${newId()}`, role: 'user', content: prompt, fullContent: prompt, complete: true },
   537→      ])
   538→
   539→      const placeholderId = `img-${newId()}`
   540→      setMessages((prev) => [
   541→        ...prev,
   542→        { id: placeholderId, role: 'assistant', content: '', fullContent: '', complete: false, kind: 'image', pendingImage: true },
   543→      ])
   544→
   545→      const controller = new AbortController()
   546→      abortRef.current = controller
   547→
   548→      try {
   549→        const token = await auth.getAccessToken()
   550→        const res = await fetch('/api/tools/image', {
   551→          method: 'POST',
   552→          headers: {
   553→            'Content-Type': 'application/json',
   554→            ...(token ? { Authorization: `Bearer ${token}` } : {}),
   555→          },
   556→          body: JSON.stringify({ sessionId: convId, prompt }),
   557→          signal: controller.signal,
   558→        })
   559→        const data = (await res.json().catch(() => null)) as {
   560→          url?: string
   561→          latencyMs?: number
   562→          error?: string
   563→          errorType?: string
   564→        } | null
   565→        if (!res.ok || !data?.url) {
   566→          throw Object.assign(new Error(data?.error ?? `No se pudo generar la imagen (error ${res.status}).`), {
   567→            errorType: data?.errorType,
   568→          })
   569→        }
   570→        setMessages((prev) =>
   571→          prev.map((m) =>
   572→            m.id === placeholderId
   573→              ? {
   574→                  ...m,
   575→                  content: prompt,
   576→                  fullContent: prompt,
   577→                  assetUrl: data.url ?? null,
   578→                  pendingImage: false,
   579→                  complete: true,
   580→                  meta: { latencyMs: data.latencyMs ?? null },
   581→                }
   582→              : m,
   583→          ),
   584→        )
   585→        void refreshConversations()
   586→        void refreshQuota()
   587→      } catch (err) {
   588→        setMessages((prev) => prev.filter((m) => m.id !== placeholderId))
   589→        if ((err as Error).name !== 'AbortError') {
   590→          const e = err as Error & { errorType?: string }
   591→          if (e.errorType === 'guest_limit_reached') {
   592→            setGuestBlocked(true)
   593→            void refreshQuota()
   594→          } else {
   595→            setError({ message: e.message, errorType: e.errorType })
   596→          }
   597→        }
   598→      } finally {
   599→        abortRef.current = null
   600→        pendingRef.current = false
   601→        setImagePending(false)
   602→        setPhase('idle')
   603→      }
   604→    },
   605→    [auth, refreshConversations, refreshQuota],
   606→  )
   607→
   608→  /** Detiene la generación, la revelación o la herramienta en curso. */
   609→  const stop = useCallback(() => {
   610→    abortRef.current?.abort()
   611→    abortRef.current = null
   612→    pendingRef.current = false
   613→    if (rafRef.current !== null) {
   614→      cancelAnimationFrame(rafRef.current)
   615→      rafRef.current = null
   616→    }
   617→    setPhase('idle')
   618→    setWebSearching(false)
   619→    setImagePending(false)
   620→    setMessages((prev) => prev.map((m) => (m.complete || m.pendingImage ? m : { ...m, content: m.fullContent || m.content, complete: true })))
   621→  }, [])
   622→
   623→  /** Regenera la última respuesta (conserva la búsqueda web y la imagen adjunta del turno). */
   624→  const regenerate = useCallback(() => {
   625→    if (!lastUserMsg || pendingRef.current) return
   626→    // Se elimina la última respuesta asistente incompleta o final
   627→    setMessages((prev) => {
   628→      const idx = prev.map((m) => m.role).lastIndexOf('assistant')
   629→      return idx === -1 ? prev : prev.filter((_, i) => i !== idx)
   630→    })
   631→    void send(lastUserMsg, {
   632→      isRetry: true,
   633→      webSearch: lastOptsRef.current.webSearch,
   634→      attachment: lastOptsRef.current.attachment,
   635→    })
   636→  }, [lastUserMsg, send])
   637→
   638→  /**
   639→   * Edita un mensaje del usuario: recorta la rama posterior (servidor y cliente)
   640→   * y reenvía el texto corregido como mensaje nuevo.
   641→   */
   642→  const editMessage = useCallback(
   643→    async (messageId: string, newContent: string) => {
   644→      const text = newContent.trim()
   645→      if (!text || pendingRef.current) return
   646→      const idx = messages.findIndex((m) => m.id === messageId)
   647→      if (idx === -1 || messages[idx]?.role !== 'user') return
   648→      const convId = activeIdRef.current
   649→      if (!convId) return
   650→
   651→      // Recorte en servidor: se conservan solo los mensajes anteriores a la edición.
   652→      try {
   653→        await fetch('/api/chat/trim', {
   654→          method: 'POST',
   655→          headers: { 'Content-Type': 'application/json' },
   656→          body: JSON.stringify({ sessionId: convId, keepCount: idx }),
   657→        })
   658→      } catch {
   659→        // si el recorte falla se continúa: la conversación local manda en la vista
   660→      }
   661→      setMessages((prev) => prev.slice(0, idx))
   662→      void send(text)
   663→    },
   664→    [messages, send],
   665→  )
   666→
   667→  /** Renombra una conversación (PATCH) y actualiza sidebar/buscador. */
   668→  const renameConversation = useCallback(
   669→    async (id: string, title: string) => {
   670→      const clean = title.trim().slice(0, 80)
   671→      if (!clean) return
   672→      setConversations((prev) => prev.map((c) => (c.id === id ? { ...c, title: clean } : c)))
   673→      useUiStore.setState((s) => ({ conversations: s.conversations.map((c) => (c.id === id ? { ...c, title: clean } : c)) }))
   674→      try {
   675→        await fetch(`/api/chat/sessions/${encodeURIComponent(id)}`, {
   676→          method: 'PATCH',
   677→          headers: { 'Content-Type': 'application/json' },
   678→          body: JSON.stringify({ title: clean }),
   679→        })
   680→      } catch {
   681→        // el título local se mantiene; el servidor quedará coherente al refrescar
   682→      }
   683→      void refreshConversations()
   684→    },
   685→    [refreshConversations],
   686→  )
   687→
   688→  /** Exporta la conversación activa como Markdown descargable. */
   689→  const exportActiveConversation = useCallback(() => {
   690→    if (messages.length === 0) return
   691→    const title = conversations.find((c) => c.id === activeId)?.title ?? 'Conversación OE AI'
   692→    exportMessagesAsMarkdown(title, messages)
   693→  }, [activeId, conversations, messages])
   694→
   695→  /** Nueva conversación (Ctrl+Shift+O). */
   696→  const newChat = useCallback(() => {
   697→    if (pendingRef.current) abortRef.current?.abort()
   698→    activeIdRef.current = null
   699→    setActiveId(null)
   700→    setMessages([])
   701→    setError(null)
   702→    setRouteNote(null)
   703→    setLastUserMsg(null)
   704→    setFocusModel(null)
   705→    setWebSearching(false)
   706→    setImagePending(false)
   707→    try {
   708→      window.localStorage.removeItem(ACTIVE_KEY)
   709→      window.sessionStorage.removeItem('oe-ai-pending-model')
   710→      useUiStore.setState({ chatModel: 'auto' })
   711→    } catch {
   712→      // sin almacenamiento
   713→    }
   714→  }, [])
   715→
   716→  /** Elimina una conversación del historial. */
   717→  const deleteConversation = useCallback(
   718→    async (id: string) => {
   719→      try {
   720→        await fetch(`/api/chat/sessions/${encodeURIComponent(id)}`, { method: 'DELETE' })
   721→      } catch {
   722→        // si falla, la lista se refresca igualmente
   723→      }
   724→      if (activeIdRef.current === id) newChat()
   725→      void refreshConversations()
   726→    },
   727→    [newChat, refreshConversations],
   728→  )
   729→
   730→  // Limpieza al desmontar
   731→  useEffect(() => {
   732→    return () => {
   733→      if (rafRef.current !== null) cancelAnimationFrame(rafRef.current)
   734→      abortRef.current?.abort()
   735→    }
   736→  }, [])
   737→
   738→  const pending = phase !== 'idle'
   739→
   740→  return {
   741→    conversations,
   742→    activeId,
   743→    messages,
   744→    phase,
   745→    pending,
   746→    error,
   747→    routeNote,
   748→    focusModel,
   749→    loadingHistory,
   750→    lastUserMsg,
   751→    webSearching,
   752→    imagePending,
   753→    guestQuota,
   754→    guestBlocked,
   755→    refreshQuota,
   756→    send,
   757→    stop,
   758→    regenerate,
   759→    newChat,
   760→    loadConversation,
   761→    deleteConversation,
   762→    refreshConversations,
   763→    setFocusModel,
   764→    generateImage,
   765→    editMessage,
   766→    renameConversation,
   767→    exportActiveConversation,
   768→  }
   769→}
   770→
   771→export type OeChatApi = ReturnType<typeof useOeChat>
