     3→/**
     4→ * useOeChat — lógica del chat premium con OE AI LLM.
     5→ *
     6→ * - Comunica con POST /api/chat (SSE: meta → stage/search → delta → done | error).
     7→ * - El backend entrega la respuesta completa (streaming de tokens no verificado);
     8→ *   la revelación progresiva es un efecto de presentación local (máquina de
     9→ *   escribir adaptativa) para una experiencia fluida, sin fingir datos.
    10→ * - Gestiona historial multi-conversación vía /api/chat/sessions y
    11→ *   /api/chat/history; herramientas: imágenes (/api/tools/image), búsqueda web
    12→ *   real con fuentes, edición de mensajes (recorte), renombrar y exportar.
    13→ */

    81→function newId(): string {
    82→  if (typeof crypto !== 'undefined' && 'randomUUID' in crypto) return crypto.randomUUID()
    83→  return `${Date.now()}-${Math.random().toString(36).slice(2)}-${Math.random().toString(36).slice(2)}`
    84→}

   114→   *  sigue viendo la cookie de invitado agotada y volvía a bloquear a usuarios
   115→   *  que YA habían iniciado sesión (la respuesta asíncrona reactivaba la
   116→   *  compuerta después de que el efecto la levantaba). */
   117→  const refreshQuota = useCallback(async () => {
   118→    try {
   119→      const token = await auth.getAccessToken().catch(() => null)
   120→      const res = await fetch('/api/chat/quota', {
   121→        cache: 'no-store',

   125→      const data = (await res.json()) as {

   183→  /** Carga los mensajes de una conversación existente (texto e imágenes). */
   184→  const loadConversation = useCallback(async (id: string) => {
   185→    if (pendingRef.current) return
   186→    setError(null)
   187→    setRouteNote(null)
   188→    setLoadingHistory(true)
   189→    try {
   190→      const res = await fetch(`/api/chat/history?sessionId=${encodeURIComponent(id)}`)
   191→      if (!res.ok) return
   192→      const data = (await res.json()) as { messages?: Array<Record<string, unknown>> }
   193→      const restored: ChatMsg[] = (data.messages ?? []).map((m) => {
   194→        const kind = m.kind === 'image' ? 'image' : 'text'
   195→        const role = m.role === 'user' ? ('user' as const) : ('assistant' as const)
   196→        let sources: WebSource[] | undefined
   197→        if (typeof m.sourcesJson === 'string' && m.sourcesJson) {
   198→          try {
   199→            const parsed = JSON.parse(m.sourcesJson) as WebSource[]

   213→          attachmentUrl: role === 'user' && typeof m.attachmentAssetId === 'string' ? `/api/assets/${m.attachmentAssetId}` : null,
   214→          sources,

   216→            providerReportedModel: (m.providerReportedModel as string) ?? null,

   226→      } catch {

   230→      setLoadingHistory(false)
   231→    }
   232→  }, [])

   235→  useEffect(() => {

   237→    let initialId: string | null = null
   238→    try {
   239→      initialId = window.localStorage.getItem(ACTIVE_KEY)
   240→    } catch {
   241→      // sin almacenamiento
   242→    }

   244→
   245→    // Enfoque de red llegado desde el Centro de APIs (/apis)
   246→    try {
   247→      const pending = window.sessionStorage.getItem('oe-ai-pending-model')

   251→      }
   252→    } catch {
   253→      // ignorar
   254→    }
   255→    const focus = useUiStore.getState().chatModel
   256→    if (focus && focus !== 'auto') setFocusModel(focus)

   259→  /** Revelación progresiva (máquina de escribir adaptativa, duración acotada).
   260→   *  Contenido estructurado (tablas/código): sin typewriter → render Markdown
   261→   *  inmediato y bien formateado (nunca se ve la sintaxis cruda). */
   262→  const startTypewriter = useCallback((assistantId: string, full: string) => {
   263→    if (/```|(^|\n)\s*\|/.test(full)) {
   264→      setMessages((prev) =>
   265→        prev.map((m) => (m.id === assistantId ? { ...m, content: full, complete: true } : m)),
   266→      )
   267→      setPhase('idle')
   268→      return
   269→    }
   270→    setPhase('typing')
   271→    const duration = Math.min(2400, Math.max(700, full.length * 4))
   272→    const start = performance.now()
   273→    const step = (now: number) => {
   274→      const p = Math.min(1, (now - start) / duration)
   275→      const eased = 1 - Math.pow(1 - p, 2.1)
   276→      const chars = Math.max(1, Math.floor(eased * full.length))
   277→      setMessages((prev) =>
   278→        prev.map((m) => (m.id === assistantId ? { ...m, content: full.slice(0, chars) } : m)),
   279→      )
   280→      if (p < 1) {
   281→        rafRef.current = requestAnimationFrame(step)
   282→      } else {
   283→        setMessages((prev) =>
   284→          prev.map((m) => (m.id === assistantId ? { ...m, content: full, complete: true } : m)),
   285→        )
   286→        setPhase('idle')
   287→      }
   288→    }
   289→    rafRef.current = requestAnimationFrame(step)
   290→  }, [])
   291→
   292→  /** Envía un mensaje al modelo unificado OE AI LLM (con búsqueda web y/o imagen adjunta opcional). */
   293→  const send = useCallback(
   294→    async (
   295→      rawText: string,
   296→      opts?: {
   297→        isRetry?: boolean
   298→        webSearch?: boolean
   299→        /** Imagen adjuntada (visión): el núcleo la analiza junto al texto. */
   300→        attachment?: { assetId: string; previewUrl: string }
   301→      },
   302→    ) => {
   303→      const content = rawText.trim()
   304→      if (!content || pendingRef.current) return
   305→      const mode = useUiStore.getState().mode
   306→      const focus = useUiStore.getState().chatModel
   307→      const webSearch = Boolean(opts?.webSearch)
   308→      const attachment = opts?.attachment
   309→      lastOptsRef.current = { webSearch, attachment }
   310→
   311→      // Conversación nueva si no hay una activa
   312→      let convId = activeIdRef.current
   313→      if (!convId) {
   314→        convId = newId()
   315→        activeIdRef.current = convId
   316→        setActiveId(convId)
   317→        try {
   318→          window.localStorage.setItem(ACTIVE_KEY, convId)
   319→        } catch {
   320→          // sin almacenamiento
   321→        }
   322→      }
   323→
   324→      setError(null)
   325→      setRouteNote(null)
   326→      setPhase('thinking')
   327→      setWebSearching(false)
   328→      pendingRef.current = true
   329→      if (!opts?.isRetry) {
   330→        setLastUserMsg(content)
   331→        setMessages((prev) => [
   332→          ...prev,
   333→          {
   334→            id: `u-${newId()}`,
   335→            role: 'user',
   336→            content,
   337→            fullContent: content,
   338→            complete: true,
   339→            attachmentUrl: attachment?.previewUrl ?? null,
   340→          },
   341→        ])
   342→      }
   343→
   344→      const assistantId = `a-${newId()}`
   345→      setMessages((prev) => [
   346→        ...prev,
   347→        { id: assistantId, role: 'assistant', content: '', fullContent: '', complete: false },
   348→      ])
   349→
   350→      const controller = new AbortController()
   351→      abortRef.current = controller
   352→      let doneMeta: ChatMsgMeta | null = null
   353→      let serverStreaming = false
   354→
   355→      const handleEvent = (event: string, data: unknown) => {
   356→        if (event === 'meta') {
   357→          const meta = data as { routeNote?: string; coreLabel?: string; mode?: 'instant' | 'reflective'; streaming?: boolean }
   358→          serverStreaming = Boolean(meta.streaming)
   359→          if (meta.routeNote) setRouteNote(meta.routeNote)
   360→          setMessages((prev) =>
   361→            prev.map((m) =>
   362→              m.id === assistantId
   363→                ? { ...m, meta: { ...m.meta, coreLabel: meta.coreLabel ?? null, routeNote: meta.routeNote ?? null, mode: meta.mode, streaming: serverStreaming } }
   364→                : m,
   365→            ),
   366→          )
   367→        } else if (event === 'stage') {
   368→          const { stage } = data as { stage?: string }
   369→          if (stage === 'web_search') setWebSearching(true)
   370→        } else if (event === 'search') {
   371→          const { sources } = data as { sources?: WebSource[] }
   372→          setWebSearching(false)
   373→          setMessages((prev) =>
   374→            prev.map((m) => (m.id === assistantId ? { ...m, sources: Array.isArray(sources) ? sources : [] } : m)),
   375→          )
   376→        } else if (event === 'search_error') {
   377→          const { message } = data as { message?: string }
   378→          setWebSearching(false)
   379→          setMessages((prev) =>
   380→            prev.map((m) => (m.id === assistantId ? { ...m, searchNote: message ?? 'La búsqueda web no estuvo disponible.' } : m)),
   381→          )
   382→        } else if (event === 'delta') {
   383→          const { text } = data as { text: string }
   384→          if (serverStreaming) {
   385→            // Streaming real del proveedor: cada delta es un incremento.
   386→            setPhase('typing')
   387→            setMessages((prev) =>
   388→              prev.map((m) => {
   389→                if (m.id !== assistantId) return m
   390→                const full = m.fullContent + text
   391→                return { ...m, fullContent: full, content: full }
   392→              }),
   393→            )
   394→          } else {
   395→            // Respuesta única: revelación progresiva local (efecto de presentación).
   396→            setMessages((prev) =>
   397→              prev.map((m) => (m.id === assistantId ? { ...m, fullContent: text } : m)),
   398→            )
   399→            startTypewriter(assistantId, text)
   400→          }
   401→        } else if (event === 'done') {
   402→          const d = data as { latencyMs?: number; providerReportedModel?: string | null; webSearchUsed?: boolean; visionUsed?: boolean }
   403→          doneMeta = {
   404→            latencyMs: d.latencyMs ?? null,
   405→            providerReportedModel: d.providerReportedModel ?? null,
   406→            webSearchUsed: Boolean(d.webSearchUsed),
   407→            visionUsed: Boolean(d.visionUsed),
   408→          }
   409→          if (serverStreaming) {
   410→            setMessages((prev) =>
   411→              prev.map((m) => (m.id === assistantId ? { ...m, content: m.fullContent, complete: true } : m)),
   412→            )
   413→            setPhase('idle')
   414→          }
   415→        } else if (event === 'error') {
   416→          const e = data as { message: string; errorType?: string }
   417→          setError({ message: e.message, errorType: e.errorType })
   418→          // Si ya había contenido parcial, se conserva completo; si no, se retira.
   419→          setMessages((prev) =>
   420→            prev
   421→              .map((m) => (m.id === assistantId && m.fullContent ? { ...m, content: m.fullContent, complete: true } : m))
   422→              .filter((m) => !(m.id === assistantId && !m.fullContent)),
   423→          )
   424→          setPhase('idle')
   425→        }
   426→      }
   427→
   428→      try {
   429→        const token = await auth.getAccessToken()
   430→        const res = await fetch('/api/chat', {
   431→          method: 'POST',
   432→          headers: {
   433→            'Content-Type': 'application/json',
   434→            ...(token ? { Authorization: `Bearer ${token}` } : {}),
   435→          },
   436→          body: JSON.stringify({
   437→            sessionId: convId,
   438→            message: content,
   439→            model: focus && focus !== 'auto' ? focus : undefined,
   440→            allowFallback: true,
   441→            mode,
   442→            webSearch,
   443→            attachmentId: attachment?.assetId,
   444→          }),
   445→          signal: controller.signal,
   446→        })
   447→
   448→        if (!res.ok || !res.body) {
   449→          const data = (await res.json().catch(() => null)) as { error?: string; errorType?: string } | null
   450→          throw Object.assign(new Error(data?.error ?? `Error ${res.status} al iniciar la conversación.`), {
   451→            errorType: data?.errorType,
   452→          })
   453→        }
   454→
   455→        const reader = res.body.getReader()
   456→        const decoder = new TextDecoder()
   457→        let buffer = ''
   458→
   459→        while (true) {
   460→          const { done, value } = await reader.read()
   461→          if (done) break
   462→          buffer += decoder.decode(value, { stream: true })
   463→          let sep: number
   464→          while ((sep = buffer.indexOf('\n\n')) !== -1) {
   465→            const rawEvent = buffer.slice(0, sep)
   466→            buffer = buffer.slice(sep + 2)
   467→            let eventName = 'message'
   468→            let dataRaw = ''
   469→            for (const line of rawEvent.split('\n')) {
   470→              if (line.startsWith('event:')) eventName = line.slice(6).trim()
   471→              else if (line.startsWith('data:')) dataRaw += line.slice(5).trim()
   472→            }
   473→            if (dataRaw) {
   474→              try {
   475→                handleEvent(eventName, JSON.parse(dataRaw))
   476→              } catch {
   477→                // evento malformado: ignorar de forma segura
   478→              }
   479→            }
   480→          }
   481→        }
   482→
   483→        // Adjuntar metadatos finales (latencia, modelo reportado)
   484→        if (doneMeta) {
   485→          const meta = doneMeta as ChatMsgMeta
   486→          setMessages((prev) => prev.map((m) => (m.id === assistantId ? { ...m, meta: { ...m.meta, ...meta } } : m)))
   487→        }
   488→        void refreshConversations()
   489→        void refreshQuota()
   490→      } catch (err) {
   491→        if ((err as Error).name === 'AbortError') {
   492→          // Cancelación: si ya hay texto, se revela completo; si no, se retira el marcador.
   493→          setMessages((prev) =>
   494→            prev
   495→              .map((m) => (m.id === assistantId && m.fullContent ? { ...m, content: m.fullContent, complete: true } : m))
   496→              .filter((m) => !(m.id === assistantId && !m.fullContent)),
   497→          )
   498→        } else {
   499→          const e = err as Error & { errorType?: string }
   500→          if (e.errorType === 'guest_limit_reached') {
   501→            // Cuota agotada: se activa la compuerta de inicio de sesión (sin error genérico).
   502→            setGuestBlocked(true)
   503→            void refreshQuota()
   504→          } else {
   505→            setError({ message: e.message, errorType: e.errorType })
   506→          }
   507→          setMessages((prev) => prev.filter((m) => m.id !== assistantId))
   508→        }
   509→      } finally {
   510→        abortRef.current = null
   511→        pendingRef.current = false
   512→        setWebSearching(false)
   513→        if (!rafRef.current) setPhase('idle')
   514→      }
   515→    },
   516→    [auth, refreshConversations, refreshQuota, startTypewriter],
   517→  )

   519→  /** Genera una imagen con la herramienta del núcleo y la inserta en la conversación. */
   520→  const generateImage = useCallback(
   521→    async (rawPrompt: string) => {
   522→      const prompt = rawPrompt.trim()
   523→      if (!prompt || pendingRef.current) return
   524→
   525→      let convId = activeIdRef.current
   526→      if (!convId) {
   527→        convId = newId()
   528→        activeIdRef.current = convId
   529→        setActiveId(convId)
   530→        try {
   531→          window.localStorage.setItem(ACTIVE_KEY, convId)
   532→        } catch {
   533→          // sin almacenamiento
   534→        }
   535→      }
   536→
   537→      setError(null)
   538→      setRouteNote(null)
   539→      setPhase('thinking')
   540→      setImagePending(true)
   541→      pendingRef.current = true
   542→      setLastUserMsg(null) // la regeneración de texto no aplica tras una imagen
   543→      setMessages((prev) => [
   544→        ...prev,
   545→        { id: `u-${newId()}`, role: 'user', content: prompt, fullContent: prompt, complete: true },
   546→      ])
   547→
   548→      const placeholderId = `img-${newId()}`
   549→      setMessages((prev) => [
   550→        ...prev,
   551→        { id: placeholderId, role: 'assistant', content: '', fullContent: '', complete: false, kind: 'image', pendingImage: true },
   552→      ])
   553→
   554→      const controller = new AbortController()
   555→      abortRef.current = controller
   556→
   557→      try {
   558→        const token = await auth.getAccessToken()
   559→        const res = await fetch('/api/tools/image', {
   560→          method: 'POST',
   561→          headers: {
   562→            'Content-Type': 'application/json',
   563→            ...(token ? { Authorization: `Bearer ${token}` } : {}),
   564→          },
   565→          body: JSON.stringify({ sessionId: convId, prompt }),
   566→          signal: controller.signal,
   567→        })
   568→        const data = (await res.json().catch(() => null)) as {
   569→          url?: string
   570→          latencyMs?: number
   571→          error?: string
   572→          errorType?: string
   573→        } | null
   574→        if (!res.ok || !data?.url) {
   575→          throw Object.assign(new Error(data?.error ?? `No se pudo generar la imagen (error ${res.status}).`), {
   576→            errorType: data?.errorType,
   577→          })
   578→        }
   579→        setMessages((prev) =>
   580→          prev.map((m) =>
   581→            m.id === placeholderId
   582→              ? {
   583→                  ...m,
   584→                  content: prompt,
   585→                  fullContent: prompt,
   586→                  assetUrl: data.url ?? null,
   587→                  pendingImage: false,
   588→                  complete: true,
   589→                  meta: { latencyMs: data.latencyMs ?? null },
   590→                }
   591→              : m,
   592→          ),
   593→        )
   594→        void refreshConversations()
   595→        void refreshQuota()
   596→      } catch (err) {
   597→        setMessages((prev) => prev.filter((m) => m.id !== placeholderId))
   598→        if ((err as Error).name !== 'AbortError') {
   599→          const e = err as Error & { errorType?: string }
   600→          if (e.errorType === 'guest_limit_reached') {
   601→            setGuestBlocked(true)
   602→            void refreshQuota()
   603→          } else {
   604→            setError({ message: e.message, errorType: e.errorType })
   605→          }
   606→        }
   607→      } finally {
   608→        abortRef.current = null
   609→        pendingRef.current = false
   610→        setImagePending(false)
   611→        setPhase('idle')
   612→      }
   613→    },
   614→    [auth, refreshConversations, refreshQuota],
   615→  )
   616→
   617→  /** Detiene la generación, la revelación o la herramienta en curso. */
   618→  const stop = useCallback(() => {
   619→    abortRef.current?.abort()
   620→    abortRef.current = null
   621→    pendingRef.current = false
   622→    if (rafRef.current !== null) {
   623→      cancelAnimationFrame(rafRef.current)
   624→      rafRef.current = null
   625→    }
   626→    setPhase('idle')
   627→    setWebSearching(false)
   628→    setImagePending(false)
   629→    setMessages((prev) => prev.map((m) => (m.complete || m.pendingImage ? m : { ...m, content: m.fullContent || m.content, complete: true })))
   630→  }, [])
   631→
   632→  /** Regenera la última respuesta (conserva la búsqueda web y la imagen adjunta del turno). */
   633→  const regenerate = useCallback(() => {
   634→    if (!lastUserMsg || pendingRef.current) return
   635→    // Se elimina la última respuesta asistente incompleta o final
   636→    setMessages((prev) => {
   637→      const idx = prev.map((m) => m.role).lastIndexOf('assistant')
   638→      return idx === -1 ? prev : prev.filter((_, i) => i !== idx)
   639→    })
   640→    void send(lastUserMsg, {
   641→      isRetry: true,
   642→      webSearch: lastOptsRef.current.webSearch,
   643→      attachment: lastOptsRef.current.attachment,
   644→    })
   645→  }, [lastUserMsg, send])
   646→
   647→  /**
   648→   * Edita un mensaje del usuario: recorta la rama posterior (servidor y cliente)
   649→   * y reenvía el texto corregido como mensaje nuevo.
   650→   */
   651→  const editMessage = useCallback(
   652→    async (messageId: string, newContent: string) => {
   653→      const text = newContent.trim()
   654→      if (!text || pendingRef.current) return
   655→      const idx = messages.findIndex((m) => m.id === messageId)
   656→      if (idx === -1 || messages[idx]?.role !== 'user') return
   657→      const convId = activeIdRef.current
   658→      if (!convId) return
   659→
   660→      // Recorte en servidor: se conservan solo los mensajes anteriores a la edición.
   661→      try {
   662→        await fetch('/api/chat/trim', {
   663→          method: 'POST',
   664→          headers: { 'Content-Type': 'application/json' },
   665→          body: JSON.stringify({ sessionId: convId, keepCount: idx }),
   666→        })
   667→      } catch {
   668→        // si el recorte falla se continúa: la conversación local manda en la vista
   669→      }
   670→      setMessages((prev) => prev.slice(0, idx))
   671→      void send(text)
   672→    },
   673→    [messages, send],
   674→  )
   675→
   676→  /** Renombra una conversación (PATCH) y actualiza sidebar/buscador. */
   677→  const renameConversation = useCallback(
   678→    async (id: string, title: string) => {
   679→      const clean = title.trim().slice(0, 80)
   680→      if (!clean) return
   681→      setConversations((prev) => prev.map((c) => (c.id === id ? { ...c, title: clean } : c)))
   682→      useUiStore.setState((s) => ({ conversations: s.conversations.map((c) => (c.id === id ? { ...c, title: clean } : c)) }))
   683→      try {
   684→        await fetch(`/api/chat/sessions/${encodeURIComponent(id)}`, {
   685→          method: 'PATCH',
   686→          headers: { 'Content-Type': 'application/json' },
   687→          body: JSON.stringify({ title: clean }),
   688→        })
   689→      } catch {
   690→        // el título local se mantiene; el servidor quedará coherente al refrescar
   691→      }
   692→      void refreshConversations()
   693→    },
   694→    [refreshConversations],
   695→  )
   696→
   697→  /** Exporta la conversación activa como Markdown descargable. */
   698→  const exportActiveConversation = useCallback(() => {
   699→    if (messages.length === 0) return
   700→    const title = conversations.find((c) => c.id === activeId)?.title ?? 'Conversación OE AI'
   701→    exportMessagesAsMarkdown(title, messages)
   702→  }, [activeId, conversations, messages])
   703→
   704→  /** Nueva conversación (Ctrl+Shift+O). */
   705→  const newChat = useCallback(() => {
   706→    if (pendingRef.current) abortRef.current?.abort()
   707→    activeIdRef.current = null
   708→    setActiveId(null)
   709→    setMessages([])
   710→    setError(null)
   711→    setRouteNote(null)
   712→    setLastUserMsg(null)
   713→    setFocusModel(null)
   714→    setWebSearching(false)
   715→    setImagePending(false)
   716→    try {
   717→      window.localStorage.removeItem(ACTIVE_KEY)
   718→      window.sessionStorage.removeItem('oe-ai-pending-model')
   719→      useUiStore.setState({ chatModel: 'auto' })
   720→    } catch {
   721→      // sin almacenamiento
   722→    }
   723→  }, [])
   724→
   725→  /** Elimina una conversación del historial. */
   726→  const deleteConversation = useCallback(
   727→    async (id: string) => {
   728→      try {
   729→        await fetch(`/api/chat/sessions/${encodeURIComponent(id)}`, { method: 'DELETE' })
   730→      } catch {
   731→        // si falla, la lista se refresca igualmente
   732→      }
   733→      if (activeIdRef.current === id) newChat()
   734→      void refreshConversations()
   735→    },
   736→    [newChat, refreshConversations],
   737→  )
   738→
   739→  // Limpieza al desmontar
   740→  useEffect(() => {
   741→    return () => {
   742→      if (rafRef.current !== null) cancelAnimationFrame(rafRef.current)
   743→      abortRef.current?.abort()
   744→    }
   745→  }, [])
   746→
   747→  const pending = phase !== 'idle'
   748→
   749→  return {
   750→    conversations,
   751→    activeId,
   752→    messages,
   753→    phase,
   754→    pending,
   755→    error,
   756→    routeNote,
   757→    focusModel,
   758→    loadingHistory,
   759→    lastUserMsg,
   760→    webSearching,
   761→    imagePending,
   762→    guestQuota,
   763→    guestBlocked,
   764→    refreshQuota,
   765→    send,
   766→    stop,
   767→    regenerate,
   768→    newChat,
   769→    loadConversation,
   770→    deleteConversation,
   771→    refreshConversations,
   772→    setFocusModel,
   773→    generateImage,
   774→    editMessage,
   775→    renameConversation,
   776→    exportActiveConversation,
   777→  }
   778→}
   779→
   780→export type OeChatApi = ReturnType<typeof useOeChat>
