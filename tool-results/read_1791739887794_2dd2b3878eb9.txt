     5→     5→ *
     6→     6→ * - Comunica con POST /api/chat (SSE: meta → stage/search → delta → done | error).

     9→     9→ *   escribir adaptativa) para una experiencia fluida, sin fingir datos.
    10→    10→ * - Gestiona historial multi-conversación vía /api/chat/sessions y
    11→    11→ *   /api/chat/history; herramientas: imágenes (/api/tools/image), búsqueda web
    12→    12→ *   real con fuentes, edición de mensajes (recorte), renombrar y exportar.

   103→   103→

   115→   115→  /** Estado de la cuota (GET /api/chat/quota): nula si hay sesión iniciada.
   116→   116→   *  IMPORTANTE: envía el token de Supabase cuando existe — sin él el endpoint
   117→   117→   *  sigue viendo la cookie de invitado agotada y volvía a bloquear a usuarios
   118→   118→   *  que YA habían iniciado sesión (la respuesta asíncrona reactivaba la

   120→   120→  const refreshQuota = useCallback(async () => {
   121→   121→    try {
   122→   122→      const token = await auth.getAccessToken().catch(() => null)
   123→   123→      const res = await fetch('/api/chat/quota', {
   124→   124→        cache: 'no-store',
   125→   125→        ...(token ? { headers: { Authorization: `Bearer ${token}` } } : {}),
   126→   126→      })
   127→   127→      if (!res.ok) return
   128→   128→      const data = (await res.json()) as {
   129→   129→        authenticated: boolean
   130→   130→        used: number | null
   131→   131→        limit: number | null
   132→   132→        remaining: number | null
   133→   133→        exhausted?: boolean
   134→   134→      }

   136→   136→        setGuestQuota(null)
   137→   137→        setGuestBlocked(false)
   138→   138→        return
   139→   139→      }

   141→   141→        used: data.used ?? 0,
   142→   142→        limit: data.limit ?? 5,
   143→   143→        remaining: data.remaining ?? 0,
   144→   144→        exhausted: Boolean(data.exhausted),

   149→   149→    } catch {

   151→   151→    }

   153→   153→

   160→   160→

   183→   183→    }
   184→   184→  }, [])
   185→   185→
   186→   186→  /** Carga los mensajes de una conversación existente (texto e imágenes). */
   187→   187→  const loadConversation = useCallback(async (id: string) => {
   188→   188→    if (pendingRef.current) return
   189→   189→    setError(null)
   190→   190→    setRouteNote(null)
   191→   191→    setLoadingHistory(true)
   192→   192→    try {
   193→   193→      const res = await fetch(`/api/chat/history?sessionId=${encodeURIComponent(id)}`)
   194→   194→      if (!res.ok) return
   195→   195→      const data = (await res.json()) as { messages?: Array<Record<string, unknown>> }
   196→   196→      const restored: ChatMsg[] = (data.messages ?? []).map((m) => {
   197→   197→        const kind = m.kind === 'image' ? 'image' : 'text'

   199→   199→        let sources: WebSource[] | undefined

   235→   235→  }, [])
   236→   236→

   254→   254→      }
   255→   255→    } catch {
   256→   256→      // ignorar
   257→   257→    }
   258→   258→    const focus = useUiStore.getState().chatModel
   259→   259→    if (focus && focus !== 'auto') setFocusModel(focus)
   260→   260→  }, [loadConversation, refreshConversations])
   261→   261→
   262→   262→  /** Revelación progresiva (máquina de escribir adaptativa, duración acotada).
   263→   263→   *  Contenido estructurado (tablas/código): sin typewriter → render Markdown
   264→   264→   *  inmediato y bien formateado (nunca se ve la sintaxis cruda). */
   265→   265→  const startTypewriter = useCallback((assistantId: string, full: string) => {
   266→   266→    if (/```|(^|\n)\s*\|/.test(full)) {
   267→   267→      setMessages((prev) =>
   268→   268→        prev.map((m) => (m.id === assistantId ? { ...m, content: full, complete: true } : m)),
   269→   269→      )
   270→   270→      setPhase('idle')
   271→   271→      return
   272→   272→    }
   273→   273→    setPhase('typing')
   274→   274→    const duration = Math.min(2400, Math.max(700, full.length * 4))
   275→   275→    const start = performance.now()
   276→   276→    const step = (now: number) => {
   277→   277→      const p = Math.min(1, (now - start) / duration)
   278→   278→      const eased = 1 - Math.pow(1 - p, 2.1)
   279→   279→      const chars = Math.max(1, Math.floor(eased * full.length))
   280→   280→      setMessages((prev) =>
   281→   281→        prev.map((m) => (m.id === assistantId ? { ...m, content: full.slice(0, chars) } : m)),
   282→   282→      )
   283→   283→      if (p < 1) {
   284→   284→        rafRef.current = requestAnimationFrame(step)
   285→   285→      } else {
   286→   286→        setMessages((prev) =>
   287→   287→          prev.map((m) => (m.id === assistantId ? { ...m, content: full, complete: true } : m)),
   288→   288→        )
   289→   289→        setPhase('idle')
   290→   290→      }
   291→   291→    }
   292→   292→    rafRef.current = requestAnimationFrame(step)
   293→   293→  }, [])
   294→   294→
   295→   295→  /** Envía un mensaje al modelo unificado OE AI LLM (con búsqueda web y/o imagen adjunta opcional). */
   296→   296→  const send = useCallback(
   297→   297→    async (
   298→   298→      rawText: string,
   299→   299→      opts?: {
   300→   300→        isRetry?: boolean
   301→   301→        webSearch?: boolean
   302→   302→        /** Imagen adjuntada (visión): el núcleo la analiza junto al texto. */
   303→   303→        attachment?: { assetId: string; previewUrl: string }
   304→   304→        /** Texto de documento leído (enlace/PDF): contexto efímero del turno. */
   305→   305→        docContext?: string
   306→   306→        /** Título del documento leído (se muestra en el prompt de sistema). */
   307→   307→        docTitle?: string
   308→   308→      },
   309→   309→    ) => {
   310→   310→      const content = rawText.trim()
   311→   311→      if (!content || pendingRef.current) return
   312→   312→      const mode = useUiStore.getState().mode
   313→   313→      const focus = useUiStore.getState().chatModel
   314→   314→      const webSearch = Boolean(opts?.webSearch)
   315→   315→      const attachment = opts?.attachment
   316→   316→      const docContext = typeof opts?.docContext === 'string' ? opts.docContext : undefined
   317→   317→      const docTitle = typeof opts?.docTitle === 'string' ? opts.docTitle : undefined
   318→   318→      lastOptsRef.current = { webSearch, attachment }
   319→   319→
   320→   320→      // Conversación nueva si no hay una activa
   321→   321→      let convId = activeIdRef.current
   322→   322→      if (!convId) {
   323→   323→        convId = newId()
   324→   324→        activeIdRef.current = convId
   325→   325→        setActiveId(convId)
   326→   326→        try {
   327→   327→          window.localStorage.setItem(ACTIVE_KEY, convId)
   328→   328→        } catch {
   329→   329→          // sin almacenamiento
   330→   330→        }
   331→   331→      }
   332→   332→
   333→   333→      setError(null)
   334→   334→      setRouteNote(null)
   335→   335→      setPhase('thinking')
   336→   336→      setWebSearching(false)
   337→   337→      pendingRef.current = true
   338→   338→      if (!opts?.isRetry) {
   339→   339→        setLastUserMsg(content)
   340→   340→        setMessages((prev) => [
   341→   341→          ...prev,
   342→   342→          {
   343→   343→            id: `u-${newId()}`,
   344→   344→            role: 'user',
   345→   345→            content,
   346→   346→            fullContent: content,
   347→   347→            complete: true,
   348→   348→            attachmentUrl: attachment?.previewUrl ?? null,
   349→   349→          },
   350→   350→        ])
   351→   351→      }
   352→   352→
   353→   353→      const assistantId = `a-${newId()}`
   354→   354→      setMessages((prev) => [
   355→   355→        ...prev,
   356→   356→        { id: assistantId, role: 'assistant', content: '', fullContent: '', complete: false },
   357→   357→      ])
   358→   358→
   359→   359→      const controller = new AbortController()
   360→   360→      abortRef.current = controller
   361→   361→      let doneMeta: ChatMsgMeta | null = null
   362→   362→      let serverStreaming = false
   363→   363→
   364→   364→      const handleEvent = (event: string, data: unknown) => {
   365→   365→        if (event === 'meta') {
   366→   366→          const meta = data as { routeNote?: string; coreLabel?: string; mode?: 'instant' | 'reflective'; streaming?: boolean }
   367→   367→          serverStreaming = Boolean(meta.streaming)
   368→   368→          if (meta.routeNote) setRouteNote(meta.routeNote)
   369→   369→          setMessages((prev) =>
   370→   370→            prev.map((m) =>
   371→   371→              m.id === assistantId
   372→   372→                ? { ...m, meta: { ...m.meta, coreLabel: meta.coreLabel ?? null, routeNote: meta.routeNote ?? null, mode: meta.mode, streaming: serverStreaming } }
   373→   373→                : m,
   374→   374→            ),
   375→   375→          )
   376→   376→        } else if (event === 'stage') {
   377→   377→          const { stage } = data as { stage?: string }
   378→   378→          if (stage === 'web_search') setWebSearching(true)
   379→   379→        } else if (event === 'search') {
   380→   380→          const { sources } = data as { sources?: WebSource[] }
   381→   381→          setWebSearching(false)
   382→   382→          setMessages((prev) =>
   383→   383→            prev.map((m) => (m.id === assistantId ? { ...m, sources: Array.isArray(sources) ? sources : [] } : m)),
   384→   384→          )
   385→   385→        } else if (event === 'search_error') {
   386→   386→          const { message } = data as { message?: string }
   387→   387→          setWebSearching(false)
   388→   388→          setMessages((prev) =>
   389→   389→            prev.map((m) => (m.id === assistantId ? { ...m, searchNote: message ?? 'La búsqueda web no estuvo disponible.' } : m)),
   390→   390→          )
   391→   391→        } else if (event === 'delta') {
   392→   392→          const { text } = data as { text: string }
   393→   393→          if (serverStreaming) {
   394→   394→            // Streaming real del proveedor: cada delta es un incremento.
   395→   395→            setPhase('typing')
   396→   396→            setMessages((prev) =>
   397→   397→              prev.map((m) => {
   398→   398→                if (m.id !== assistantId) return m
   399→   399→                const full = m.fullContent + text
   400→   400→                return { ...m, fullContent: full, content: full }
   401→   401→              }),
   402→   402→            )
   403→   403→          } else {
   404→   404→            // Respuesta única: revelación progresiva local (efecto de presentación).
   405→   405→            setMessages((prev) =>
   406→   406→              prev.map((m) => (m.id === assistantId ? { ...m, fullContent: text } : m)),
   407→   407→            )
   408→   408→            startTypewriter(assistantId, text)
   409→   409→          }
   410→   410→        } else if (event === 'done') {
   411→   411→          const d = data as { latencyMs?: number; providerReportedModel?: string | null; webSearchUsed?: boolean; visionUsed?: boolean }
   412→   412→          doneMeta = {
   413→   413→            latencyMs: d.latencyMs ?? null,
   414→   414→            providerReportedModel: d.providerReportedModel ?? null,
   415→   415→            webSearchUsed: Boolean(d.webSearchUsed),
   416→   416→            visionUsed: Boolean(d.visionUsed),
   417→   417→          }
   418→   418→          if (serverStreaming) {
   419→   419→            setMessages((prev) =>
   420→   420→              prev.map((m) => (m.id === assistantId ? { ...m, content: m.fullContent, complete: true } : m)),
   421→   421→            )
   422→   422→            setPhase('idle')
   423→   423→          }
   424→   424→        } else if (event === 'error') {
   425→   425→          const e = data as { message: string; errorType?: string }
   426→   426→          setError({ message: e.message, errorType: e.errorType })
   427→   427→          // Si ya había contenido parcial, se conserva completo; si no, se retira.
   428→   428→          setMessages((prev) =>
   429→   429→            prev
   430→   430→              .map((m) => (m.id === assistantId && m.fullContent ? { ...m, content: m.fullContent, complete: true } : m))
   431→   431→              .filter((m) => !(m.id === assistantId && !m.fullContent)),
   432→   432→          )
   433→   433→          setPhase('idle')
   434→   434→        }
   435→   435→      }
   436→   436→
   437→   437→      try {

   491→   491→          }

   519→   519→        }

   521→   521→        abortRef.current = null
   522→   522→        pendingRef.current = false
   523→   523→        setWebSearching(false)

   905→   905→      setMessages((prev) => prev.slice(0, idx))

   909→   909→  )

   924→   924→      } catch {

   957→   957→    }

   971→   971→    [newChat, refreshConversations],
   972→   972→  )

   975→   975→  useEffect(() => {

   978→   978→      abortRef.current?.abort()
   979→   979→    }
   980→   980→  }, [])
