   198→   198→/**
   199→   199→ * Extrae la directiva ```oe-build del texto del asistente (constructor de apps).
   200→   200→ * Devuelve el brief refinado y el texto de anuncio previo a la valla.
   201→   201→ */
   202→   202→interface BuildDirectivePayload {
   203→   203→  name?: string
   204→   204→  brief: string
   205→   205→  mode?: string
   206→   206→  projectId?: string
   207→   207→  kind: 'app' | 'web'
   208→   208→  prefix: string
   209→   209→}
   210→   210→
   211→   211→function extractBuildDirective(text: string): BuildDirectivePayload | null {
   212→   212→  const webIdx = text.indexOf('```oe-web')
   213→   213→  const appIdx = text.indexOf('```oe-build')
   214→   214→  const kind: 'app' | 'web' = webIdx !== -1 && (appIdx === -1 || webIdx < appIdx) ? 'web' : 'app'
   215→   215→  const fenceStart = kind === 'web' ? webIdx : appIdx
   216→   216→  if (fenceStart === -1) return null
   217→   217→  const bodyStart = text.indexOf('\n', fenceStart)
   218→   218→  if (bodyStart === -1) return null
   219→   219→  const fenceEnd = text.indexOf('```', bodyStart)
   220→   220→  if (fenceEnd === -1) return null
   221→   221→  const raw = text.slice(bodyStart, fenceEnd).trim()
   222→   222→  const start = raw.indexOf('{')
   223→   223→  if (start === -1) return null
   224→   224→  let depth = 0
   225→   225→  let inString = false
   226→   226→  let escaped = false
   227→   227→  let end = -1
   228→   228→  for (let i = start; i < raw.length; i++) {
   229→   229→    const ch = raw[i]
   230→   230→    if (inString) {
   231→   231→      if (escaped) escaped = false
   232→   232→      else if (ch === '\\') escaped = true
   233→   233→      else if (ch === '"') inString = false
   234→   234→      continue
   235→   235→    }
   236→   236→    if (ch === '"') inString = true
   237→   237→    else if (ch === '{') depth++
   238→   238→    else if (ch === '}') {
   239→   239→      depth--
   240→   240→      if (depth === 0) {
   241→   241→        end = i
   242→   242→        break
   243→   243→      }
   244→   244→    }
   245→   245→  }
   246→   246→  if (end === -1) return null
   247→   247→  try {
   248→   248→    const parsed = JSON.parse(raw.slice(start, end + 1)) as {
   249→   249→      name?: unknown
   250→   250→      brief?: unknown
   251→   251→      mode?: unknown
   252→   252→      projectId?: unknown
   253→   253→    }
   254→   254→    const brief = typeof parsed.brief === 'string' ? parsed.brief.trim() : ''
   255→   255→    if (!brief) return null
   256→   256→    const prefix = text
   257→   257→      .slice(0, fenceStart)
   258→   258→      .replace(/```[a-z]*\s*$/, '')
   259→   259→      .trim()
   260→   260→    return {
   261→   261→      brief: brief.slice(0, 6000),
   262→   262→      name: typeof parsed.name === 'string' ? parsed.name : undefined,
   263→   263→      mode: typeof parsed.mode === 'string' ? parsed.mode : undefined,
   264→   264→      projectId: typeof parsed.projectId === 'string' ? parsed.projectId : undefined,
   265→   265→      kind,
   266→   266→      prefix,
   267→   267→    }
   268→   268→  } catch {
   269→   269→    return null
   270→   270→  }
   271→   271→}

   274→   274→function extractLastBuildFromHistory(
   275→   275→  messages: Array<{ content: string }>,
   276→   276→): { projectId: string; title: string; files: string[]; kind: 'app' | 'web' } | null {
   277→   277→  const re = /```oeproject\s*\n([\s\S]*?)```/g
   278→   278→  let last: { projectId: string; title: string; files: string[]; kind: 'app' | 'web' } | null = null
   279→   279→  for (const m of messages) {
   280→   280→    let match: RegExpExecArray | null
   281→   281→    re.lastIndex = 0
   282→   282→    while ((match = re.exec(m.content)) !== null) {
   283→   283→      try {
   284→   284→        const data = JSON.parse(match[1]) as { id?: unknown; title?: unknown; kind?: unknown; files?: Array<{ path?: unknown }> }
   285→   285→        if (typeof data.id === 'string' && Array.isArray(data.files)) {
   286→   286→          last = {
   287→   287→            projectId: data.id,
   288→   288→            title: typeof data.title === 'string' ? data.title : 'App construida',
   289→   289→            files: data.files
   290→   290→              .map((f) => (typeof f?.path === 'string' ? f.path : ''))
   291→   291→              .filter(Boolean)
   292→   292→              .slice(0, 12),
   293→   293→            kind: data.kind === 'web' ? 'web' : 'app',
   294→   294→          }
   295→   295→        }
   296→   296→      } catch {
   297→   297→        // bloque inválido: ignorar
   298→   298→      }
   299→   299→    }
   300→   300→  }
   301→   301→  return last
