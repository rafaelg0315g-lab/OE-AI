   249→   249→
   250→   250→  /** Revelación progresiva (máquina de escribir adaptativa, duración acotada).
   251→   251→   *  Contenido estructurado (tablas/código): sin typewriter → render Markdown
   252→   252→   *  inmediato y bien formateado (nunca se ve la sintaxis cruda). */
   253→   253→  const startTypewriter = useCallback((assistantId: string, full: string) => {
   254→   254→    if (/```|(^|\n)\s*\|/.test(full)) {
   255→   255→      setMessages((prev) =>
   256→   256→        prev.map((m) => (m.id === assistantId ? { ...m, content: full, complete: true } : m)),
   257→   257→      )
   258→   258→      setPhase('idle')
   259→   259→      return
   260→   260→    }
   261→   261→    setPhase('typing')
   262→   262→    const duration = Math.min(2400, Math.max(700, full.length * 4))
   263→   263→    const start = performance.now()
   264→   264→    const step = (now: number) => {
   265→   265→      const p = Math.min(1, (now - start) / duration)
   266→   266→      const eased = 1 - Math.pow(1 - p, 2.1)
   267→   267→      const chars = Math.max(1, Math.floor(eased * full.length))
   268→   268→      setMessages((prev) =>
   269→   269→        prev.map((m) => (m.id === assistantId ? { ...m, content: full.slice(0, chars) } : m)),
   270→   270→      )
   271→   271→      if (p < 1) {
   272→   272→        rafRef.current = requestAnimationFrame(step)
   273→   273→      } else {
   274→   274→        setMessages((prev) =>
   275→   275→          prev.map((m) => (m.id === assistantId ? { ...m, content: full, complete: true } : m)),
   276→   276→        )
   277→   277→        setPhase('idle')
   278→   278→      }
   279→   279→    }
   280→   280→    rafRef.current = requestAnimationFrame(step)
   281→   281→  }, [])
   282→   282→
   283→   283→  /** Envía un mensaje al modelo unificado OE AI LLM (con búsqueda web y/o imagen adjunta opcional). */
   284→   284→  const send = useCallback(
   285→   285→    async (
   286→   286→      rawText: string,
   287→   287→      opts?: {
   288→   288→        isRetry?: boolean
   289→   289→        webSearch?: boolean
   290→   290→        /** Imagen adjuntada (visión): el núcleo la analiza junto al texto. */
   291→   291→        attachment?: { assetId: string; previewUrl: string }
   292→   292→      },
   293→   293→    ) => {
   294→   294→      const content = rawText.trim()
   295→   295→      if (!content || pendingRef.current) return
   296→   296→      const mode = useUiStore.getState().mode
   297→   297→      const focus = useUiStore.getState().chatModel
   298→   298→      const webSearch = Boolean(opts?.webSearch)
   299→   299→      const attachment = opts?.attachment
   300→   300→      lastOptsRef.current = { webSearch, attachment }
   301→   301→
   302→   302→      // Conversación nueva si no hay una activa
   303→   303→      let convId = activeIdRef.current
   304→   304→      if (!convId) {
   305→   305→        convId = newId()
   306→   306→        activeIdRef.current = convId
   307→   307→        setActiveId(convId)
   308→   308→        try {
   309→   309→          window.localStorage.setItem(ACTIVE_KEY, convId)
   310→   310→        } catch {
   311→   311→          // sin almacenamiento
   312→   312→        }
   313→   313→      }
   314→   314→
   315→   315→      setError(null)
   316→   316→      setRouteNote(null)
   317→   317→      setPhase('thinking')
   318→   318→      setWebSearching(false)
   319→   319→      pendingRef.current = true
   320→   320→      if (!opts?.isRetry) {
   321→   321→        setLastUserMsg(content)
   322→   322→        setMessages((prev) => [
   323→   323→          ...prev,
   324→   324→          {
   325→   325→            id: `u-${newId()}`,
   326→   326→            role: 'user',
   327→   327→            content,
   328→   328→            fullContent: content,
   329→   329→            complete: true,
   330→   330→            attachmentUrl: attachment?.previewUrl ?? null,
   331→   331→          },
   332→   332→        ])
   333→   333→      }
   334→   334→
   335→   335→      const assistantId = `a-${newId()}`
   336→   336→      setMessages((prev) => [
   337→   337→        ...prev,
   338→   338→        { id: assistantId, role: 'assistant', content: '', fullContent: '', complete: false },
   339→   339→      ])
   340→   340→
   341→   341→      const controller = new AbortController()
   342→   342→      abortRef.current = controller
   343→   343→      let doneMeta: ChatMsgMeta | null = null
   344→   344→      let serverStreaming = false
   345→   345→
   346→   346→      const handleEvent = (event: string, data: unknown) => {
   347→   347→        if (event === 'meta') {
   348→   348→          const meta = data as { routeNote?: string; coreLabel?: string; mode?: 'instant' | 'reflective'; streaming?: boolean }
   349→   349→          serverStreaming = Boolean(meta.streaming)
   350→   350→          if (meta.routeNote) setRouteNote(meta.routeNote)
   351→   351→          setMessages((prev) =>
   352→   352→            prev.map((m) =>
   353→   353→              m.id === assistantId
   354→   354→                ? { ...m, meta: { ...m.meta, coreLabel: meta.coreLabel ?? null, routeNote: meta.routeNote ?? null, mode: meta.mode, streaming: serverStreaming } }
   355→   355→                : m,
   356→   356→            ),
   357→   357→          )
   358→   358→        } else if (event === 'stage') {
   359→   359→          const { stage } = data as { stage?: string }
   360→   360→          if (stage === 'web_search') setWebSearching(true)
   361→   361→        } else if (event === 'search') {
   362→   362→          const { sources } = data as { sources?: WebSource[] }
   363→   363→          setWebSearching(false)
   364→   364→          setMessages((prev) =>
   365→   365→            prev.map((m) => (m.id === assistantId ? { ...m, sources: Array.isArray(sources) ? sources : [] } : m)),
   366→   366→          )
   367→   367→        } else if (event === 'search_error') {
   368→   368→          const { message } = data as { message?: string }
   369→   369→          setWebSearching(false)
   370→   370→          setMessages((prev) =>
   371→   371→            prev.map((m) => (m.id === assistantId ? { ...m, searchNote: message ?? 'La búsqueda web no estuvo disponible.' } : m)),
   372→   372→          )
   373→   373→        } else if (event === 'delta') {
   374→   374→          const { text } = data as { text: string }
   375→   375→          if (serverStreaming) {
   376→   376→            // Streaming real del proveedor: cada delta es un incremento.
   377→   377→            setPhase('typing')
   378→   378→            setMessages((prev) =>
   379→   379→              prev.map((m) => {
   380→   380→                if (m.id !== assistantId) return m
   381→   381→                const full = m.fullContent + text
   382→   382→                return { ...m, fullContent: full, content: full }
   383→   383→              }),
   384→   384→            )
   385→   385→          } else {
   386→   386→            // Respuesta única: revelación progresiva local (efecto de presentación).
   387→   387→            setMessages((prev) =>
   388→   388→              prev.map((m) => (m.id === assistantId ? { ...m, fullContent: text } : m)),
   389→   389→            )
   390→   390→            startTypewriter(assistantId, text)
   391→   391→          }
   392→   392→        } else if (event === 'done') {
   393→   393→          const d = data as { latencyMs?: number; providerReportedModel?: string | null; webSearchUsed?: boolean; visionUsed?: boolean }
   394→   394→          doneMeta = {
   395→   395→            latencyMs: d.latencyMs ?? null,
   396→   396→            providerReportedModel: d.providerReportedModel ?? null,
   397→   397→            webSearchUsed: Boolean(d.webSearchUsed),
   398→   398→            visionUsed: Boolean(d.visionUsed),
   399→   399→          }
   400→   400→          if (serverStreaming) {
   401→   401→            setMessages((prev) =>
   402→   402→              prev.map((m) => (m.id === assistantId ? { ...m, content: m.fullContent, complete: true } : m)),
   403→   403→            )
   404→   404→            setPhase('idle')
   405→   405→          }
   406→   406→        } else if (event === 'error') {
   407→   407→          const e = data as { message: string; errorType?: string }
   408→   408→          setError({ message: e.message, errorType: e.errorType })
   409→   409→          // Si ya había contenido parcial, se conserva completo; si no, se retira.
   410→   410→          setMessages((prev) =>
   411→   411→            prev
   412→   412→              .map((m) => (m.id === assistantId && m.fullContent ? { ...m, content: m.fullContent, complete: true } : m))
   413→   413→              .filter((m) => !(m.id === assistantId && !m.fullContent)),
   414→   414→          )
   415→   415→          setPhase('idle')
   416→   416→        }
   417→   417→      }
   418→   418→
   419→   419→      try {
   420→   420→        const token = await auth.getAccessToken()
   421→   421→        const res = await fetch('/api/chat', {
   422→   422→          method: 'POST',
   423→   423→          headers: {
   424→   424→            'Content-Type': 'application/json',
   425→   425→            ...(token ? { Authorization: `Bearer ${token}` } : {}),
   426→   426→          },
   427→   427→          body: JSON.stringify({
   428→   428→            sessionId: convId,
   429→   429→            message: content,
   430→   430→            model: focus && focus !== 'auto' ? focus : undefined,
   431→   431→            allowFallback: true,
   432→   432→            mode,
   433→   433→            webSearch,
   434→   434→            attachmentId: attachment?.assetId,
   435→   435→          }),
   436→   436→          signal: controller.signal,
   437→   437→        })
   438→   438→
   439→   439→        if (!res.ok || !res.body) {
   440→   440→          const data = (await res.json().catch(() => null)) as { error?: string; errorType?: string } | null
   441→   441→          throw Object.assign(new Error(data?.error ?? `Error ${res.status} al iniciar la conversación.`), {
   442→   442→            errorType: data?.errorType,
   443→   443→          })
   444→   444→        }
   445→   445→
   446→   446→        const reader = res.body.getReader()
   447→   447→        const decoder = new TextDecoder()
   448→   448→        let buffer = ''
   449→   449→
   450→   450→        while (true) {
   451→   451→          const { done, value } = await reader.read()
   452→   452→          if (done) break
   453→   453→          buffer += decoder.decode(value, { stream: true })
   454→   454→          let sep: number
   455→   455→          while ((sep = buffer.indexOf('\n\n')) !== -1) {
   456→   456→            const rawEvent = buffer.slice(0, sep)
   457→   457→            buffer = buffer.slice(sep + 2)
   458→   458→            let eventName = 'message'
   459→   459→            let dataRaw = ''
   460→   460→            for (const line of rawEvent.split('\n')) {
   461→   461→              if (line.startsWith('event:')) eventName = line.slice(6).trim()
   462→   462→              else if (line.startsWith('data:')) dataRaw += line.slice(5).trim()
   463→   463→            }
   464→   464→            if (dataRaw) {
   465→   465→              try {
   466→   466→                handleEvent(eventName, JSON.parse(dataRaw))
   467→   467→              } catch {
   468→   468→                // evento malformado: ignorar de forma segura
   469→   469→              }
   470→   470→            }
   471→   471→          }
   472→   472→        }
   473→   473→
   474→   474→        // Adjuntar metadatos finales (latencia, modelo reportado)
   475→   475→        if (doneMeta) {
   476→   476→          const meta = doneMeta as ChatMsgMeta
   477→   477→          setMessages((prev) => prev.map((m) => (m.id === assistantId ? { ...m, meta: { ...m.meta, ...meta } } : m)))
   478→   478→        }
   479→   479→        void refreshConversations()
   480→   480→        void refreshQuota()
   481→   481→      } catch (err) {
   482→   482→        if ((err as Error).name === 'AbortError') {
   483→   483→          // Cancelación: si ya hay texto, se revela completo; si no, se retira el marcador.
   484→   484→          setMessages((prev) =>
   485→   485→            prev
   486→   486→              .map((m) => (m.id === assistantId && m.fullContent ? { ...m, content: m.fullContent, complete: true } : m))
   487→   487→              .filter((m) => !(m.id === assistantId && !m.fullContent)),
   488→   488→          )
   489→   489→        } else {
   490→   490→          const e = err as Error & { errorType?: string }
   491→   491→          if (e.errorType === 'guest_limit_reached') {
   492→   492→            // Cuota agotada: se activa la compuerta de inicio de sesión (sin error genérico).
   493→   493→            setGuestBlocked(true)
   494→   494→            void refreshQuota()
   495→   495→          } else {
   496→   496→            setError({ message: e.message, errorType: e.errorType })
   497→   497→          }
   498→   498→          setMessages((prev) => prev.filter((m) => m.id !== assistantId))
   499→   499→        }
   500→   500→      } finally {
   501→   501→        abortRef.current = null
   502→   502→        pendingRef.current = false
   503→   503→        setWebSearching(false)
   504→   504→        if (!rafRef.current) setPhase('idle')
   505→   505→      }
   506→   506→    },
   507→   507→    [auth, refreshConversations, refreshQuota, startTypewriter],
   508→   508→  )
   509→   509→
   510→   510→  /** Genera una imagen con la herramienta del núcleo y la inserta en la conversación. */
   511→   511→  const generateImage = useCallback(
   512→   512→    async (rawPrompt: string) => {
   513→   513→      const prompt = rawPrompt.trim()
   514→   514→      if (!prompt || pendingRef.current) return
   515→   515→
   516→   516→      let convId = activeIdRef.current
   517→   517→      if (!convId) {
   518→   518→        convId = newId()
   519→   519→        activeIdRef.current = convId
   520→   520→        setActiveId(convId)
   521→   521→        try {
   522→   522→          window.localStorage.setItem(ACTIVE_KEY, convId)
   523→   523→        } catch {
   524→   524→          // sin almacenamiento
   525→   525→        }
   526→   526→      }
   527→   527→
   528→   528→      setError(null)
   529→   529→      setRouteNote(null)
   530→   530→      setPhase('thinking')
   531→   531→      setImagePending(true)
   532→   532→      pendingRef.current = true
   533→   533→      setLastUserMsg(null) // la regeneración de texto no aplica tras una imagen
   534→   534→      setMessages((prev) => [
   535→   535→        ...prev,
   536→   536→        { id: `u-${newId()}`, role: 'user', content: prompt, fullContent: prompt, complete: true },
   537→   537→      ])
   538→   538→
   539→   539→      const placeholderId = `img-${newId()}`
   540→   540→      setMessages((prev) => [
   541→   541→        ...prev,
   542→   542→        { id: placeholderId, role: 'assistant', content: '', fullContent: '', complete: false, kind: 'image', pendingImage: true },
   543→   543→      ])
   544→   544→
   545→   545→      const controller = new AbortController()
   546→   546→      abortRef.current = controller
   547→   547→
   548→   548→      try {
   549→   549→        const token = await auth.getAccessToken()
   550→   550→        const res = await fetch('/api/tools/image', {
   551→   551→          method: 'POST',
   552→   552→          headers: {
   553→   553→            'Content-Type': 'application/json',
   554→   554→            ...(token ? { Authorization: `Bearer ${token}` } : {}),
   555→   555→          },
   556→   556→          body: JSON.stringify({ sessionId: convId, prompt }),
   557→   557→          signal: controller.signal,
   558→   558→        })
   559→   559→        const data = (await res.json().catch(() => null)) as {
   560→   560→          url?: string
   561→   561→          latencyMs?: number
   562→   562→          error?: string
   563→   563→          errorType?: string
   564→   564→        } | null
   565→   565→        if (!res.ok || !data?.url) {
   566→   566→          throw Object.assign(new Error(data?.error ?? `No se pudo generar la imagen (error ${res.status}).`), {
   567→   567→            errorType: data?.errorType,
   568→   568→          })
   569→   569→        }
   570→   570→        setMessages((prev) =>
   571→   571→          prev.map((m) =>
   572→   572→            m.id === placeholderId
   573→   573→              ? {
   574→   574→                  ...m,
   575→   575→                  content: prompt,
   576→   576→                  fullContent: prompt,
   577→   577→                  assetUrl: data.url ?? null,
   578→   578→                  pendingImage: false,
   579→   579→                  complete: true,
   580→   580→                  meta: { latencyMs: data.latencyMs ?? null },
   581→   581→                }
   582→   582→              : m,
   583→   583→          ),
   584→   584→        )
   585→   585→        void refreshConversations()
   586→   586→        void refreshQuota()
   587→   587→      } catch (err) {
   588→   588→        setMessages((prev) => prev.filter((m) => m.id !== placeholderId))
   589→   589→        if ((err as Error).name !== 'AbortError') {
   590→   590→          const e = err as Error & { errorType?: string }
   591→   591→          if (e.errorType === 'guest_limit_reached') {
   592→   592→            setGuestBlocked(true)
   593→   593→            void refreshQuota()
   594→   594→          } else {
   595→   595→            setError({ message: e.message, errorType: e.errorType })
   596→   596→          }
   597→   597→        }
   598→   598→      } finally {
   599→   599→        abortRef.current = null
   600→   600→        pendingRef.current = false
   601→   601→        setImagePending(false)
   602→   602→        setPhase('idle')
   603→   603→      }
   604→   604→    },
   605→   605→    [auth, refreshConversations, refreshQuota],
   606→   606→  )
   607→   607→
   608→   608→  /** Detiene la generación, la revelación o la herramienta en curso. */
   609→   609→  const stop = useCallback(() => {
   610→   610→    abortRef.current?.abort()
   611→   611→    abortRef.current = null
   612→   612→    pendingRef.current = false
   613→   613→    if (rafRef.current !== null) {
   614→   614→      cancelAnimationFrame(rafRef.current)
   615→   615→      rafRef.current = null
   616→   616→    }
   617→   617→    setPhase('idle')
   618→   618→    setWebSearching(false)
   619→   619→    setImagePending(false)
   620→   620→    setMessages((prev) => prev.map((m) => (m.complete || m.pendingImage ? m : { ...m, content: m.fullContent || m.content, complete: true })))
   621→   621→  }, [])
   622→   622→
   623→   623→  /** Regenera la última respuesta (conserva la búsqueda web y la imagen adjunta del turno). */
   624→   624→  const regenerate = useCallback(() => {
   625→   625→    if (!lastUserMsg || pendingRef.current) return
   626→   626→    // Se elimina la última respuesta asistente incompleta o final
   627→   627→    setMessages((prev) => {
   628→   628→      const idx = prev.map((m) => m.role).lastIndexOf('assistant')
   629→   629→      return idx === -1 ? prev : prev.filter((_, i) => i !== idx)
   630→   630→    })
   631→   631→    void send(lastUserMsg, {
   632→   632→      isRetry: true,
   633→   633→      webSearch: lastOptsRef.current.webSearch,
   634→   634→      attachment: lastOptsRef.current.attachment,
   635→   635→    })
   636→   636→  }, [lastUserMsg, send])
   637→   637→
   638→   638→  /**
   639→   639→   * Edita un mensaje del usuario: recorta la rama posterior (servidor y cliente)
   640→   640→   * y reenvía el texto corregido como mensaje nuevo.
   641→   641→   */
   642→   642→  const editMessage = useCallback(
   643→   643→    async (messageId: string, newContent: string) => {
   644→   644→      const text = newContent.trim()
   645→   645→      if (!text || pendingRef.current) return
   646→   646→      const idx = messages.findIndex((m) => m.id === messageId)
   647→   647→      if (idx === -1 || messages[idx]?.role !== 'user') return
   648→   648→      const convId = activeIdRef.current
   649→   649→      if (!convId) return
   650→   650→
   651→   651→      // Recorte en servidor: se conservan solo los mensajes anteriores a la edición.
   652→   652→      try {
   653→   653→        await fetch('/api/chat/trim', {
   654→   654→          method: 'POST',
   655→   655→          headers: { 'Content-Type': 'application/json' },
   656→   656→          body: JSON.stringify({ sessionId: convId, keepCount: idx }),
   657→   657→        })
   658→   658→      } catch {
   659→   659→        // si el recorte falla se continúa: la conversación local manda en la vista
   660→   660→      }
   661→   661→      setMessages((prev) => prev.slice(0, idx))
   662→   662→      void send(text)
   663→   663→    },
   664→   664→    [messages, send],
   665→   665→  )
   666→   666→
   667→   667→  /** Renombra una conversación (PATCH) y actualiza sidebar/buscador. */
   668→   668→  const renameConversation = useCallback(
   669→   669→    async (id: string, title: string) => {
   670→   670→      const clean = title.trim().slice(0, 80)
   671→   671→      if (!clean) return
   672→   672→      setConversations((prev) => prev.map((c) => (c.id === id ? { ...c, title: clean } : c)))
   673→   673→      useUiStore.setState((s) => ({ conversations: s.conversations.map((c) => (c.id === id ? { ...c, title: clean } : c)) }))
   674→   674→      try {
   675→   675→        await fetch(`/api/chat/sessions/${encodeURIComponent(id)}`, {
   676→   676→          method: 'PATCH',
   677→   677→          headers: { 'Content-Type': 'application/json' },
   678→   678→          body: JSON.stringify({ title: clean }),
   679→   679→        })
   680→   680→      } catch {
   681→   681→        // el título local se mantiene; el servidor quedará coherente al refrescar
   682→   682→      }
   683→   683→      void refreshConversations()
   684→   684→    },
   685→   685→    [refreshConversations],
   686→   686→  )
   687→   687→
   688→   688→  /** Exporta la conversación activa como Markdown descargable. */
   689→   689→  const exportActiveConversation = useCallback(() => {
   690→   690→    if (messages.length === 0) return
   691→   691→    const title = conversations.find((c) => c.id === activeId)?.title ?? 'Conversación OE AI'
   692→   692→    exportMessagesAsMarkdown(title, messages)
   693→   693→  }, [activeId, conversations, messages])
   694→   694→
   695→   695→  /** Nueva conversación (Ctrl+Shift+O). */
   696→   696→  const newChat = useCallback(() => {
   697→   697→    if (pendingRef.current) abortRef.current?.abort()
   698→   698→    activeIdRef.current = null
   699→   699→    setActiveId(null)
   700→   700→    setMessages([])
   701→   701→    setError(null)
   702→   702→    setRouteNote(null)
   703→   703→    setLastUserMsg(null)
   704→   704→    setFocusModel(null)
   705→   705→    setWebSearching(false)
   706→   706→    setImagePending(false)
   707→   707→    try {
   708→   708→      window.localStorage.removeItem(ACTIVE_KEY)
   709→   709→      window.sessionStorage.removeItem('oe-ai-pending-model')
   710→   710→      useUiStore.setState({ chatModel: 'auto' })
   711→   711→    } catch {
   712→   712→      // sin almacenamiento
   713→   713→    }
   714→   714→  }, [])
   715→   715→
   716→   716→  /** Elimina una conversación del historial. */
   717→   717→  const deleteConversation = useCallback(
   718→   718→    async (id: string) => {
   719→   719→      try {
   720→   720→        await fetch(`/api/chat/sessions/${encodeURIComponent(id)}`, { method: 'DELETE' })
   721→   721→      } catch {
   722→   722→        // si falla, la lista se refresca igualmente
   723→   723→      }
   724→   724→      if (activeIdRef.current === id) newChat()
   725→   725→      void refreshConversations()
   726→   726→    },
   727→   727→    [newChat, refreshConversations],
   728→   728→  )
   729→   729→
   730→   730→  // Limpieza al desmontar
   731→   731→  useEffect(() => {
   732→   732→    return () => {
   733→   733→      if (rafRef.current !== null) cancelAnimationFrame(rafRef.current)
   734→   734→      abortRef.current?.abort()
   735→   735→    }
   736→   736→  }, [])
   737→   737→
   738→   738→  const pending = phase !== 'idle'
   739→   739→
   740→   740→  return {
   741→   741→    conversations,
   742→   742→    activeId,
   743→   743→    messages,
   744→   744→    phase,
   745→   745→    pending,
   746→   746→    error,
   747→   747→    routeNote,
   748→   748→    focusModel,
   749→   749→    loadingHistory,
   750→   750→    lastUserMsg,
   751→   751→    webSearching,
   752→   752→    imagePending,
   753→   753→    guestQuota,
   754→   754→    guestBlocked,
   755→   755→    refreshQuota,
   756→   756→    send,
   757→   757→    stop,
   758→   758→    regenerate,
   759→   759→    newChat,
   760→   760→    loadConversation,
   761→   761→    deleteConversation,
   762→   762→    refreshConversations,
   763→   763→    setFocusModel,
   764→   764→    generateImage,
   765→   765→    editMessage,
   766→   766→    renameConversation,
   767→   767→    exportActiveConversation,
   768→   768→  }
   769→   769→}
