    44→/* ── Lectura en voz alta (una sola reproducción global) ─────────────────── */
    45→let globalAudio: HTMLAudioElement | null = null
    46→
    47→type TtsState = 'idle' | 'loading' | 'playing'
    48→
    49→function stopGlobalAudio() {
    50→  if (globalAudio) {
    51→    globalAudio.pause()
    52→    globalAudio.src = ''
    53→    globalAudio = null
    54→  }
    55→}

    57→function SpeakButton({ msg }: { msg: ChatMsg }) {
    58→  const [state, setState] = useState<TtsState>('idle')
    59→  const [note, setNote] = useState<string | null>(null)
    60→
    61→  // Al desmontar, si este audio es el global activo se detiene.
    62→  useEffect(() => {
    63→    return () => {
    64→      if (globalAudio && globalAudio.dataset.msgId === msg.id) stopGlobalAudio()
    65→    }
    66→  }, [msg.id])
    67→
    68→  async function toggle() {
    69→    if (state === 'playing') {
    70→      stopGlobalAudio()
    71→      setState('idle')
    72→      return
    73→    }
    74→    if (state === 'loading') return
    75→    setState('loading')
    76→    setNote(null)
    77→    stopGlobalAudio()
    78→    try {
    79→      const res = await fetch('/api/tools/tts', {
    80→        method: 'POST',
    81→        headers: { 'Content-Type': 'application/json' },
    82→        body: JSON.stringify({ text: msg.fullContent || msg.content }),
    83→      })
    84→      if (!res.ok) {
    85→        const data = (await res.json().catch(() => null)) as { error?: string } | null
    86→        throw new Error(data?.error ?? 'No se pudo generar el audio.')
    87→      }
    88→      const blob = await res.blob()
    89→      const audio = new Audio(URL.createObjectURL(blob))
    90→      audio.dataset.msgId = msg.id
    91→      globalAudio = audio
    92→      audio.onended = () => {
    93→        setState('idle')
    94→        if (globalAudio === audio) globalAudio = null
    95→      }
    96→      audio.onerror = () => setState('idle')
    97→      setState('playing')
    98→      void audio.play()
    99→    } catch (err) {
   100→      setState('idle')
   101→      setNote(err instanceof Error ? err.message : 'No se pudo generar el audio.')
   102→    }
   103→  }
   104→
   105→  const label =
   106→    state === 'playing'
   107→      ? 'Detener lectura'
   108→      : state === 'loading'
   109→        ? 'Generando audio…'
   110→        : 'Leer en voz alta (voz premium)'
   111→
   112→  return (
   113→    <>
   114→      <button
   115→        type="button"
   116→        onClick={() => void toggle()}
   117→        disabled={state === 'loading'}
   118→        aria-label={label}
   119→        title={label}
   120→        className={cn(
   121→          'inline-flex items-center gap-1 rounded-md px-1.5 py-0.5 text-[11px] transition-colors hover:bg-accent hover:text-foreground disabled:opacity-40',
   122→          state === 'playing' ? 'text-emerald-300' : 'text-muted-foreground',
   123→        )}
   124→      >
   125→        {state === 'loading' ? (
   126→          <motion.span
   127→            aria-hidden
   128→            animate={{ opacity: [0.4, 1, 0.4] }}
   129→            transition={{ duration: 1.1, repeat: Infinity }}
   130→            className="inline-flex"
   131→          >
   132→            <Volume2 className="size-3" />
   133→          </motion.span>
   134→        ) : state === 'playing' ? (
   135→          <Square className="size-3 fill-current" aria-hidden />
   136→        ) : (
   137→          <Volume2 className="size-3" aria-hidden />
   138→        )}
   139→        {state === 'playing' ? 'Detener' : 'Escuchar'}
   140→      </button>
   141→      {note && <span className="text-[11px] text-amber-300/90">{note}</span>}
   142→    </>
   143→  )
   144→}

   305→/* ── Burbuja del usuario con edición ────────────────────────────────────── */
   306→
   307→function UserBubble({ msg }: { msg: ChatMsg }) {
   308→  const chat = useChat()
   309→  const [editing, setEditing] = useState(false)
   310→  const [draft, setDraft] = useState(msg.content)
   311→
   312→  function save() {
   313→    const text = draft.trim()
   314→    setEditing(false)
   315→    if (!text || text === msg.content) return
   316→    void chat.editMessage(msg.id, text)
   317→  }
   318→
   319→  if (editing) {
   320→    return (
   321→      <div className="w-full min-w-[240px] rounded-3xl rounded-br-lg border border-emerald-500/40 bg-secondary/60 p-3">
   322→        <label htmlFor={`edit-${msg.id}`} className="sr-only">
   323→          Editar mensaje
   324→        </label>
   325→        <Textarea
   326→          id={`edit-${msg.id}`}
   327→          value={draft}
   328→          onChange={(e) => setDraft(e.target.value)}
   329→          onKeyDown={(e) => {
   330→            if (e.key === 'Enter' && !e.shiftKey) {
   331→              e.preventDefault()
   332→              save()
   333→            }
   334→            if (e.key === 'Escape') setEditing(false)
   335→          }}
   336→          rows={3}
   337→          autoFocus
   338→          className="scrollbar-slim min-h-0 w-full resize-none border-0 bg-transparent px-0 py-0 text-sm leading-relaxed shadow-none focus-visible:ring-0 dark:bg-transparent"
   339→        />
   340→        <div className="mt-2 flex items-center justify-end gap-2">
   341→          <Button size="sm" variant="ghost" onClick={() => setEditing(false)}>
   342→            Cancelar
   343→          </Button>
   344→          <Button size="sm" className="gap-1.5" onClick={save} disabled={!draft.trim() || chat.pending}>
   345→            <Check className="size-3.5" aria-hidden />
   346→            Guardar y regenerar
   347→          </Button>
   348→        </div>
   349→      </div>
   350→    )
   351→  }
   352→
   353→  return (
   354→    <div className="group/user flex flex-col items-end gap-1">
   355→      {/* Imagen adjuntada por el usuario (turno de visión) */}
   356→      {msg.attachmentUrl && (
   357→        <a
   358→          href={msg.attachmentUrl}
   359→          target="_blank"
   360→          rel="noreferrer"
   361→          aria-label="Abrir imagen adjunta en tamaño completo"
   362→          className="block max-w-[85%] overflow-hidden rounded-3xl rounded-br-lg ring-1 ring-border/60 transition-shadow hover:ring-emerald-500/40"
   363→        >
   364→          <img
   365→            src={msg.attachmentUrl}
   366→            alt={msg.content ? `Imagen adjunta: ${msg.content}` : 'Imagen adjunta por el usuario'}
   367→            className="max-h-56 w-auto max-w-full bg-secondary/30"
   368→            loading="lazy"
   369→          />
   370→        </a>
   371→      )}
   372→      {msg.content && (
   373→        <div className="rounded-3xl rounded-br-lg bg-primary/15 px-4 py-3 text-sm leading-relaxed whitespace-pre-wrap break-words text-foreground ring-1 ring-primary/20">
   374→          {msg.content}
   375→        </div>
   376→      )}
   377→      <button
   378→        type="button"
   379→        onClick={() => {
   380→          setDraft(msg.content)
   381→          setEditing(true)
   382→        }}
   383→        disabled={chat.pending}
   384→        aria-label="Editar mensaje y reenviar"
   385→        title="Editar y reenviar"
   386→        className="mr-1 inline-flex items-center gap-1 rounded-md px-1.5 py-0.5 text-[11px] text-muted-foreground opacity-0 transition-opacity hover:bg-accent hover:text-foreground focus-visible:opacity-100 group-hover/user:opacity-100 disabled:pointer-events-none [@media(pointer:coarse)]:opacity-100"
   387→      >
   388→        <Pencil className="size-3" aria-hidden />
   389→        Editar
   390→      </button>
   391→    </div>
   392→  )
   393→}

   528→          {typeof msg.meta?.latencyMs === 'number' && msg.meta.latencyMs > 0 && (
   529→            <Badge variant="outline" className="border-border/60 text-[10px] text-muted-foreground">
   530→              {(msg.meta.latencyMs / 1000).toFixed(1)} s
   531→            </Badge>
   532→          )}
   533→          {msg.meta?.mode === 'reflective' && (
   534→            <Badge variant="outline" className="border-emerald-500/30 text-[10px] text-emerald-300">
   535→              Reflexivo
   536→            </Badge>
   537→          )}
   538→          <CopyTextButton text={msg.fullContent} label="Copiar respuesta" />
   539→          <button
   540→            type="button"
   541→            onClick={chat.regenerate}
   542→            disabled={chat.pending}
   543→            aria-label="Regenerar respuesta"
   544→            className="inline-flex items-center gap-1 rounded-md px-1.5 py-0.5 text-[11px] text-muted-foreground transition-colors hover:bg-accent hover:text-foreground disabled:opacity-40"
   545→          >
   546→            <RotateCcw className="size-3" aria-hidden />
   547→            Regenerar
   548→          </button>
   549→          <SpeakButton msg={msg} />
   550→        </div>
   551→      )}
   552→    </div>
   553→  )
   554→}

   558→function ImageContent({ msg }: { msg: ChatMsg }) {
   559→  const chat = useChat()
   560→  const [copied, setCopied] = useState(false)
   561→  const caption = msg.content
   562→

   589→          {caption}
   590→        </p>

   592→
   593→      <div className="mt-1 flex flex-wrap items-center gap-1.5 opacity-70 transition-opacity group-hover/img:opacity-100">

   634→          <RotateCcw className="size-3" aria-hidden />
   635→          Volver a generar
   636→        </button>
   637→      </div>
   638→    </div>
   639→  )
