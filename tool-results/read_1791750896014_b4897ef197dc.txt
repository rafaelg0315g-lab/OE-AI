     4→ * Flujo por solicitud:

    16→ *      licencias y cómo ejecutar.
    17→ *   6. Persistencia en BuildProject/BuildFile → descargable (ZIP) y con vista

    20→ * Modificación: "update" reescribe SOLO los archivos afectados con el contexto

    22→ *
    23→ * Reutilización responsable: el código SIEMPRE es propio (lo escribe el modelo

    25→ * licencia queda registrada en la procedencia del proyecto.

   106→function s(v: unknown, fallback: string, max = 120): string {
   107→  const str = typeof v === 'string' && v.trim() ? v.trim() : fallback
   108→  return str.slice(0, max)
   109→}

   111→function sanitizePagePath(raw: unknown): string | null {
   112→  const p = s(raw, '', 60).toLowerCase().replace(/\\/g, '/').replace(/^\.?\//, '')
   113→  if (!p || !/^[\w-]+\.html$/.test(p) || p.includes('..')) return null
   114→  return p
   115→}

   117→function parsePlan(data: RawPlan, fallbackRefs: CatalogTemplate[]): FactoryPlan {
   118→  const title = s(data.title, 'Sitio Web', 60)
   119→  const summary = s(data.summary, '', 400)
   120→
   121→  const rawPages = Array.isArray(data.pages) ? data.pages : []
   122→  const pages: FactoryPage[] = []
   123→  for (const item of rawPages) {
   124→    if (!item || typeof item !== 'object') continue
   125→    const rec = item as { path?: unknown; title?: unknown; sections?: unknown }
   126→    const path = sanitizePagePath(rec.path)
   127→    if (!path) continue
   128→    const sections = Array.isArray(rec.sections)
   129→      ? rec.sections.map((x) => s(x, 'sección', 80)).slice(0, 8)
   130→      : []
   131→    pages.push({ path, title: s(rec.title, title, 80), sections })
   132→    if (pages.length >= MAX_PAGES) break
   133→  }
   134→  if (pages.length === 0) throw new WebFactoryError('El plan no produjo páginas utilizables.', 'invalid_plan')
   135→  // index.html SIEMPRE existe.
   136→  if (!pages.some((p) => p.path === 'index.html')) {
   137→    pages.unshift({ path: 'index.html', title, sections: ['hero', 'características', 'cta', 'footer'] })
   138→  }
   139→
   140→  const ds = data.designSystem ?? {}
   141→  const designSystem: Record<string, string> = {
   142→    paletteName: s(ds.paletteName, 'Elegante oscuro', 40),
   143→    bg: s(ds.bg, '#0c0f14', 20),
   144→    surface: s(ds.surface, '#121722', 20),
   145→    text: s(ds.text, '#e8eaf0', 20),
   146→    muted: s(ds.muted, '#9aa3b2', 20),
   147→    primary: s(ds.primary, '#10b981', 20),
   148→    accent: s(ds.accent, '#d4b483', 20),
   149→    radius: s(ds.radius, '14px', 12),
   150→    shadow: s(ds.shadow, '0 18px 40px rgba(0,0,0,.35)', 80),
   151→    headingFont: s(ds.headingFont, 'Georgia, "Times New Roman", serif', 90),
   152→    bodyFont: s(ds.bodyFont, 'system-ui, -apple-system, "Segoe UI", Roboto, sans-serif', 90),
   153→    motion: s(ds.motion, 'revelación sutil al hacer scroll, microinteracciones en botones', 140),
   154→  }
   155→
   156→  const images: FactoryPlan['images'] = []
   157→  if (Array.isArray(data.images)) {
   158→    for (const img of data.images.slice(0, 2)) {
   159→      if (!img || typeof img !== 'object') continue
   160→      const rec = img as { slot?: unknown; prompt?: unknown; size?: unknown }
   161→      const slot = s(rec.slot, '', 40).replace(/[^\w-]/g, '')
   162→      const prompt = s(rec.prompt, '', 300)
   163→      if (!slot || !prompt) continue
   164→      const size = ['1344x768', '1152x864', '1024x1024'].includes(String(rec.size)) ? String(rec.size) : '1344x768'
   165→      images.push({ slot, prompt, size })
   166→    }
   167→  }
   168→
   169→  const catalogUrls = new Set(fallbackRefs.map((r) => r.repoUrl))
   170→  const references: FactoryPlan['references'] = []
   171→  if (Array.isArray(data.references)) {
   172→    for (const ref of data.references.slice(0, 4)) {
   173→      if (!ref || typeof ref !== 'object') continue
   174→      const rec = ref as { name?: unknown; repoUrl?: unknown; license?: unknown; usage?: unknown }
   175→      const repoUrl = s(rec.repoUrl, '', 200)
   176→      if (!catalogUrls.has(repoUrl)) continue // solo referencias reales del catálogo
   177→      const cat = fallbackRefs.find((r) => r.repoUrl === repoUrl)
   178→      references.push({
   179→        name: s(rec.name, cat?.name ?? 'Recurso', 80),
   180→        repoUrl,
   181→        license: cat?.licenseSpdx ?? (cat?.status === 'referencia' ? 'sin licencia clara' : 'MIT'),
   182→        usage: 'referencia-visual',
   183→      })
   184→    }
   185→  }
   186→  if (references.length === 0 && fallbackRefs.length > 0) {
   187→    for (const cat of fallbackRefs.slice(0, 3)) {
   188→      references.push({
   189→        name: cat.name,
   190→        repoUrl: cat.repoUrl,
   191→        license: cat.licenseSpdx ?? (cat.status === 'referencia' ? 'sin licencia clara' : 'MIT'),
   192→        usage: 'referencia-visual',
   193→      })
   194→    }
   195→  }
   196→
   197→  return {
   198→    title,
   199→    summary: summary || 'Sitio web profesional generado con OE AI Web Factory.',
   200→    siteType: s(data.siteType, 'sitio web profesional', 60),
   201→    aesthetic: s(data.aesthetic, 'moderno y elegante', 120),
   202→    designSystem,
   203→    pages,
   204→    interactions: Array.isArray(data.interactions) ? data.interactions.map((x) => s(x, '', 100)).filter(Boolean).slice(0, 6) : [],
   205→    images,
   206→    references,
   207→  }
   208→}

   316→  'data:image/svg+xml;utf8,' +

   321→async function generatePlanImages(
   322→  sessionId: string,
   323→  images: FactoryPlan['images'],
   324→): Promise<Array<{ slot: string; assetId: string; url: string }>> {
   325→  if (images.length === 0) return []
   326→  const out: Array<{ slot: string; assetId: string; url: string }> = []
   327→  await Promise.all(
   328→    images.map(async (img) => {
   329→      try {
   330→        const { default: ZAI } = await import('z-ai-web-dev-sdk')
   331→        const zai = await ZAI.create()
   332→        const result = await zai.images.generations.create({ prompt: img.prompt, size: img.size as '1344x768' })
   333→        const base64 = result?.data?.[0]?.base64
   334→        if (!base64) return
   335→        const buffer = Buffer.from(base64, 'base64')
   336→        const asset = await db.generatedAsset.create({
   337→          data: { sessionId, prompt: img.prompt.slice(0, 900), size: img.size, mime: 'image/png', data: buffer },
   338→        })
   339→        out.push({ slot: img.slot, assetId: asset.id, url: `/api/assets/${asset.id}` })
   340→      } catch (err) {
   341→        console.error(`[web-factory] imagen "${img.slot}" falló (continúa con degradado):`, err instanceof Error ? err.message.slice(0, 120) : err)
   342→      }
   343→    }),
   344→  )
   345→  return out
   346→}

   348→function injectImages(html: string, images: Array<{ slot: string; url: string }>): string {
   349→  let out = html
   350→  for (const img of images) {
   351→    out = out.replaceAll(`{{IMG:${img.slot}}}`, img.url)
   352→  }
   353→  // Slots sin imagen (o fallo): placeholder mínimo válido (el degradado .wf-media manda).
   354→  out = out.replace(/{{IMG:[\w-]+}}/g, TINY_SVG)
   355→  return out
   356→}

   385→    fileNames.map((f) => `- ${f}`).join('\n'),
   386→    '',
   387→    '## Personalización',
   388→    '- Colores/radios/sombras: edita las variables `:root` al inicio de `styles.css`.',
   389→    '- Textos: cada página es HTML autocontenido y legible.',
   390→    '- Imágenes: sustituye las URLs `/api/assets/...` por tus propios archivos (o borra el `<img>`; el degradado `.wf-media` mantiene el diseño).',
   391→    '',
   392→    '## Procedencia y licencias (transparencia)',
   393→    'El código de este sitio fue escrito íntegramente por OE AI (sin copia de repos). Referencias de diseño consultadas:',
   394→    refs || '- Ninguna (diseño propio).',
   395→    '',
   396→    '## Validación automática',
   397→    checkLines,
   398→    '',
   399→    '— OE AI Web Factory · OE BUSINESS',
   400→  ].join('\n')
   401→}

   403→/* ── Pipeline ─────────────────────────────────────────────────────────────── */
   404→
   405→export async function runWebFactory(input: {
   406→  brief: string
   407→  sessionId: string
   408→  existingProjectId?: string
   409→  signal?: AbortSignal
   410→}): Promise<Omit<FactoryResult, 'id' | 'previewUrl' | 'latencyMs'> & { planJson: string; readme: string }> {
   411→  const start = Date.now()
   412→  const isUpdate = Boolean(input.existingProjectId)
   413→
   414→  // Modo edición: cargar proyecto existente.
   415→  let existing: { title: string; files: Array<{ path: string; content: string }> } | null = null
   416→  if (input.existingProjectId) {
   417→    const project = await db.buildProject.findUnique({
   418→      where: { id: input.existingProjectId },
   419→      select: { title: true, files: { select: { path: true, content: true } } },
   420→    })
   421→    if (!project) throw new WebFactoryError('El sitio a modificar no existe o fue eliminado.', 'no_files')
   422→    existing = { title: project.title, files: project.files.map((f) => ({ path: f.path, content: f.content })) }
   423→  }
   424→
   425→  let plan: FactoryPlan
   426→  if (!existing) {
   427→    // 1) Plan con catálogo real
   428→    const [refs, componentKinds] = await Promise.all([pickReferences(input.brief), listComponentKinds()])
   429→    const planStart = Date.now()
   430→    const planRes = await buildChatJson<RawPlan>(
   431→      plannerSystemPrompt(),
   432→      plannerUserPrompt(input.brief, refs, componentKinds, null),
   433→      input.signal,
   434→    )
   435→    plan = parsePlan(planRes.data, refs)
   436→    console.error(`[web-factory] plan ${((Date.now() - planStart) / 1000).toFixed(1)}s vía ${planRes.core} · ${plan.pages.length} páginas · ${plan.references.length} referencias`)
   437→  } else {
   438→    // Plan mínimo heredado del proyecto existente (para README/procedencia).
   439→    plan = {
   440→      title: existing.title,
   441→      summary: s('', 'Sitio actualizado con OE AI Web Factory.', 400),
   442→      siteType: 'sitio web',
   443→      aesthetic: 'consistente con el diseño existente',
   444→      designSystem: {},
   445→      pages: existing.files.filter((f) => f.path.endsWith('.html')).map((f) => ({
   446→        path: f.path,
   447→        title: f.path.replace('.html', ''),
   448→        sections: [],
   449→      })),
   450→      interactions: [],
   451→      images: [],
   452→      references: [],
   453→    }
   454→  }
   455→
   456→  let files: Array<{ path: string; content: string }>
   457→  let images: Array<{ slot: string; assetId: string; url: string }> = []
   458→
   459→  if (!existing) {
   460→    // 2) Shell compartido (styles.css + app.js)
   461→    const shellStart = Date.now()
   462→    const shell = await buildChatJson<{ css?: unknown; js?: unknown }>(
   463→      shellSystemPrompt(),
   464→      `Sistema de diseño:\n${JSON.stringify(plan.designSystem)}\n\nInteracciones pedidas: ${plan.interactions.join('; ') || 'menú móvil y reveal al scroll'}.`,
   465→      input.signal,
   466→    )
   467→    const css = s(shell.data.css, '', 60_000)
   468→    const js = s(shell.data.js, '', 30_000)
   469→    if (!css.includes('--wf-bg') || !js.includes('addEventListener')) {
   470→      throw new WebFactoryError('El shell generado no cumplió el contrato mínimo.', 'invalid_plan')
   471→    }
   472→    files = [
   473→      { path: 'styles.css', content: css },
   474→      { path: 'app.js', content: js },
   475→    ]
   476→    console.error(`[web-factory] shell ${((Date.now() - shellStart) / 1000).toFixed(1)}s vía ${shell.core} · css ${css.length}B · js ${js.length}B`)
   477→
   478→    // 3) Imágenes (best-effort, en paralelo)
   479→    images = await generatePlanImages(input.sessionId, plan.images)
   480→
   481→    // 4) Páginas (comparten el shell → coherencia). Arranque escalonado para no
   482→    //    golpear el núcleo con todas las peticiones a la vez, y UN reintento por
   483→    //    página fallida: es la fase más cara y perder una página no debe tirar
   484→    //    todo el sitio (causa raíz histórica del error no_files).
   485→    const planJson = JSON.stringify(plan)
   486→    const imageSlots = plan.images.map((i) => i.slot)
   487→    const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms))
   488→    const buildPageOnce = async (page: FactoryPlan['pages'][number]): Promise<string | null> => {
   489→      try {
   490→        const res = await buildChatJson<{ html?: unknown }>(
   491→          pageSystemPrompt(),
   492→          pageUserPrompt(planJson, page, plan.pages, imageSlots),
   493→        )
   494→        const html = s(res.data.html, '', 90_000)
   495→        return /<html/i.test(html) ? html : null
   496→      } catch (err) {
   497→        console.error(`[web-factory] página ${page.path} falló: ${err instanceof Error ? err.message : 'sin detalle'}`)
   498→        return null
   499→      }
   500→    }
   501→    const pageHtmls = await Promise.all(
   502→      plan.pages.map(async (page, i) => {
   503→        if (i > 0) await sleep(Math.min(i * 1200, 4000))
   504→        let html = await buildPageOnce(page)
   505→        if (!html) {
   506→          await sleep(2500)
   507→          html = await buildPageOnce(page)
   508→          if (html) console.error(`[web-factory] página ${page.path} recuperada en el reintento`)
   509→        }
   510→        return html
   511→      }),
   512→    )
   513→    for (let i = 0; i < pageHtmls.length; i++) {
   514→      const html = pageHtmls[i]
   515→      if (html) files.push({ path: plan.pages[i].path, content: injectImages(html, images) })
   516→    }
   517→    const okPages = files.filter((f) => f.path.endsWith('.html')).length
   518→    if (okPages === 0) {
   519→      throw new WebFactoryError('Ninguna página se pudo componer. Inténtalo de nuevo.', 'no_files')
   520→    }
   521→    const failedPages = pageHtmls.filter((h) => !h).length
   522→    if (failedPages > 0) console.error(`[web-factory] ${failedPages} página(s) fallaron; se entregan ${okPages}`)
   523→  } else {
   524→    // Modo edición: una llamada con el contexto completo.
   525→    const res = await buildChatJson<{ summary?: unknown; files?: unknown }>(
   526→      updaterSystemPrompt(),
   527→      [
   528→        `Sitio: ${existing.title}`,
   529→        `Archivos actuales:\n${existing.files.map((f) => `--- ${f.path} ---\n${f.content.slice(0, 9000)}`).join('\n\n').slice(0, 60_000)}`,
   530→        `Petición del usuario:\n"""\n${input.brief.slice(0, 2500)}\n"""`,
   531→        'Devuelve el JSON con los archivos completos modificados.',
   532→      ].join('\n\n'),
   533→      input.signal,
   534→    )
   535→    const changed = Array.isArray(res.data.files)
   536→      ? (res.data.files as Array<{ path?: unknown; content?: unknown }>)
   537→          .map((f) => ({ path: sanitizePagePath(f.path) ?? (typeof f.path === 'string' && /^(styles\.css|app\.js|README\.md)$/.test(f.path) ? f.path : null), content: typeof f.content === 'string' ? f.content : '' }))
   538→          .filter((f): f is { path: string; content: string } => Boolean(f.path) && f.content.trim().length > 0)
   539→          .slice(0, 5)
   540→      : []
   541→    if (changed.length === 0) throw new WebFactoryError('El editor no produjo cambios utilizables.', 'no_files')
   542→    files = changed.map((f) => ({ path: f.path, content: injectImages(f.content, images) }))
   543→    plan.summary = s(res.data.summary, plan.summary, 400)
   544→  }
   545→
   546→  // Límite de archivos
   547→  if (files.length > MAX_FILES) files = files.slice(0, MAX_FILES)
   548→
   549→  // 5) Validación honesta (puede autofijar metadatos/alt)
   550→  const { checks, warnings } = validateSite({ files })
   551→
   552→  // 6) README determinista con procedencia y licencias
   553→  files = files.filter((f) => f.path !== 'README.md')
   554→  files.push({ path: 'README.md', content: buildReadme(plan, files.map((f) => f.path), checks) })
   555→
   556→  return {
   557→    title: plan.title,
   558→    summary: plan.summary,
   559→    mode: 'web',
   560→    isUpdate,
   561→    changedPaths: files.map((f) => f.path),
   562→    files,
   563→    designSystem: plan.designSystem,
   564→    references: plan.references,
   565→    checks,
   566→    warnings: warnings.slice(0, 12),
   567→    images: images.map(({ slot, assetId, url }) => ({ slot, assetId, url })),
   568→    planJson: JSON.stringify(plan),
   569→    readme: '',
   570→    // latencia la calcula el llamador
   571→    ...({ latencyMs: Date.now() - start } as object),
   572→  }
   573→}
