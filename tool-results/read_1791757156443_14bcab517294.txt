   133→  switch (errorType) {
   134→    case 'no_files':
   135→      return 'el núcleo no devolvió archivos utilizables (inténtalo de nuevo).'
   136→    case 'rate_limited':
   137→      return 'el núcleo alcanzó su límite de peticiones; espera un momento e inténtalo de nuevo.'
   138→    case 'empty_response':
   139→      return 'el núcleo devolvió una respuesta vacía.'
   140→    case 'json_parse':
   141→      return 'la respuesta del núcleo no se pudo interpretar.'
   142→    case 'provider_error':
   143→      return 'el proveedor de código falló.'
   144→    case 'invalid_plan':
   145→      return 'el plan generado no fue válido.'
   146→    case 'aborted':
   147→      return 'la construcción se canceló.'
   148→    default:
   149→      return 'error interno del constructor.'

   198→/**
   199→ * Extrae la directiva ```oe-build del texto del asistente (constructor de apps).
   200→ * Devuelve el brief refinado y el texto de anuncio previo a la valla.
   201→ */
   202→interface BuildDirectivePayload {
   203→  name?: string
   204→  brief: string
   205→  mode?: string
   206→  projectId?: string
   207→  kind: 'app' | 'web'
   208→  prefix: string
   209→}

   211→function extractBuildDirective(text: string): BuildDirectivePayload | null {
   212→  const webIdx = text.indexOf('```oe-web')
   213→  const appIdx = text.indexOf('```oe-build')
   214→  const kind: 'app' | 'web' = webIdx !== -1 && (appIdx === -1 || webIdx < appIdx) ? 'web' : 'app'
   215→  const fenceStart = kind === 'web' ? webIdx : appIdx
   216→  if (fenceStart === -1) return null
   217→  const bodyStart = text.indexOf('\n', fenceStart)
   218→  if (bodyStart === -1) return null
   219→  const fenceEnd = text.indexOf('```', bodyStart)
   220→  if (fenceEnd === -1) return null
   221→  const raw = text.slice(bodyStart, fenceEnd).trim()
   222→  const start = raw.indexOf('{')
   223→  if (start === -1) return null
   224→  let depth = 0
   225→  let inString = false
   226→  let escaped = false
   227→  let end = -1
   228→  for (let i = start; i < raw.length; i++) {
   229→    const ch = raw[i]
   230→    if (inString) {
   231→      if (escaped) escaped = false
   232→      else if (ch === '\\') escaped = true
   233→      else if (ch === '"') inString = false
   234→      continue
   235→    }
   236→    if (ch === '"') inString = true
   237→    else if (ch === '{') depth++
   238→    else if (ch === '}') {
   239→      depth--
   240→      if (depth === 0) {
   241→        end = i
   242→        break
   243→      }
   244→    }
   245→  }
   246→  if (end === -1) return null
   247→  try {
   248→    const parsed = JSON.parse(raw.slice(start, end + 1)) as {
   249→      name?: unknown
   250→      brief?: unknown
   251→      mode?: unknown
   252→      projectId?: unknown
   253→    }
   254→    const brief = typeof parsed.brief === 'string' ? parsed.brief.trim() : ''
   255→    if (!brief) return null
   256→    const prefix = text
   257→      .slice(0, fenceStart)
   258→      .replace(/```[a-z]*\s*$/, '')
   259→      .trim()
   260→    return {
   261→      brief: brief.slice(0, 6000),
   262→      name: typeof parsed.name === 'string' ? parsed.name : undefined,
   263→      mode: typeof parsed.mode === 'string' ? parsed.mode : undefined,
   264→      projectId: typeof parsed.projectId === 'string' ? parsed.projectId : undefined,
   265→      kind,
   266→      prefix,
   267→    }
   268→  } catch {
   269→    return null
   270→  }
   271→}

   273→/** Restaura el último proyecto construido desde el historial (bloques oeproject). */
   274→function extractLastBuildFromHistory(
   275→  messages: Array<{ content: string }>,
   276→): { projectId: string; title: string; files: string[]; kind: 'app' | 'web' } | null {
   277→  const re = /```oeproject\s*\n([\s\S]*?)```/g
   278→  let last: { projectId: string; title: string; files: string[]; kind: 'app' | 'web' } | null = null
   279→  for (const m of messages) {
   280→    let match: RegExpExecArray | null
   281→    re.lastIndex = 0
   282→    while ((match = re.exec(m.content)) !== null) {
   283→      try {
   284→        const data = JSON.parse(match[1]) as { id?: unknown; title?: unknown; kind?: unknown; files?: Array<{ path?: unknown }> }
   285→        if (typeof data.id === 'string' && Array.isArray(data.files)) {
   286→          last = {
   287→            projectId: data.id,
   288→            title: typeof data.title === 'string' ? data.title : 'App construida',
   289→            files: data.files
   290→              .map((f) => (typeof f?.path === 'string' ? f.path : ''))
   291→              .filter(Boolean)
   292→              .slice(0, 12),
   293→            kind: data.kind === 'web' ? 'web' : 'app',
   294→          }
   295→        }
   296→      } catch {
   297→        // bloque inválido: ignorar
   298→      }
   299→    }
   300→  }
   301→  return last
   302→}

   417→  /**
   418→   * Sondeo tras un fallo de conexión del POST de construcción (502 del gateway,
   419→   * red, recarga): el build sigue en el servidor. Cuando el proyecto queda
   420→   * listo, las tarjetas de archivos aparecen solas, sin recargar la página.
   421→   */
   422→  const resolveBuildByStatus = useCallback(
   423→    async (args: {
   424→      convId: string
   425→      assistantId: string
   426→      serverMessageId: string | null
   427→      projectId?: string
   428→      startedAt: number
   429→      prefix: string
   430→      isWeb: boolean
   431→      fallbackError: string
   432→    }) => {
   433→      const { convId, assistantId, serverMessageId, projectId, startedAt, prefix, isWeb, fallbackError } = args
   434→      const persist = (content: string) => {
   435→        if (!serverMessageId) return
   436→        void fetch('/api/chat/messages', {
   437→          method: 'PATCH',
   438→          headers: { 'Content-Type': 'application/json' },
   439→          body: JSON.stringify({ id: serverMessageId, content }),
   440→        }).catch(() => undefined)
   441→      }
   442→      const finish = (content: string, persistIt: boolean) => {
   443→        setMessages((prev) =>
   444→          prev.map((m) =>
   445→            m.id === assistantId ? { ...m, content, fullContent: content, pendingProject: false, complete: true } : m,
   446→          ),
   447→        )
   448→        if (persistIt) persist(content)
   449→      }
   450→      const deadline = Date.now() + BUILD_POLL_MAX_MS
   451→      let lastSeenBuilding = false
   452→      while (Date.now() < deadline) {
   453→        await new Promise((r) => setTimeout(r, BUILD_POLL_INTERVAL_MS))
   454→        const p = await fetchBuildStatus(convId, projectId)
   455→        if (!p) continue
   456→        const matches = p.id === projectId || new Date(p.createdAt).getTime() >= startedAt - 5000
   457→        if (!matches) continue
   458→        if (p.status === 'ready' && Array.isArray(p.files) && p.files.length > 0) {
   459→          const newContent = oeProjectContent(prefix, p, isWeb)
   460→          finish(newContent, true)
   461→          lastBuildRef.current = {
   462→            projectId: p.id,
   463→            title: p.title ?? 'App construida',
   464→            files: (p.files ?? []).map((f) => f.path).slice(0, 12),
   465→            kind: isWeb ? 'web' : 'app',
   466→          }
   467→          void refreshQuota()
   468→          void refreshConversations()
   469→          return
   470→        }
   471→        if (p.status === 'error') {
   472→          // Fallo definitivo del pipeline: nota honesta persistida.
   473→          finish(`${prefix ? `${prefix}\n\n` : ''}⚠️ No se pudo completar la construcción: ${describeBuildError(p.errorType)}`, true)
   474→          return
   475→        }
   476→        lastSeenBuilding = p.status === 'building'
   477→      }
   478→      if (lastSeenBuilding) {
   479→        // El build sigue en curso tras el plazo: aviso SOLO local (sin PATCH)
   480→        // para no borrar la directiva; al recargar, el healing la recupera.
   481→        finish(`${prefix ? `${prefix}\n\n` : ''}⏳ La construcción sigue en curso en el servidor. Recarga la conversación en un momento para ver el resultado.`, false)
   482→        return
   483→      }
   484→      finish(`${prefix ? `${prefix}\n\n` : ''}⚠️ No se pudo iniciar la construcción: ${fallbackError}`, true)
   485→    },
   486→    [fetchBuildStatus, refreshConversations, refreshQuota],
   487→  )

   489→  /**
   490→   * Recuperación al cargar una conversación: mensajes con directiva
   491→   * ```oe-build/```oe-web sin resolver (o nota de error del constructor) se
   492→   * contrastan con el estado real del servidor. Si el proyecto terminó listo,
   493→   * las tarjetas se muestran sin que el usuario tenga que hacer nada.
   494→   */
   495→  const healUnresolvedBuilds = useCallback(
   496→    async (convId: string, msgs: ChatMsg[]) => {
   497→      if (msgs.length === 0) return
   498→      let projects: BuildStatusSummary[] | null = null
   499→      try {
   500→        const res = await fetch(`/api/tools/build-app/status?sessionId=${encodeURIComponent(convId)}`, { cache: 'no-store' })
   501→        if (res.ok) {
   502→          const data = (await res.json().catch(() => null)) as { projects?: BuildStatusSummary[] } | null
   503→          projects = data?.projects ?? null
   504→        }
   505→      } catch {
   506→        return
   507→      }
   508→      if (!projects || projects.length === 0) return
   509→
   510→      for (let i = 0; i < msgs.length; i++) {
   511→        const msg = msgs[i]
   512→        if (msg.role !== 'assistant') continue
   513→        const content = msg.fullContent || msg.content
   514→        if (content.includes('```oeproject')) continue
   515→        const directive = extractBuildDirective(content)
   516→        const isBuildErrorNote = !directive && BUILD_ERROR_NOTE_RE.test(content)
   517→        if (!directive && !isBuildErrorNote) continue
