   133→  async function submitWithDoc(text: string) {
   134→    const doc = pendingDoc
   135→    if (!doc || docReading) return
   136→    setDocReading(true)
   137→    try {
   138→      let body: Record<string, unknown>
   139→      if (doc.kind === 'url') {
   140→        body = { url: doc.url, name: doc.name }
   141→      } else {
   142→        const file = doc.file
   143→        if (!file) throw new Error('El archivo del documento ya no está disponible.')
   144→        const dataUrl = await new Promise<string>((resolve, reject) => {
   145→          const reader = new FileReader()
   146→          reader.onload = () => resolve(String(reader.result))
   147→          reader.onerror = () => reject(new Error('No se pudo leer el PDF.'))
   148→          reader.readAsDataURL(file)
   149→        })
   150→        body = { fileDataUrl: dataUrl, name: doc.name }
   151→      }
   152→      const res = await fetch('/api/tools/read-doc', {
   153→        method: 'POST',
   154→        headers: { 'Content-Type': 'application/json' },
   155→        body: JSON.stringify(body),
   156→      })
   157→      const data = (await res.json().catch(() => null)) as { title?: string; text?: string; error?: string } | null
   158→      if (!res.ok || !data?.text) {
   159→        throw new Error(data?.error ?? 'No se pudo leer el documento.')
   160→      }
   161→      const title = (data.title ?? '').trim() || doc.name
   162→      toast.success(`Documento leído: ${title}`)
   163→      setPendingDoc(null)
   164→      setInput('')
   165→      try {
   166→        window.localStorage.setItem(`${DRAFT_PREFIX}${chat.activeId ?? 'new'}`, '')
   167→      } catch {
   168→        // sin almacenamiento
   169→      }
   170→      void chat.send(text, { docContext: data.text, docTitle: title })
   171→    } catch (err) {
   172→      toast.error(err instanceof Error ? err.message : 'No se pudo leer el documento.')
   173→    } finally {
   174→      setDocReading(false)
   175→    }
   176→  }

   178→  function submit() {
   179→    const text = input.trim()

   181→

   185→      return

   187→
   188→    // Imagen adjuntada: turno de visión (el núcleo analiza la imagen con el texto).
   189→    if (attachment) {

   191→      const att = attachment
   192→      setInput('')
   193→      setAttachment(null)
   194→      setAttachNote(null)
   195→      try {
   196→        window.localStorage.setItem(`${DRAFT_PREFIX}${chat.activeId ?? 'new'}`, '')

   199→      }
   200→      void chat.send(finalText, { attachment: { assetId: att.assetId, previewUrl: att.previewUrl } })
   201→      return

   203→
   204→    // Comandos del núcleo: /imagen y /buscar
   205→    if (text.startsWith('/imagen ')) {

   237→  /** Prepara y sube una imagen adjuntada (botón, arrastre o portapapeles). */
   238→  async function handleFiles(files: FileList | File[] | null) {
   239→    const file = Array.from(files ?? [])[0]
   240→    if (!file || uploading) return
   241→    setAttachNote(null)
   242→    if (!file.type.startsWith('image/')) {
   243→      setAttachNote('Solo se pueden adjuntar imágenes (PNG, JPEG, WebP, GIF).')
   244→      return
   245→    }
   246→    if (file.size > 12 * 1024 * 1024) {
   247→      setAttachNote('La imagen supera los 12 MB. Elige una más ligera.')
   248→      return
   249→    }
   250→    setUploading(true)
   251→    try {
   252→      const dataUrl = await fileToDownscaledDataUrl(file)
   253→      const res = await fetch('/api/assets/upload', {
   254→        method: 'POST',
   255→        headers: { 'Content-Type': 'application/json' },
   256→        body: JSON.stringify({ dataUrl }),
   257→      })
   258→      const data = (await res.json().catch(() => null)) as { id?: string; url?: string; error?: string } | null
   259→      if (!res.ok || !data?.id || !data.url) {
   260→        throw new Error(data?.error ?? 'No se pudo subir la imagen.')
   261→      }
   262→      setAttachment({ assetId: data.id, previewUrl: data.url })
   263→    } catch (err) {
   264→      setAttachNote(err instanceof Error ? err.message : 'No se pudo subir la imagen.')
   265→    } finally {
   266→      setUploading(false)
   267→    }

   270→  function toggleDictation() {

   277→    if (!Ctor) return
   278→    const rec = new Ctor()
   279→    rec.lang = 'es-CO'
   280→    rec.continuous = false
   281→    rec.interimResults = false
   282→    rec.onresult = (e) => {
   283→      const transcript = Array.from({ length: e.results.length }, (_, i) => e.results[i][0]?.transcript ?? '')
   284→        .join(' ')
   285→        .trim()
   286→      if (transcript) setInput((prev) => (prev ? `${prev} ${transcript}` : transcript))
   287→    }
   288→    rec.onend = () => {
   289→      setListening(false)
   290→      recognitionRef.current = null
   291→    }
   292→    rec.onerror = () => {
   293→      setListening(false)
   294→      recognitionRef.current = null
   295→    }
   296→    recognitionRef.current = rec
   297→    setListening(true)
   298→    rec.start()
   299→  }

   694→/* Botón con tooltip ligero (title + aria) para el micrófono. */
   695→function TooltipishButton({
   696→  label,
   697→  disabled,
   698→  active,
   699→  onClick,
   700→  children,
   701→}: {
   702→  label: string
   703→  disabled?: boolean
   704→  active?: boolean
   705→  onClick: () => void
   706→  children: React.ReactNode
   707→}) {
   708→  return (
   709→    <button
   710→      type="button"
   711→      title={label}
   712→      aria-label={label}
   713→      aria-pressed={active}
   714→      disabled={disabled}
   715→      onClick={onClick}
   716→      className={cn(
   717→        'grid size-10 place-items-center rounded-full border transition-all',
   718→        active
   719→          ? 'border-emerald-500/40 bg-emerald-500/15'
   720→          : 'border-border/60 bg-secondary/40 hover:bg-accent',
   721→        disabled ? 'cursor-not-allowed opacity-40' : 'hover:scale-105',
   722→      )}
   723→    >
   724→      {children}
   725→    </button>
   726→  )
   727→}
